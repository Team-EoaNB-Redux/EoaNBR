# Banking and base money (M11). Money: 1,000 = GBP 1M. Only government *_k debt uses thousands of this unit.
# Borrower records are authoritative. bank_update recomputes the balance sheet; bank_monthly moves reserves
# and fixes the month's rate. bank_close_monthly settles income after ALL countries, independent of tick order.
# Approved household spending consumes capacity until drawn next month. Shocks restrict NEW credit.

    set_variable = { global.base_money_gdp_months = 12 }
    set_variable = { global.bank_initial_capital_months = 2 }
    set_variable = { global.bank_capital_requirement = 0.08 }
    set_variable = { global.bank_reserve_requirement = 0.2 }
    set_variable = { global.bank_reserve_recovery = 0.1 }

    set_variable = { global.bank_lending_spread = 4 }
    set_variable = { global.bank_scarcity_premium = 2 }
    set_variable = { global.bank_retained_share = 0.5 }
    set_variable = { global.bank_crisis_confidence_loss = 40 }
    set_variable = { global.bank_pop_debt_months = 24 }
    set_variable = { global.bank_pool_debt_months = 6 }

money_startup = {
    # All GDPs and markets exist. Initial money is a stock, not treasury revenue.
    every_country = {
        set_variable = { price_level = 1 }
        set_variable = { base_money = 0 }
        set_variable = { currency_reserves = 0 }
        set_variable = { currency_market = market_leader }
        set_variable = { currency_price_level = 1 }
    }
    every_country = {
        if = {
            limit = { NOT = { check_variable = { market_leader = 0 } } }
            add_to_variable = { market_leader:base_money = { value = gdp_amount max = 0 multiply = global.base_money_gdp_months } }
        }
    }
    money_shares_update = yes
    every_country = { bank_init = yes }
    every_state = { pops_update = yes }
    every_country = { update_country_pop_stats = yes national_market_states_update = yes }
}

money_shares_update = {
    # Global. Use current GDPs, not the market screen's potentially stale market_gdp.
    every_country = {
        set_variable = { money_market_gdp = 0 }
        set_variable = { money_market_emergency = 0 }
    }
    every_country = {
        if = {
            limit = { NOT = { check_variable = { market_leader = 0 } } }
            add_to_variable = { market_leader:money_market_gdp = { value = gdp_amount max = 0 } }
            add_to_variable = { market_leader:money_market_emergency = bank_central_bank_loan }
        }
    }
    every_country = {
        set_variable = { base_money_share = 0 }
        set_variable = { currency_reserves_share = 0 }
        set_variable = { bank_base_money_share = 0 }
        if = {
            limit = {
                check_variable = { market_leader:money_market_gdp > 0 }
                check_variable = { market_leader:price_level > 0 }
            }
            set_variable = { base_money_share = { value = market_leader:base_money divide = market_leader:money_market_gdp multiply = gdp_amount max = 0 } }
            set_variable = { currency_reserves_share = { value = market_leader:currency_reserves divide = market_leader:money_market_gdp multiply = gdp_amount max = 0 } }
            set_variable = { bank_base_money_share = { value = market_leader:base_money divide = market_leader:price_level subtract = market_leader:money_market_emergency max = 0 divide = market_leader:money_market_gdp multiply = gdp_amount max = 0 } }
        }
        set_variable = { currency_price_level = market_leader:price_level }
    }
}

money_transfer_market = {
    # Country. Membership changed; transfer the share cached BEFORE any membership changed (Q65, Q69).
    if = {
        limit = {
            has_variable = currency_market
            NOT = { check_variable = { currency_market = market_leader } }
            NOT = { check_variable = { market_leader = 0 } }
        }
        set_temp_variable = { money_old = currency_market }
        set_temp_variable = { money_new = market_leader }
        if = {
            limit = { NOT = { check_variable = { money_new:price_level > 0 } } }
            set_variable = { money_new:price_level = { value = currency_price_level max = 1 } }
        }
        set_temp_variable = { money_move = { value = base_money_share max = 0 min = money_old:base_money } }
        subtract_from_variable = { money_old:base_money = money_move }
        if = {
            limit = { check_variable = { currency_price_level > 0 } }
            set_temp_variable = { money_move = { value = money_move divide = currency_price_level multiply = money_new:price_level } }
        }
        add_to_variable = { money_new:base_money = money_move }
        set_temp_variable = { money_move = { value = currency_reserves_share max = 0 min = money_old:currency_reserves } }
        subtract_from_variable = { money_old:currency_reserves = money_move }
        add_to_variable = { money_new:currency_reserves = money_move } # Physical valuables, not currency
        set_variable = { currency_market = market_leader }
    }
}

bank_init = {
    set_variable = { bank_initialized = 1 }
    set_variable = { central_bank_interest_rate = global.bank_policy_rate }
    set_variable = { bank_capital = { value = gdp_amount max = 0 multiply = global.bank_initial_capital_months } }
    set_variable = { bank_central_bank_loan = 0 }
    set_variable = { bank_interest_received = 0 }
    set_variable = { bank_dividends = 0 }
    set_variable = { bank_defaults = 0 }
    set_variable = { bank_new_lending = 0 }
    set_variable = { bank_repayments = 0 }
    set_variable = { bank_reserves = { value = bank_base_money_share multiply = modifier@banking_confidence_modifier divide = 100 max = 0 min = bank_base_money_share } }
    bank_update = yes
    bank_rate_update = yes
}

bank_update = {
    # Idempotent. Ownership, not occupation, moves the household creditor (Q43, Q59).
    set_variable = { bank_loans_pops = 0 }
    set_variable = { bank_credit_committed = 0 }
    every_owned_state = {
        for_each_loop = {
            array = global.pops_array
            value = bank_pop_token
            add_to_variable = { PREV.bank_loans_pops = pops_debt@var:bank_pop_token }
            add_to_variable = { PREV.bank_credit_committed = pops_credit@var:bank_pop_token }
        }
    }
    set_variable = { bank_loans_sectors = 0 }
    for_each_loop = {
        array = sectors
        value = bank_sector
        add_to_variable = { bank_loans_sectors = sector_debt@var:bank_sector }
    }
    set_variable = { bank_loans_companies = 0 }
    for_each_loop = {
        array = companies
        value = bank_company
        add_to_variable = { bank_loans_companies = global.company_debt@var:bank_company }
    }
    set_variable = { bank_loans_foreign = 0 }
    for_each_loop = {
        array = money_lent_country
        value = bank_borrower
        if = {
            limit = { NOT = { check_variable = { bank_borrower = THIS } } }
            add_to_variable = { bank_loans_foreign = { value = money_lent_k@var:bank_borrower multiply = 1000 } }
        }
    }
    set_variable = { bank_loans = { value = bank_loans_pops add = bank_loans_sectors add = bank_loans_companies add = bank_loans_foreign add = investment_fund_debt } }
    bank_credit_update = yes
}

bank_credit_update = {
    # Cheap recompute after each transaction. Rates remain fixed throughout settlement.
    set_variable = { bank_reserve_requirement = { value = modifier@reserve_ratio_modifier clamp = { min = 0.01 max = 1 } } }
    if = {
        limit = { NOT = { check_variable = { modifier@reserve_ratio_modifier > 0 } } }
        set_variable = { bank_reserve_requirement = global.bank_reserve_requirement }
    }
    set_variable = { bank_capital_requirement = { value = global.bank_capital_requirement add = modifier@bank_capital_requirement_modifier clamp = { min = 0.01 max = 1 } } }
    set_variable = { bank_credit_capacity = { value = bank_reserves max = 0 divide = bank_reserve_requirement min = { value = bank_capital max = 0 divide = bank_capital_requirement } } }
    set_variable = { bank_credit_available = { value = bank_credit_capacity subtract = bank_loans subtract = bank_credit_committed max = 0 } }
    set_variable = { bank_money = bank_loans }
    set_variable = { bank_crisis = 0 }
    set_variable = { bank_status = 4 }
    if = {
        limit = { check_variable = { bank_capital < 0.00001 } }
        set_variable = { bank_crisis = 1 }
        set_variable = { bank_status = 0 }
    }
    else_if = {
        limit = { NOT = { check_variable = { bank_credit_available > 0 } } }
        set_variable = { bank_status = 1 }
    }
    set_variable = { bank_capital_ratio = 0 }
    set_variable = { bank_reserve_ratio = 0 }
    if = {
        limit = { check_variable = { bank_loans > 0 } }
        set_variable = { bank_capital_ratio = { value = bank_capital divide = bank_loans } }
        set_variable = { bank_reserve_ratio = { value = bank_reserves divide = bank_loans } }
    }
}

bank_rate_update = {
    set_temp_variable = { bank_usage = 1 }
    if = {
        limit = { check_variable = { bank_credit_capacity > 0 } }
        set_temp_variable = { bank_usage = { value = bank_loans add = bank_credit_committed divide = bank_credit_capacity clamp = { min = 0 max = 1 } } }
    }
    set_variable = { bank_interest_rate = { value = bank_usage multiply = global.bank_scarcity_premium add = central_bank_interest_rate add = global.bank_lending_spread add = modifier@interest_rate_modifier multiply = { value = 1 add = modifier@interest_rate_factor max = 0 } clamp = { min = 0 max = 30 } } }
}

bank_monthly = {
    if = { limit = { NOT = { has_variable = bank_initialized } } bank_init = yes }
    bank_update = yes
    set_variable = { bank_defaults = 0 }
    set_variable = { bank_new_lending = 0 }
    set_variable = { bank_repayments = 0 }
    set_variable = { bank_confidence = { value = bank_crisis multiply = global.bank_crisis_confidence_loss multiply = -1 add = modifier@banking_confidence_modifier add = bank_confidence_shock clamp = { min = 0 max = 100 } } }
    set_variable = { bank_reserve_target = { value = bank_base_money_share multiply = bank_confidence divide = 100 add = bank_central_bank_loan } }
    set_temp_variable = { bank_speed = global.bank_reserve_recovery }
    if = {
        limit = { check_variable = { bank_reserves > bank_reserve_target } }
        set_temp_variable = { bank_speed = global.bank_run_speed }
    }
    add_to_variable = { bank_reserves = { value = bank_reserve_target subtract = bank_reserves multiply = bank_speed } }
    bank_credit_update = yes
    bank_rate_update = yes
}

bank_borrow = {
    # Country (creditor). Input: bank_request. Output: bank_borrowed. Caller updates its borrower record.
    set_temp_variable = { bank_borrowed = { value = bank_request max = 0 min = bank_credit_available } }
    add_to_variable = { bank_loans = bank_borrowed }
    add_to_variable = { bank_new_lending = bank_borrowed }
    bank_credit_update = yes
}

bank_repay = {
    # Country (creditor). Input: bank_payment, capped by the caller to the borrower's principal.
    subtract_from_variable = { bank_loans = bank_payment }
    add_to_variable = { bank_repayments = bank_payment }
    bank_credit_update = yes
}

bank_write_off = {
    # Country (creditor). Input: bank_loss, already removed from the borrower. Losses hit capital, not reserves.
    subtract_from_variable = { bank_loans = bank_loss }
    subtract_from_variable = { bank_capital = bank_loss }
    add_to_variable = { bank_defaults = bank_loss }
    bank_credit_update = yes
}

bank_close_monthly = {
    # Only interest received is income. Liquidity assistance is repayable and cannot recapitalise an insolvent bank.
    set_variable = { bank_interest_income = bank_interest_received }
    set_variable = { bank_interest_received = 0 }
    set_temp_variable = { bank_cb_interest = { value = bank_central_bank_loan multiply = central_bank_interest_rate divide = 1200 } }
    subtract_from_variable = { bank_interest_income = bank_cb_interest }
    add_to_variable = { market_leader:treasury = bank_cb_interest }
    set_variable = { bank_dividends = 0 }
    add_to_variable = { bank_capital = bank_interest_income }
    set_temp_variable = { bank_required_capital = { value = bank_loans add = bank_credit_committed multiply = bank_capital_requirement } }
    if = {
        limit = { check_variable = { bank_interest_income > 0 } }
        set_variable = { bank_dividends = { value = 1 subtract = global.bank_retained_share multiply = bank_interest_income min = { value = bank_capital subtract = bank_required_capital max = 0 } } }
        subtract_from_variable = { bank_capital = bank_dividends }
    }
    set_temp_variable = { bank_cb_repay = { value = bank_loans add = bank_credit_committed multiply = bank_reserve_requirement multiply = -1 add = bank_reserves max = 0 min = { value = bank_central_bank_loan multiply = global.bank_emergency_repayment } } }
    subtract_from_variable = { bank_central_bank_loan = bank_cb_repay }
    subtract_from_variable = { bank_reserves = bank_cb_repay }
    subtract_from_variable = { market_leader:base_money = { value = bank_cb_repay multiply = market_leader:price_level } }
    bank_update = yes
}

bank_foreign_interest_update = {
    # Borrower country. Idempotent quotes in real money, with annual percent rates per creditor.
    set_variable = { foreign_debt_interest = 0 }
    set_variable = { credit_rating_interest = { value = debt_to_gdp_ratio divide = 25 add = global.base_international_interest_rate clamp = { min = 2 max = 10 } } }
    for_each_loop = {
        array = debt_to_bank_country
        value = fi_creditor
        if = {
            limit = { NOT = { check_variable = { fi_creditor = THIS } } }
            set_variable = { foreign_interest_rate@var:fi_creditor = 0 }
            if = {
                limit = { check_variable = { global.eco_debt_enabled = 1 } }
                set_variable = { foreign_interest_rate@var:fi_creditor = { value = fi_creditor:bank_interest_rate add = credit_rating_interest add = modifier@foreign_interest_rate_modifier multiply = { value = 1 add = modifier@foreign_interest_rate_factor max = 0 } max = 0 } }
            }
            set_variable = { foreign_interest@var:fi_creditor = { value = debt_to_bank_k@var:fi_creditor multiply = foreign_interest_rate@var:fi_creditor divide = 1.2 } }
            add_to_variable = { foreign_debt_interest = foreign_interest@var:fi_creditor }
        }
    }
    set_variable = { foreign_interest_rate = 0 }
    if = {
        limit = { check_variable = { debt_to_foreign_countries_k > 0 } }
        set_variable = { foreign_interest_rate = { value = foreign_debt_interest multiply = 1.2 divide = debt_to_foreign_countries_k } }
    }
}

bank_foreign_interest_monthly = {
    # The budget just charged these quotes. A recompute must never pay interest twice.
    for_each_loop = {
        array = debt_to_bank_country
        value = fi_creditor
        if = {
            limit = { NOT = { check_variable = { fi_creditor = THIS } } }
            add_to_variable = { fi_creditor:bank_interest_received = foreign_interest@var:fi_creditor }
        }
    }
}

# Construction is paid before it advances, including its final month. A refused payment pauses only that project.
# Quote all owners of a shared upgrade before charging any of them (Q133).
bank_project_quote = {
    # Country. Inputs: bc_financer (0 generic, otherwise company id), bc_building, bc_cost.
    # Outputs: bc_cash, bc_room, bc_shortfall, bc_ok. No mutation.
    if = {
        limit = { check_variable = { bc_financer = 0 } }
        set_temp_variable = { bc_cash = { value = sector_capital@var:bc_building max = 0 add = sector_construction_reserve@var:bc_building } }
        set_temp_variable = { bc_room = { value = sector_asset_value@var:bc_building multiply = global.sector_debt_limit subtract = sector_debt@var:bc_building max = 0 } }
    }
    else = {
        set_temp_variable = { bc_cash = { value = global.company_treasury@var:bc_financer max = 0 } }
        set_temp_variable = { bc_room = { value = global.company_asset_value@var:bc_financer multiply = global.sector_debt_limit subtract = global.company_debt@var:bc_financer max = 0 } }
    }
    set_temp_variable = { bc_shortfall = { value = bc_cost subtract = bc_cash max = 0 } }
    set_temp_variable = { bc_ok = 1 }
    if = {
        limit = {
            OR = {
                check_variable = { bc_shortfall > bc_room }
                AND = { check_variable = { bc_financer = 0 } check_variable = { sector_construction_paused@var:bc_building = 1 } }
                AND = { check_variable = { bc_financer > 0 } check_variable = { global.company_construction_paused@var:bc_financer = 1 } }
            }
        }
        set_temp_variable = { bc_ok = 0 }
    }
}

bank_project_pay = {
    # Country. Same inputs as bank_project_quote. Called only after the entire project is affordable.
    bank_project_quote = yes
    set_temp_variable = { bank_request = bc_shortfall }
    bank_borrow = yes
    if = {
        limit = { check_variable = { bc_financer = 0 } }
        add_to_variable = { sector_debt@var:bc_building = bank_borrowed }
        set_temp_variable = { bc_paid = { value = bc_cost min = sector_construction_reserve@var:bc_building max = 0 } }
        subtract_from_variable = { sector_construction_reserve@var:bc_building = bc_paid }
        subtract_from_variable = { sector_capital@var:bc_building = { value = bc_cost subtract = bc_paid subtract = bank_borrowed max = 0 } }
    }
    else = {
        add_to_variable = { global.company_debt@var:bc_financer = bank_borrowed }
        subtract_from_variable = { global.company_treasury@var:bc_financer = { value = bc_cost subtract = bank_borrowed max = 0 } }
    }
}

bank_construction_monthly = {
    # State, before on_monthly_industries. Prices have moved; price the bill before deciding whether to buy its goods.
    for_each_loop = {
        array = industries
        value = bc_building
        if = {
            limit = {
                OR = {
                    check_variable = { construction@var:bc_building > 0 }
                    NOT = { check_variable = { retooling_pm@var:bc_building = 0 } }
                }
            }
            set_variable = { construction_credit_paused@var:bc_building = 0 }
            set_variable = { retooling_credit_paused@var:bc_building = 0 }
            set_temp_variable = { bank_quote_construction = 1 }
            set_temp_variable = { iuc_industry = bc_building }
            industry_update_construction_retooling = yes
            set_temp_variable = { bank_quote_construction = 0 }
            set_temp_variable = { bc_financer = construction_financier@var:bc_building }
            if = {
                limit = {
                    check_variable = { construction@var:bc_building > 0 }
                    NOT = { check_variable = { bc_financer > 10500 } }
                }
                set_temp_variable = { bc_cost = construction_cost@var:bc_building }
                controller = {
                    bank_project_quote = yes
                    if = {
                        limit = { check_variable = { bc_shortfall > bank_credit_available } }
                        set_temp_variable = { bc_ok = 0 }
                    }
                    if = { limit = { check_variable = { bc_ok = 1 } } bank_project_pay = yes }
                }
                if = { limit = { check_variable = { bc_ok = 0 } } set_variable = { construction_credit_paused@var:bc_building = 1 } }
            }
            if = {
                limit = {
                    NOT = { check_variable = { retooling_pm@var:bc_building = 0 } }
                    check_variable = { retooling_financier@var:bc_building > 0 }
                    NOT = { check_variable = { retooling_financier@var:bc_building > 10500 } }
                }
                set_temp_variable = { bc_financer = retooling_financier@var:bc_building }
                set_temp_variable = { bc_cost = retooling_cost@var:bc_building }
                controller = {
                    bank_project_quote = yes
                    if = {
                        limit = { check_variable = { bc_shortfall > bank_credit_available } }
                        set_temp_variable = { bc_ok = 0 }
                    }
                    if = { limit = { check_variable = { bc_ok = 1 } } bank_project_pay = yes }
                }
                if = { limit = { check_variable = { bc_ok = 0 } } set_variable = { retooling_credit_paused@var:bc_building = 1 } }
            }
            if = {
                limit = {
                    NOT = { check_variable = { retooling_pm@var:bc_building = 0 } }
                    check_variable = { retooling_financier@var:bc_building = 0 }
                    check_variable = { size@var:bc_building > 0 }
                }
                set_temp_variable = { bc_total = retooling_cost@var:bc_building }
                set_temp_variable = { bc_generic_share = 1 }
                set_temp_variable = { bc_credit = 0 }
                set_temp_variable = { bc_all_ok = 1 }
                for_each_loop = {
                    array = owner_companies@var:bc_building
                    value = bc_financer
                    index = bc_idx
                    if = {
                        limit = { check_variable = { global.company_owner@var:bc_financer = controller } }
                        set_temp_variable = { bc_share = owner_sizes@var:bc_building^bc_idx }
                        divide_temp_variable = { bc_share = size@var:bc_building }
                        subtract_from_temp_variable = { bc_generic_share = bc_share }
                        set_temp_variable = { bc_cost = { value = bc_total multiply = bc_share } }
                        controller = { bank_project_quote = yes }
                        add_to_temp_variable = { bc_credit = bc_shortfall }
                        if = { limit = { check_variable = { bc_ok = 0 } } set_temp_variable = { bc_all_ok = 0 } }
                    }
                }
                set_temp_variable = { bc_financer = 0 }
                set_temp_variable = { bc_cost = { value = bc_total multiply = bc_generic_share max = 0 } }
                controller = { bank_project_quote = yes }
                add_to_temp_variable = { bc_credit = bc_shortfall }
                if = {
                    limit = {
                        OR = {
                            check_variable = { bc_ok = 0 }
                            check_variable = { bc_credit > controller:bank_credit_available }
                        }
                    }
                    set_temp_variable = { bc_all_ok = 0 }
                }
                if = {
                    limit = { check_variable = { bc_all_ok = 1 } }
                    controller = { bank_project_pay = yes }
                    for_each_loop = {
                        array = owner_companies@var:bc_building
                        value = bc_financer
                        index = bc_idx
                        if = {
                            limit = { check_variable = { global.company_owner@var:bc_financer = controller } }
                            set_temp_variable = { bc_share = owner_sizes@var:bc_building^bc_idx }
                            set_temp_variable = { bc_cost = { value = bc_share divide = size@var:bc_building multiply = bc_total } }
                            controller = { bank_project_pay = yes }
                        }
                    }
                }
                else = { set_variable = { retooling_credit_paused@var:bc_building = 1 } }
            }
        }
    }
}

bank_pool_monthly = {
    # Country. Pool income services its own debt before any new funding. It does not hide debt in sectors.
    set_variable = { investment_fund_interest = { value = investment_fund_debt multiply = bank_interest_rate divide = 1200 } }
    set_temp_variable = { pool_paid = { value = investment_fund max = 0 min = investment_fund_interest } }
    subtract_from_variable = { investment_fund = pool_paid }
    add_to_variable = { bank_interest_received = pool_paid }
    set_temp_variable = { bank_payment = { value = investment_fund_income multiply = global.bank_pool_repayment_share max = 0 min = investment_fund_debt min = { value = investment_fund max = 0 } } }
    subtract_from_variable = { investment_fund = bank_payment }
    subtract_from_variable = { investment_fund_debt = bank_payment }
    bank_repay = yes
    # A persistent loss of contributions writes off the unserviceable portion, without destroying plants.
    set_temp_variable = { pool_limit = { value = investment_fund_income max = 0 multiply = global.bank_pool_debt_months } }
    if = {
        limit = { check_variable = { investment_fund_debt > pool_limit } }
        set_temp_variable = { bank_loss = { value = investment_fund_debt subtract = pool_limit } }
        set_variable = { investment_fund_debt = pool_limit }
        bank_write_off = yes
    }
    set_variable = { investment_funded_credit = 0 }
}

bank_pool_credit = {
    # Country, within a tier after companies and sector capital. inv_pool_need_temp is unmet capital cost.
    # Borrow for investment only when the tier's ROI clears the lending rate and its money is not held by Q126.
    if = {
        limit = {
            NOT = { check_variable = { inv_t > global.bank_credit_max_tier } }
            NOT = { check_variable = { inv_pool_held_temp > 0.01 } }
        }
        set_temp_variable = { pool_roi = global.investment_tier_roi^inv_ti }
        set_temp_variable = { pool_rate = { value = bank_interest_rate divide = 100 } }
        if = {
            limit = { check_variable = { pool_roi > pool_rate } }
            set_temp_variable = { bank_request = { value = inv_pool_need_temp multiply = inv_full_temp subtract = investment_fund max = 0 min = { value = investment_fund_income multiply = global.bank_pool_debt_months subtract = investment_fund_debt max = 0 } } }
            bank_borrow = yes
            add_to_variable = { investment_fund_debt = bank_borrowed }
            add_to_variable = { investment_fund = bank_borrowed }
            add_to_variable = { investment_funded_credit = bank_borrowed }
        }
    }
}

pops_credit_spending_update = {
    # State, p_pops_* context. Replace this class's reservation: recomputing never reserves twice.
    set_temp_variable = { p_credit_ration = 1 }
    set_temp_variable = { p_old_credit = pops_credit@var:current_pop_token }
    set_temp_variable = { p_new_credit = 0 }
    set_temp_variable = { p_unfunded = 0 }
    if = {
        limit = {
            check_variable = { global.pops_borrows@var:current_pop_token = 1 }
            check_variable = { owner:bank_initialized = 1 }
        }
        set_temp_variable = { p_credit_wanted = { value = p_pops_spending add = p_pops_investment subtract = p_pops_earnings max = 0 } }
        set_temp_variable = { p_credit_room = { value = p_pops_earnings subtract = p_pops_income_tax_spending max = 0 multiply = global.bank_pop_debt_months subtract = pops_debt@var:current_pop_token max = 0 } }
        set_temp_variable = { p_new_credit = { value = owner:bank_credit_available add = p_old_credit max = 0 min = p_credit_wanted min = p_credit_room } }
        set_temp_variable = { p_unfunded = { value = p_credit_wanted subtract = p_new_credit max = 0 } }
        set_temp_variable = { p_goods_cost = { value = p_pops_needs_spending add = p_pops_sales_tax_spending } }
        if = {
            limit = { check_variable = { p_goods_cost > 0 } }
            set_temp_variable = { p_credit_ration = { value = p_goods_cost subtract = p_unfunded max = 0 divide = p_goods_cost min = 1 } }
            set_temp_variable = { p_unfunded = { value = p_goods_cost min = p_unfunded } }
            subtract_from_temp_variable = { p_pops_spending = p_unfunded }
            multiply_temp_variable = { p_pops_needs_spending = p_credit_ration }
            multiply_temp_variable = { p_pops_sales_tax_spending = p_credit_ration }
            multiply_temp_variable = { p_pops_col = p_credit_ration }
        }
        owner = {
            subtract_from_variable = { bank_credit_committed = p_old_credit }
            add_to_variable = { bank_credit_committed = p_new_credit }
            bank_credit_update = yes
        }
    }
    set_variable = { pops_credit@var:current_pop_token = p_new_credit }
    set_variable = { pops_unfunded_spending@var:current_pop_token = p_unfunded }
    # The earlier get_pop_needs call quoted prices; add only the financed quantities to demand.
    set_temp_variable = { wealth_temp = pops_wealth@var:current_pop_token }
    set_temp_variable = { pop_temp = { value = p_pops multiply = p_credit_ration } }
    set_temp_variable = { market_share_temp = 1 }
    if = { limit = { check_variable = { current_pop_token = token:peasants } } set_temp_variable = { market_share_temp = global.peasant_market_interaction } }
    get_pop_needs = yes
}
