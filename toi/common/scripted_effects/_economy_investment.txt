        set_temp_variable = { ioprune_idx_temp = active_investment_opportunities^num }
        while_loop_effect = {
            limit = { check_variable = { ioprune_idx_temp > 0 } }
            subtract_from_temp_variable = { ioprune_idx_temp = 1 }
            set_temp_variable = { io_num_temp = active_investment_opportunities^ioprune_idx_temp }
            if = {
                limit = { check_variable = { investment_opportunity_attractiveness@var:io_num_temp < 0 } }
                if = {
                    limit = { check_variable = { investment_opportunity_investment@var:io_num_temp > 0 } }
                    add_to_variable = { investment_fund = investment_opportunity_investment@var:io_num_temp }
                }
                remove_investment_opportunity = yes
            }
        }
        while_loop_effect = {
            limit = {
                check_variable = { active_investment_opportunities^num > investment_opportunities_cap }
            }
            remove_worst_investment_opportunity = yes
        }

    investment_opportunities_sort = {
        
        clear_array = investment_opportunities_sort
        for_each_loop = {
            array = investment_opportunities
            value = io_num_temp
            
            set_temp_variable = { effective_wage_om = investment_opportunity_attractiveness@var:io_num_temp }

            for_each_loop = {
                array = investment_opportunities_sort
                index = ii
                value = vv
                break = bb

                set_temp_variable = { effective_wage_vv = investment_opportunity_attractiveness@var:vv }

                if = {
                    limit = {
                        check_variable = { effective_wage_vv < effective_wage_om }
                    }
                    add_to_array = { 
                        array = investment_opportunities_sort 
                        value = io_num_temp
                        index = ii
                    }
                    set_temp_variable = { bb = 1 }
                }
            }
            if = {
                limit = {
                    check_variable = { bb = 0 }
                }
                add_to_array = { investment_opportunities_sort = io_num_temp }
            }
        }
        
        clear_array = investment_opportunities
        for_each_loop = {
            array = investment_opportunities_sort
            value = io_num_temp
            add_to_array = { investment_opportunities = io_num_temp }
        }
        clear_array = investment_opportunities_sort
    }

    investment_opportunities_cap_calc = {

        if = {

            }
            set_variable = { investment_opportunities_cap = global.max_landowners_investment_opportunities }

        divide_temp_variable = { cap_temp = global.capitalists_per_investment_opportunity }
        add_to_variable = { investment_opportunities_cap = cap_temp }
        add_to_variable = { investment_opportunities_cap = global.base_investment_opportunities_cap }
        round_variable = investment_opportunities_cap

        # Monthly
        set_variable = { investment_opportunities_monthly = investment_opportunities_cap }
        multiply_variable = { investment_opportunities_monthly = 0.5 }
        add_to_variable = { investment_opportunities_monthly = 0.5 }
        round_variable = investment_opportunities_monthly

# IO Generation
    create_investment_opportunity = {
        # Tries 10 times to generate an investment opportunity. For when you want to ensure the country gets one
        set_temp_variable = { investment_opportunities_amount_temp = investment_opportunities^num }
        set_temp_variable = { tries_temp = 0 }
        while_loop_effect = {
            limit = {
                check_variable = { tries_temp < 10 }
                check_variable = { investment_opportunities_amount_temp = investment_opportunities^num }
            }
            generate_investment_opportunity = yes
            add_to_temp_variable = { tries_temp = 1 }
        }
    }
    generate_investment_opportunity = {
        clr_country_flag = investment_opportunity_done
        if = {
            limit = {
                check_variable = { investment_opportunities^num < investment_opportunities_cap }
            }
            # BUILD/EXPAND industrial capacity
            generate_investment_opportunity_railway = yes

            if = { # BUILD a NEW Building
                limit = {
                    NOT = { has_country_flag = investment_opportunity_done } # Nothing else was done
                    OR = {
                        AND = {
                            any_controlled_state = {
                                check_variable = { buildings^num = 0 } # DOESN'T have a building
                                check_variable = { available_workforce > 20 } # Has more than enough to fill a building
                                NOT = { has_state_flag = current_investment_opportunity_flag } # State doesn't have an IO for it.
                            }
                            check_variable = { random < 0.1 }
                        }
                        check_variable = { random < 0.05 }
                    }
                }
                # log = "Attemping BUILD"
                generate_investment_opportunity_build = yes
            }

            if = { # UPGRADE a CURRENT building 
                limit = { 
                    NOT = { has_country_flag = investment_opportunity_done } # Nothing else was done
                    check_variable = { random < 0.5 }
                }
                if = { # Upgrade PM
                    limit = {
                        check_variable = { industries_pm_upgradable^num > 0 }
                        check_variable = { random < 0.5 }
                    }
                    # log = "Attemping UPGRADE PM"
                    generate_investment_opportunity_upgrade = yes
                }
                else_if = { # Upgrade AM
                    limit = {
                        check_variable = { industries_am_upgradable^num > 0 }
                    }
                    # log = "Attemping UPGRADE AM"
                    generate_investment_opportunity_upgrade_automation = yes
                }
            }

            if = { # EXPAND a CURRENT building 
                limit = { 
                    NOT = { has_country_flag = investment_opportunity_done } # Nothing else was done
                }
                # log = "Attemping EXPAND"
                generate_investment_opportunity_expand = yes
            }

            add_investment_opportunity = yes
        }
    }
    generate_investment_opportunity_railway = {
        set_temp_variable = { industry_token = token:railway }
        if = {
            limit = {
                is_valid_industry = yes
            }
            # Selects a semi-random state for infrastructure
            # Outputs chosen_state

            set_temp_variable = { chosen_state = 0 }
            random_controlled_state = {
                limit = {
                    set_temp_variable = { industrial_cap_temp = industrial_capacity_use }
                    multiply_temp_variable = { industrial_cap_temp = 1.05 }
                    check_variable = { industrial_cap_temp > industrial_capacity } # Either Lacks industrial capacity or is getting close
                    NOT = { check_variable = { construction@token:railway > 0 } } # Currently not building a railway
                    NOT = {
                        any_of = {
                            array = PREV.investment_opportunities
                            value = io_num_temp
                            check_variable = { PREV.investment_opportunity_building@var:io_num_temp = industry_token }
                            check_variable = { PREV.investment_opportunity_state@var:io_num_temp = THIS }
                        }

    io_state_list = {
        # Regenerates the list of states for potential investment (in order of workforce, modified by gdp/c & other factors)

        for_each_loop = {
            array = investment_states_array
            value = vv
            clear_variable = investment_weight@var:vv
        }
        set_variable = { investment_weight_total = 0 }
        clear_array = investment_states_array
        every_controlled_state = {

            set_temp_variable = { investment_weight_temp = pops_workforce } # Population is base

            # Non-core states get reduced by 90%
            if = { limit = { NOT = { is_core_of = PREV } } multiply_temp_variable = { investment_weight_temp = 0.1 } }
            # Non-owned states get reduced by 90%
            if = { limit = { NOT = { is_owned_by = PREV } } multiply_temp_variable = { investment_weight_temp = 0.1 } }

            # Wealthy states get invested in more often
            set_temp_variable = { gdp_temp = gdp_per_capita }
            if = { limit = { check_variable = { gdp_temp < 0.1 } } set_temp_variable = { gdp_temp = 0.1 } }
            divide_temp_variable = { gdp_temp = 10 }
            multiply_temp_variable = { investment_weight_temp = gdp_temp }
            
            set_variable = { PREV.investment_weight@THIS = investment_weight_temp }
            add_to_variable = { PREV.investment_weight_total = investment_weight_temp }

            # Add to investment_states_array sorted
            for_each_loop = {
                array = PREV.investment_states_array
                index = ii
                value = vv
                break = bb

                if = {
                    limit = {
                        check_variable = { PREV.investment_weight@var:vv < investment_weight_temp }
                    }
                    add_to_array = { 
                        array = PREV.investment_states_array 
                        value = THIS
                        index = ii
                    }
                    set_temp_variable = { bb = 1 }
                }
            }
            if = {
                limit = {
                    check_variable = { bb = 0 }
                }
                add_to_array = { PREV.investment_states_array = THIS }
            }
        }
    }

	io_add_random = {
		# Selects one random IO to add to the active list
		# If what it selects is already in the active list, then it "walks" the list until it finds something
		# needed_temp

		clamp_temp_variable = { var = needed_temp min = 1 max = 1000 }

		set_temp_variable = { attempts_temp = 0 }
		while_loop_effect = {
			limit = { 
				check_variable = { attempts_temp < 10 }
				check_variable = { needed_temp > 0 }
			}

			# Pick random starting point
			set_temp_variable = { start_idx = investment_opportunities^num }
			multiply_temp_variable = { start_idx = random }
			multiply_temp_variable = { start_idx = random } # Weighted to top of list
			round_temp_variable = start_idx

			# Walk from that point
			for_loop_effect = {
				start = 0
				end = investment_opportunities^num
				value = offset
				break = io_brk

				set_temp_variable = { current_index = start_idx }
				add_to_temp_variable = { current_index = offset }
				if = {
					limit = {
						NOT = { check_variable = { current_index < investment_opportunities^num } }
					}
					subtract_from_temp_variable = { current_index = investment_opportunities^num }
				}

				set_temp_variable = { candidate_io = investment_opportunities^current_index }

				# Check if active
				if = {
					limit = {
						NOT = { is_in_array = { active_investment_opportunities = candidate_io } }
					}
					add_to_array = { active_investment_opportunities = candidate_io }
					set_temp_variable = { io_brk = 1 }
					subtract_from_temp_variable = { needed_temp = 1 }
					set_temp_variable = { attempts_temp = 0 }
				}
			}

			add_to_temp_variable = { attempts_temp = 1 }
		}
	}

	score_io_attractiveness = {
		# State Scope
		# industry_token, io_type

		# Base
		# * ROI (calculated at country level where 10% = +-0.0 - unless it's prospecting where it's based on price)
		# + Available Inputs (TODO)  
		# + Economies of Scale
		# +- Price of Output
		# +- Infra (TODO)
		# +- Labor (TODO)
		# + Other Modifiers
		# * Other Factors

		set_temp_variable = { attr_base_temp = 100.0 }

		# Return on Investment (Every 10 years)
		# If roi_temp > 1.0, that means it's a good investment (Full ROI < 10 Years)
		if = {
			limit = {
				check_variable = { io_type = token:io_upgrade_pm }
			}
			# ROI = ((New Revenue - New Expenses) - (Current Revenue - Current Expenses)) / Retooling Cost
			# Note: Upgrading to highest prod PM
			set_temp_variable = { current_am_temp = automation_method@var:industry_token }
			set_temp_variable = { am_mult_temp = controller:base_am_wage_mult@var:current_am_temp }
			multiply_temp_variable = { am_mult_temp = global.building_base_employees@var:industry_token }
			set_temp_variable = { am_expenses_temp = controller:base_am_expenses@var:current_am_temp }
			multiply_temp_variable = { am_expenses_temp = global.building_base_output_value@var:industry_token }

			set_temp_variable = { roi_temp = controller:highest_productivity_revenue@var:industry_token }

			set_temp_variable = { expenses_temp = controller:highest_productivity_wage_mult@var:industry_token }
			add_to_temp_variable = { expenses_temp = am_mult_temp }
			multiply_temp_variable = { expenses_temp = base_wage }
			add_to_temp_variable = { expenses_temp = controller:highest_productivity_expenses@var:industry_token }
			add_to_temp_variable = { expenses_temp = am_expenses_temp }
			subtract_from_temp_variable = { roi_temp = expenses_temp } # Profit
			
			set_temp_variable = { old_profit_temp = productivity@var:industry_token }
			multiply_temp_variable = { old_profit_temp = payroll_multiplier@var:industry_token }
			set_temp_variable = { old_payroll_temp = payroll_multiplier@var:industry_token }
			multiply_temp_variable = { old_payroll_temp = base_wage }
			subtract_from_temp_variable = { old_profit_temp = old_payroll_temp }

			subtract_from_temp_variable = { roi_temp = old_profit_temp }

			calc_io_cost = yes # output: total_cost_temp
			divide_temp_variable = { roi_temp = total_cost_temp } # Divide by Total Cost
		
			multiply_temp_variable = { roi_temp = 0.67 } # What's left after bonuses
		}
		else_if = {
			limit = {
				check_variable = { io_type = token:io_upgrade_am }
			}

			set_temp_variable = { throughput_modifier_temp = controller:industry_throughput_factor@var:industry_token }
			add_to_temp_variable = { throughput_modifier_temp = 1 }

			set_temp_variable = { input_modifier_temp = controller:industry_input_factor@var:industry_token }
			add_to_temp_variable = { input_modifier_temp = 1 }
			multiply_temp_variable = { input_modifier_temp = throughput_modifier_temp }

			# Get Curr & Next PM
			set_temp_variable = { pm_curr_temp = automation_method@var:industry_token }
			set_temp_variable = { i_temp = 0 }
			for_each_loop = {
				array = global.building_automation_methods@var:industry_token
				value = am_check_temp
				index = pm_i
				break = pm_brk
				if = { limit = { check_variable = { am_check_temp = automation_method@var:industry_token } } set_temp_variable = { i_temp = pm_i } set_temp_variable = { pm_brk = 1 } }
			}
			add_to_temp_variable = { i_temp = 1 }
			set_temp_variable = { pm_next_temp = global.building_automation_methods@var:industry_token^i_temp }

			# Calculation
			set_temp_variable = { curr_expenses_temp = global.building_base_output_value@var:industry_token }
			multiply_temp_variable = { curr_expenses_temp = input_modifier_temp }
			set_temp_variable = { next_expenses_temp = curr_expenses_temp }
			
			multiply_temp_variable = { curr_expenses_temp = controller:base_am_expenses@var:pm_curr_temp }
			multiply_temp_variable = { next_expenses_temp = controller:base_am_expenses@var:pm_next_temp }

			set_temp_variable = { curr_wage = global.building_base_employees@var:industry_token }
			multiply_temp_variable = { curr_wage = base_wage }
			set_temp_variable = { next_wage = curr_wage }

			multiply_temp_variable = { curr_wage = controller:base_am_wage_mult@var:pm_curr_temp }
			multiply_temp_variable = { next_wage = controller:base_am_wage_mult@var:pm_next_temp }

			set_temp_variable = { roi_temp = curr_expenses_temp }
			subtract_from_temp_variable = { roi_temp = next_expenses_temp }
			add_to_temp_variable = { roi_temp = curr_wage }
			subtract_from_temp_variable = { roi_temp = next_wage }
			
			set_temp_variable = { pm_temp = pm_next_temp }
			calc_io_cost = yes # output: total_cost_temp
			divide_temp_variable = { roi_temp = total_cost_temp } # Divide by Total Cost
		
			multiply_temp_variable = { roi_temp = 0.67 } # What's left after bonuses
		}
		else = {
			if = {
				limit = {
					check_variable = { size@var:industry_token > 0 }
				}
				set_temp_variable = { roi_temp = profit@var:industry_token }
				divide_temp_variable = { roi_temp = size@var:industry_token }
			}
			else = {
				set_temp_variable = { latest_am_temp = controller:latest_automation_method@var:industry_token }
				set_temp_variable = { am_mult_temp = controller:base_am_wage_mult@var:latest_am_temp }
				multiply_temp_variable = { am_mult_temp = global.building_base_employees@var:industry_token }
				set_temp_variable = { am_expenses_temp = controller:base_am_expenses@var:latest_am_temp }
				multiply_temp_variable = { am_expenses_temp = global.building_base_output_value@var:industry_token }

				set_temp_variable = { roi_temp = controller:highest_productivity_revenue@var:industry_token }
				set_temp_variable = { expenses_temp = controller:highest_productivity_wage_mult@var:industry_token }
				add_to_temp_variable = { expenses_temp = am_mult_temp }
				multiply_temp_variable = { expenses_temp = base_wage }
				add_to_temp_variable = { expenses_temp = controller:highest_productivity_expenses@var:industry_token }
				add_to_temp_variable = { expenses_temp = am_expenses_temp }
				subtract_from_temp_variable = { roi_temp = expenses_temp } # Profit
		
				multiply_temp_variable = { roi_temp = 0.67 } # What's left after bonuses
			}

			calc_io_cost = yes # output: total_cost_temp
			
			if = {
				limit = {
					check_variable = { io_type = token:io_prospect }
				}
				set_temp_variable = { number_resources_expected = prospecting_high@var:industry_token }
				add_to_temp_variable = { number_resources_expected = prospecting_low@var:industry_token }
				divide_temp_variable = { number_resources_expected = 2 }
				
				set_temp_variable = { ultimate_total_temp = prospecting_cost@var:industry_token }
				divide_temp_variable = { ultimate_total_temp = number_resources_expected }
				add_to_temp_variable = { ultimate_total_temp = total_cost_temp }
				divide_temp_variable = { roi_temp = ultimate_total_temp}
			}
			else = {
				divide_temp_variable = { roi_temp = total_cost_temp } # Divide by Total Cost
			}

		}
		multiply_temp_variable = { roi_temp = 120 } # monthly roi * 12 months / 10%

    check_investment_opportunity = {
        # Checks if IO is still valid. If not, removes it (and refunds money if any was put in)
        # io_num_temp

        set_temp_variable = { io_type = investment_opportunity_type@var:io_num_temp }
        set_temp_variable = { state_temp = investment_opportunity_state@var:io_num_temp }
        set_temp_variable = { industry_token = investment_opportunity_building@var:io_num_temp }

        set_temp_variable = { io_invalid_temp = 0 }

        # An IO becomes invalid if its target is already being built/upgraded by someone else (e.g. the player
        # built it directly, or a company started it). Without this the fund would "adopt" that construction and
        # treat the other party's work as its own investment.
        var:state_temp = {
            if = {
                limit = {
                    OR = {
                        check_variable = { io_type = token:io_build }
                        check_variable = { io_type = token:io_expand }
                        check_variable = { io_type = token:io_exploit }
                    }
                    NOT = { check_variable = { construction@var:industry_token = 0 } }
                }
                set_temp_variable = { io_invalid_temp = 1 }
            }
            else_if = {
                limit = {
                    OR = {
                        check_variable = { io_type = token:io_upgrade_pm }
                        check_variable = { io_type = token:io_upgrade_am }
                    }
                    NOT = { check_variable = { retooling_pm@var:industry_token = 0 } }
                }
                set_temp_variable = { io_invalid_temp = 1 }
            }
        }

        if = {
            limit = { check_variable = { io_invalid_temp = 1 } }
            # Refund any money already committed to the fund before dropping the opportunity.
            if = {
                limit = { check_variable = { investment_opportunity_investment@var:io_num_temp > 0 } }
                add_to_variable = { investment_fund = investment_opportunity_investment@var:io_num_temp }
            }
            remove_investment_opportunity = yes
        }
    }

        set_variable = { investment_fund_spending_corporate_tax = 0 }
        set_variable = { investment_fund_spending_construction = 0 }

            add_to_variable = { PREV.investment_fund_spending_corporate_tax = pops_investment_tax@token:landowners }

                array = industries
                index = sp_i
                value = sp_token

                #Construction
                if = {
                    limit = {
                        check_variable = { construction@var:sp_token > 0 }
                        check_variable = { construction_financier@var:sp_token = 0 }
                    }
                    add_to_variable = { PREV.investment_fund_spending_construction = construction_cost@var:sp_token }
                }
                if = {
                    limit = {
                        NOT = { check_variable = { retooling_pm@var:sp_token = 0 } }
                        check_variable = { retooling_financier@var:sp_token = 0 }
                    }
                    add_to_variable = { PREV.investment_fund_spending_retooling = retooling_cost@var:sp_token }
                }
            }
        }

        # Income
        set_variable = { investment_fund_income = pops_investment }

        # Debt
        set_variable = { investment_fund_spending_interest = 0 }

        # Spending
        set_variable = { investment_fund_target_spending = {
			value = modifier@investment_confidence_modifier
			divide = 100
			max = 0
			multiply = investment_fund_income

			add = {
				value = gdp_amount
				multiply = 0.1
				min = investment_fund
				multiply = 0.2
			}
		} }

        set_variable = { investment_fund_spending = investment_fund_spending_construction }
        add_to_variable = { investment_fund_spending = investment_fund_spending_retooling }
        add_to_variable = { investment_fund_spending = investment_fund_spending_interest }
        add_to_variable = { investment_fund_spending = investment_fund_spending_corporate_tax }

        # Balance
        set_variable = { investment_fund_balance = investment_fund_income }
        subtract_from_variable = { investment_fund_balance = investment_fund_spending }
    }
    investment_funds_monthly = {
        add_to_variable = { investment_fund = investment_fund_balance }

        # Validity pre-pass: drop & refund EVERY invalid active IO up-front (not just the one being funded),
        # so the player's manual funding can never land on a stale opportunity. Reverse-walk by index so
        # removals don't shift entries we haven't visited yet.
        set_temp_variable = { iocheck_idx_temp = active_investment_opportunities^num }
        while_loop_effect = {
            limit = { check_variable = { iocheck_idx_temp > 0 } }
            subtract_from_temp_variable = { iocheck_idx_temp = 1 }
            set_temp_variable = { io_num_temp = active_investment_opportunities^iocheck_idx_temp }
            check_investment_opportunity = yes
        }

        set_temp_variable = { ifm_temp = 0 }
        set_temp_variable = { spending_cap_temp = investment_fund_target_spending }
        subtract_from_temp_variable = { spending_cap_temp = investment_fund_spending }
        while_loop_effect = {
            limit = {
                check_variable = { spending_cap_temp > 0 }
                check_variable = { ifm_temp < investment_opportunities_cap }
                OR = {
                    check_variable = { active_investment_opportunities^num > 0 }
                    AND = { # an active national opportunity still needs funding
                        check_variable = { active_national_opportunity > 0 }
                        check_variable = { nopp_cost@var:active_national_opportunity > nopp_investment@var:active_national_opportunity }
                    }
                }
            }

            # Is the active national opportunity still in need of funding this pass?
            set_temp_variable = { nopp_fundable_temp = 0 }
            if = {
                limit = {
                    check_variable = { active_national_opportunity > 0 }
                    check_variable = { nopp_cost@var:active_national_opportunity > nopp_investment@var:active_national_opportunity }
                }
                set_temp_variable = { nopp_fundable_temp = 1 }
                set_temp_variable = { active_nopp_temp = active_national_opportunity }
            }

            # Re-validate the top IO (construction started earlier this loop can invalidate a sibling), then
            # see if it's fundable.
            if = {
                limit = { check_variable = { active_investment_opportunities^num > 0 } }
                set_temp_variable = { io_num_temp = active_investment_opportunities^0 }
                check_investment_opportunity = yes
            }
            set_temp_variable = { io_available_temp = 0 }
            if = {
                limit = { check_variable = { active_investment_opportunities^num > 0 } }
                set_temp_variable = { io_num_temp = active_investment_opportunities^0 }
                if = {
                    limit = {
                        has_variable = investment_opportunity_state@var:io_num_temp
                        NOT = { check_variable = { investment_opportunity_attractiveness@var:io_num_temp < 100 } } # Gate: capitalists won't fund IOs under 100 attractiveness
                    }
                    set_temp_variable = { io_available_temp = 1 }
                }
            }

            # Fund the national opportunity if it exists and either there's no IO or it out-scores the top IO.
            # NOTE: IO and national-opportunity attractiveness are on different scales right now - tune the
            # founding score (score_national_opportunity) to make this comparison fair.
            if = {
                limit = {
                    check_variable = { nopp_fundable_temp = 1 }
                    OR = {
                        check_variable = { io_available_temp = 0 }
                        check_variable = { nopp_attractiveness@var:active_nopp_temp > investment_opportunity_attractiveness@var:io_num_temp }
                    }
                }
                log = "Attractiveness: [?nopp_attractiveness@var:active_nopp_temp] | [?investment_opportunity_attractiveness@var:io_num_temp]"
                set_temp_variable = { spending_amount_temp = nopp_cost@var:active_nopp_temp }
                subtract_from_temp_variable = { spending_amount_temp = nopp_investment@var:active_nopp_temp }
                if = { # Enough money to finish pooling
                    limit = { check_variable = { spending_cap_temp > spending_amount_temp } }
                    subtract_from_variable = { investment_fund = spending_amount_temp }
                    subtract_from_temp_variable = { spending_cap_temp = spending_amount_temp }
                    set_variable = { nopp_investment@var:active_nopp_temp = nopp_cost@var:active_nopp_temp }
                    # Fulfillment (founding the company) happens in national_opportunities_monthly this same tick.
                }
                else = { # Partial - pool what's left and stop
                    subtract_from_variable = { investment_fund = spending_cap_temp }
                    add_to_variable = { nopp_investment@var:active_nopp_temp = spending_cap_temp }
                    set_temp_variable = { spending_cap_temp = 0 }
                }
            }
            else_if = {
                limit = { check_variable = { io_available_temp = 1 } }

                set_temp_variable = { spending_amount_temp = investment_opportunity_cost@var:io_num_temp }
                subtract_from_temp_variable = { spending_amount_temp = investment_opportunity_investment@var:io_num_temp }

                if = { # Enough money
                    limit = {
                        check_variable = { spending_cap_temp > spending_amount_temp }
                    }
                    subtract_from_variable = { investment_fund = spending_amount_temp }
                    subtract_from_temp_variable = { spending_cap_temp = spending_amount_temp }
                    set_variable = { investment_opportunity_investment@var:io_num_temp = investment_opportunity_cost@var:io_num_temp }
                    evaluate_investment_opportunity = yes
                }
                else = { # Not enough money
                    subtract_from_variable = { investment_fund = spending_cap_temp }
                    add_to_variable = { investment_opportunity_investment@var:io_num_temp = spending_cap_temp }
                    set_temp_variable = { spending_cap_temp = 0 }
                }
            }
            else = {
                # Nothing worth funding this pass (no national opportunity, and the top IO is under the
                # 100-attractiveness floor). Stop and leave the money in the fund - the unspent surplus below
                # accumulates in investment_fund and feeds back into a lower pop investment appetite.
                set_temp_variable = { spending_cap_temp = 0 }
            }

            add_to_temp_variable = { ifm_temp = 1 }
        }

        # Target spending that couldn't be deployed this tick (it stays in investment_fund, growing the pool).
        # Made permanent so pops can damp their contribution when the fund can't put money to work.
        set_variable = { investment_fund_unspent = spending_cap_temp }

        if = {
            limit = {
                check_variable = { investment_fund < 0 }
            }
            # TODO - Add debt
            set_variable = { investment_fund = 0 }
        }
    }

# Incentives
	calculate_investment_appetite = {
		# Inputs (temp):
		#   top_attractiveness_temp - attractiveness of the best available IO (top of the descending-sorted
		#                             controller:active_investment_opportunities; up to 1 month stale; base ~100)
		#   incentives_temp         - modifier@incentives_modifier (base 100)
		# Output (temp):
		#   investment_appetite_temp

		set_temp_variable = {
			incentives_mod_temp = {
				value = incentives_temp
				add = 200
			}
		}
		set_temp_variable = {
			investment_appetite_temp = {
				value = incentives_temp
				multiply = 3
				divide = incentives_mod_temp
				max = 0
				min = 2
			}
		}

		# Pool-overhang damping: if the fund is accumulating undeployed capital (can't put money to work),
		# pops throw in less. overhang = investment_fund / monthly target spend (months of idle capital).
		set_temp_variable = { pool_damping_temp = {
			value = controller:investment_fund
			if = { limit = { value = controller:gdp_amount greater_than = 0 }
				divide = controller:gdp_amount
				multiply = global.investment_pool_tolerance
				multiply = -1
				add = 1
			}
			else = {
				multiply = 0
			}
			clamp = { min = global.investment_pool_appetite_floor  max = 1 }
		} }
		multiply_temp_variable = { investment_appetite_temp = pool_damping_temp }
	}
#

# NATIONAL OPPORTUNITIES ----------------------------------------------------------
# Country-level investment actions funded by the investment fund, SEPARATE from the
# state-tied IO list (active_investment_opportunities). Generic typed-record frame; only
# the `nopp_founding` type is implemented (adding a type = add a branch, not a restructure).
# Lifecycle mirrors the IO system one scope up:
#   allocate id -> select/propose -> score -> cost -> check (stringent + clock) -> fulfill -> remove.
# The money POOLING is NOT here: it happens in investment_funds_monthly's spending loop, where the
# active founding competes with state IOs by attractiveness under the same target-spending cap.
# national_opportunities_monthly only validates, fulfils the fully-funded, and proposes new ones.
#
# DATA (country scope):
#   national_opportunities      - array of active nopp ids
#   active_national_opportunity - id of the one in-flight founding (0 = none); what the fund pools into
#   nopp_num                    - id allocator counter (mirrors io_num)
#   nopp_type@id                - token:nopp_founding
#   nopp_state@id               - seed BUILDING state (stored scope, restore via var:); where the slice is
#   nopp_hq_state@id            - HQ / dividend state (stored scope); the financial hub, often != build state
#   nopp_focus@id               - seed industry token
#   nopp_holding_size@id        - size of the local slice the new company incorporates on birth
#   nopp_cost@id / nopp_investment@id - total to raise (gross) / capitalist money pooled so far
#   nopp_reimburse@id           - asset value refunded to the IF on founding (only the net is truly spent)
#   nopp_subsidy@id             - govt subsidy money (-> state-shares later; 0 = none yet)
#   nopp_attractiveness@id
#   nopp_months_left@id         - funding clock; founding fails if it hits 0 unfunded
#
# Founding CAN fail: select conditions are loose, but check_national_opportunity re-validates
# stringent continuation conditions every month and counts down the clock. On failure the
# pooled money (+ subsidy) is refunded to the fund and the record dropped.

	allocate_nopp_number = {
		add_to_variable = { nopp_num = 1 }
		if = { limit = { check_variable = { nopp_num > 100000 } } set_variable = { nopp_num = 1 } }
		while_loop_effect = {
			limit = { has_variable = nopp_type@var:nopp_num }
			add_to_variable = { nopp_num = 1 }
		}
	}

	# Country scope. Output: founding_cost_mult_temp. Progressive penalty - each company a country already
	# owns compounds the cost of founding the next one (so a sprawling owner founds ever more rarely).
	# Uses a math expression: founding_cost_mult_temp = 1.2 ^ (number of companies owned)
	# (~7 companies to reach parity with a full acquisition's cost).
	calc_founding_cost_mult = {
		set_temp_variable = { founding_cost_mult_temp = { value = 1.2  pow = companies^num } }
	}

	# State scope. Input: nopp_type_temp, industry_token. Output: nopp_attr_temp
	score_national_opportunity = {
		set_temp_variable = { nopp_attr_temp = 0 }
		if = {
			limit = { check_variable = { nopp_type_temp = token:nopp_founding } }

			# ROI base, mirroring score_io_attractiveness so this is COMPARABLE to IO attractiveness:
			# 100 at a 10% annualized ROI, linear (clamped). ROI = monthly profit per size of the seed
			# industry divided by its per-size NET cost (the asset value is reimbursed on founding, so the
			# real spend is just the investment portion - making founding read ~4x better than an acquisition).
			if = {
				limit = { check_variable = { size@var:industry_token > 0 } }
				set_temp_variable = { roi_temp = profit@var:industry_token }
				divide_temp_variable = { roi_temp = size@var:industry_token } # monthly profit per size
			}
			else = { set_temp_variable = { roi_temp = 0 } }

			set_temp_variable = { per_size_cost_temp = asset_value@var:industry_token }
			if = { limit = { check_variable = { size@var:industry_token > 0 } } divide_temp_variable = { per_size_cost_temp = size@var:industry_token } }
			multiply_temp_variable = { per_size_cost_temp = global.start_construction_cost_multiplier } # only the investment portion is real cost...
			multiply_temp_variable = { per_size_cost_temp = founding_cost_mult_temp } # ...escalating with companies owned

			if = {
				limit = { check_variable = { per_size_cost_temp > 0 } }
				divide_temp_variable = { roi_temp = per_size_cost_temp }
			}
			else = { set_temp_variable = { roi_temp = 0 } }

			multiply_temp_variable = { roi_temp = 120 } # monthly roi * 12 months / 10%
			clamp_temp_variable = { var = roi_temp min = -10.0 max = 10.0 }

			set_temp_variable = { nopp_attr_temp = 100 }
			multiply_temp_variable = { nopp_attr_temp = roi_temp }

			# Founding factor: scale by the local fraction (room to incorporate - a mostly-incorporated
			# industry is a poor founding seed). Keeps the ROI base dominant, like the IO 'factors' stage.
			set_temp_variable = { local_frac_temp = owner_sizes_local@var:industry_token }
			if = { limit = { check_variable = { size@var:industry_token > 0 } } divide_temp_variable = { local_frac_temp = size@var:industry_token } }
			multiply_temp_variable = { nopp_attr_temp = local_frac_temp }
		}
	}

	# State scope. Input: nopp_type_temp, industry_token, nopp_size_temp, founding_cost_mult_temp.
	# Output: nopp_cost_temp (gross, what must be pooled) and nopp_reimburse_temp (asset value handed
	# back to the IF on founding - the fund is buying the assets off the locals, so that part is circular).
	calc_national_opportunity_cost = {
		set_temp_variable = { nopp_cost_temp = 0 }
		set_temp_variable = { nopp_reimburse_temp = 0 }
		if = {
			limit = { check_variable = { nopp_type_temp = token:nopp_founding } }
			# Asset value of the seed slice (per size * slice) - the reimbursed portion.
			set_temp_variable = { nopp_reimburse_temp = asset_value@var:industry_token }
			divide_temp_variable = { nopp_reimburse_temp = size@var:industry_token } # per size
			multiply_temp_variable = { nopp_reimburse_temp = nopp_size_temp } # the slice being incorporated

			# Net (real) cost = investment portion only: start_construction_cost_multiplier, escalated by the
			# progressive penalty. Gross = asset value + net; on founding the asset value is reimbursed, so
			# only the net actually leaves the fund.
			set_temp_variable = { nopp_net_cost_temp = nopp_reimburse_temp }
			multiply_temp_variable = { nopp_net_cost_temp = global.start_construction_cost_multiplier }
			multiply_temp_variable = { nopp_net_cost_temp = founding_cost_mult_temp }

			set_temp_variable = { nopp_cost_temp = nopp_reimburse_temp }
			add_to_temp_variable = { nopp_cost_temp = nopp_net_cost_temp }
		}
	}

	# Country scope. Inputs (temps): nopp_type_temp, nopp_state_temp, nopp_hq_state_temp, nopp_focus_temp,
	# nopp_size_temp, nopp_cost_temp, nopp_reimburse_temp, nopp_attr_temp.
	add_national_opportunity = {
		allocate_nopp_number = yes
		add_to_array = { national_opportunities = nopp_num }
		set_variable = { nopp_type@var:nopp_num = nopp_type_temp }
		set_variable = { nopp_state@var:nopp_num = nopp_state_temp } # seed building state
		set_variable = { nopp_hq_state@var:nopp_num = nopp_hq_state_temp } # HQ (dividend) state - may differ
		set_variable = { nopp_focus@var:nopp_num = nopp_focus_temp }
		set_variable = { nopp_holding_size@var:nopp_num = nopp_size_temp }
		set_variable = { nopp_cost@var:nopp_num = nopp_cost_temp }
		set_variable = { nopp_reimburse@var:nopp_num = nopp_reimburse_temp } # asset value returned to the IF on founding
		set_variable = { nopp_investment@var:nopp_num = 0 }
		set_variable = { nopp_subsidy@var:nopp_num = 0 }
		set_variable = { nopp_attractiveness@var:nopp_num = nopp_attr_temp }
		set_variable = { nopp_months_left@var:nopp_num = 12 } # founding deadline (months)

		# The one in-flight opportunity that investment_funds_monthly competes against IOs to fund.
		if = {
			limit = {
				check_variable = { active_national_opportunity = 0 }
			}
			set_variable = { active_national_opportunity = nopp_num }
		}
	}

	# Country scope. Input: nopp_num_temp.
	remove_national_opportunity = {
		if = {
			limit = { check_variable = { active_national_opportunity = nopp_num_temp } }
			set_variable = { active_national_opportunity = 0 }
		}
		remove_from_array = { national_opportunities = nopp_num_temp }
		clear_variable = nopp_type@var:nopp_num_temp
		clear_variable = nopp_state@var:nopp_num_temp
		clear_variable = nopp_hq_state@var:nopp_num_temp
		clear_variable = nopp_focus@var:nopp_num_temp
		clear_variable = nopp_holding_size@var:nopp_num_temp
		clear_variable = nopp_cost@var:nopp_num_temp
		clear_variable = nopp_reimburse@var:nopp_num_temp
		clear_variable = nopp_investment@var:nopp_num_temp
		clear_variable = nopp_subsidy@var:nopp_num_temp
		clear_variable = nopp_attractiveness@var:nopp_num_temp
		clear_variable = nopp_months_left@var:nopp_num_temp
	}

	# Country scope. Proposes ONE founding if none is currently active. Loose selection: any
	# controlled state holding a profitable, locally-owned industry. Picks the best by founding
	# attractiveness, sizes the seed slice (half the local holding), and records the opportunity.
	national_opportunity_select = {
		# Only one founding in flight at a time.
		set_temp_variable = { founding_active_temp = 0 }
		for_each_loop = {
			array = national_opportunities
			value = nopp_scan_temp
			if = {
				limit = { check_variable = { nopp_type@var:nopp_scan_temp = token:nopp_founding } }
				set_temp_variable = { founding_active_temp = 1 }
			}
		}

		if = {
			limit = { 
                check_variable = { founding_active_temp = 0 }
                check_variable = { gdp_amount > 10000 }
            }

			# Progressive cost penalty for this country (read by both score and cost below).
			calc_founding_cost_mult = yes

			set_temp_variable = { best_score_temp = 0 }
			set_temp_variable = { best_state_temp = 0 }
			set_temp_variable = { best_focus_temp = 0 }
			set_temp_variable = { has_candidate_temp = 0 }

			# Phase 1 (industry-first, country scope): pick the most productive PRIVATE building type.
			# global.building_token_array excludes govt buildings AND RGOs, so RGO founding is naturally
			# deferred until exploit/prospect handling exists. A type already focused by one of this country's
			# companies is penalized (placeholder until market share exists) so foundings spread across sectors.
			set_temp_variable = { founding_industry_temp = 0 }
			set_temp_variable = { founding_best_prod_temp = 0 }
			for_each_loop = {
				array = global.building_token_array
				value = industry_token
				if = {
					limit = {
						check_variable = { highest_productivity@var:industry_token > 2 }
						check_variable = { size@var:industry_token > 10 }
                        any_controlled_state = {
							check_variable = { profitability@var:industry_token > 0.1 }
                        }
					}
					set_temp_variable = { fi_score_temp = highest_productivity@var:industry_token }
					# Penalty if a company already focuses this industry.
					set_temp_variable = { fi_taken_temp = 0 }
					for_each_loop = {
						array = companies
						value = fi_cid_temp
						if = { limit = { check_variable = { global.company_focus@var:fi_cid_temp = industry_token } } set_temp_variable = { fi_taken_temp = 1 } }
					}
					if = { limit = { check_variable = { fi_taken_temp = 1 } } multiply_temp_variable = { fi_score_temp = 0.25 } }
					if = {
						limit = { check_variable = { fi_score_temp > founding_best_prod_temp } }
						set_temp_variable = { founding_best_prod_temp = fi_score_temp }
						set_temp_variable = { founding_industry_temp = industry_token }
					}
				}
			}

			# Phase 2 (state, country scope): the best controlled state to seed that one industry, by ROI fit.
			if = {
				limit = { NOT = { check_variable = { founding_industry_temp = 0 } } }
				every_controlled_state = {
					if = {
						limit = {
							check_variable = { owner_sizes_local@var:founding_industry_temp > 0 }
							check_variable = { profitability@var:founding_industry_temp > 0.1 }
						}
						set_temp_variable = { industry_token = founding_industry_temp }
						set_temp_variable = { nopp_type_temp = token:nopp_founding }
						score_national_opportunity = yes
						if = {
							limit = { check_variable = { nopp_attr_temp > best_score_temp } }
							set_temp_variable = { best_score_temp = nopp_attr_temp }
							set_temp_variable = { best_state_temp = THIS }
							set_temp_variable = { best_focus_temp = founding_industry_temp }
							set_temp_variable = { has_candidate_temp = 1 }
						}
					}
				}
			}

			if = {
				limit = { check_variable = { has_candidate_temp = 1 } }

				# How much GROSS the fund can plausibly pool toward one founding: a few months of its target
				# spending. Sizing must use the gross cost (not the net) because the entire asset value has to
				# be fronted before the founding-completion reimbursement lands - that upfront pool, not the
				# eventual net cost, is the real constraint. Kept modest since the founding competes with IOs
				# for the fund and must finish within the ~12-month clock.
				set_temp_variable = { founding_budget_temp = investment_fund_target_spending }
				multiply_temp_variable = { founding_budget_temp = 3 } # tunable: months of fund budget per founding

				# Size the seed slice to that budget and price it inside the chosen state.
				var:best_state_temp = {
					set_temp_variable = { industry_token = best_focus_temp }
					set_temp_variable = { nopp_type_temp = token:nopp_founding }

					# Gross per-size cost = asset/size * (1 + start_construction_cost_multiplier * progressive penalty).
					set_temp_variable = { gross_ps_temp = asset_value@var:best_focus_temp }
					if = { limit = { check_variable = { size@var:best_focus_temp > 0 } } divide_temp_variable = { gross_ps_temp = size@var:best_focus_temp } }
					set_temp_variable = { net_mult_temp = global.start_construction_cost_multiplier }
					multiply_temp_variable = { net_mult_temp = founding_cost_mult_temp }
					add_to_temp_variable = { net_mult_temp = 1 }
					multiply_temp_variable = { gross_ps_temp = net_mult_temp }

					# size = 50% * budget / gross-per-size, clamped to what's locally available.
					set_temp_variable = { nopp_size_temp = founding_budget_temp }
					if = { limit = { check_variable = { gross_ps_temp > 0 } } divide_temp_variable = { nopp_size_temp = gross_ps_temp } }
					multiply_temp_variable = { nopp_size_temp = 0.5 }
					round_temp_variable = nopp_size_temp
					clamp_temp_variable = { var = nopp_size_temp min = 1 max = owner_sizes_local@var:best_focus_temp }

					calc_national_opportunity_cost = yes
				}

				# Pick the HQ state SEPARATELY: the country's standout financial hub (most industrialised +
				# wealthiest capitalists), which is often not the seed building's state (e.g. UK -> London).
				# Dividends will flow to this state's capitalists. Weighting of the two terms is tunable.
				set_temp_variable = { hq_state_temp = best_state_temp } # fallback: the build state
				set_temp_variable = { hq_best_temp = -1 }
				every_controlled_state = {
					set_temp_variable = { hq_score_temp = total_industry_size }
                    if = {
                        limit = {
                            check_variable = { controller:market_capital = THIS }
                        }
                        add_to_temp_variable = { hq_score_temp = 50 }
                    }
					if = {
						limit = { check_variable = { pops@token:capitalists > 0 } }
						set_temp_variable = { cap_pc_temp = pops_earnings@token:capitalists }
						divide_temp_variable = { cap_pc_temp = pops@token:capitalists } # capitalist per-capita earnings
						divide_temp_variable = { cap_pc_temp = controller:gdp_per_capita } # capitalist per-capita earnings
						add_to_temp_variable = { hq_score_temp = cap_pc_temp }
					}
					if = {
						limit = { check_variable = { hq_score_temp > hq_best_temp } }
						set_temp_variable = { hq_best_temp = hq_score_temp }
						set_temp_variable = { hq_state_temp = THIS }
					}
				}

				set_temp_variable = { nopp_type_temp = token:nopp_founding }
				set_temp_variable = { nopp_state_temp = best_state_temp }
				set_temp_variable = { nopp_hq_state_temp = hq_state_temp }
				set_temp_variable = { nopp_focus_temp = best_focus_temp }
				set_temp_variable = { nopp_attr_temp = best_score_temp }
				add_national_opportunity = yes
			}
		}
	}

	# Country scope. Input: nopp_num_temp. Re-validates stringent continuation conditions and
	# ticks the clock; refreshes cost/attractiveness. On failure refunds pooled money + subsidy
	# to the fund and removes the record.
	check_national_opportunity = {
		set_temp_variable = { nopp_type_temp = nopp_type@var:nopp_num_temp }
		set_temp_variable = { nopp_invalid_temp = 0 }

		# Clock.
		subtract_from_variable = { nopp_months_left@var:nopp_num_temp = 1 }
		if = {
			limit = { NOT = { check_variable = { nopp_months_left@var:nopp_num_temp > 0 } } }
			set_temp_variable = { nopp_invalid_temp = 1 }
		}

		if = {
			limit = { check_variable = { nopp_type_temp = token:nopp_founding } }
			set_temp_variable = { nopp_size_temp = nopp_holding_size@var:nopp_num_temp }
			set_temp_variable = { nopp_focus_temp = nopp_focus@var:nopp_num_temp }
			set_temp_variable = { nopp_state_temp = nopp_state@var:nopp_num_temp }

			# Progressive cost penalty (read by score & cost inside the state scope below).
			calc_founding_cost_mult = yes

			var:nopp_state_temp = {
				set_temp_variable = { industry_token = nopp_focus_temp }
				# Stringent: seed must stay solidly profitable
				if = {
					limit = { NOT = { check_variable = { profitability@var:industry_token > 0.05 } } }
					set_temp_variable = { nopp_invalid_temp = 1 }
				}
				# Seed slice must still be available locally (could have been acquired/sold off).
				if = {
					limit = { check_variable = { nopp_size_temp > owner_sizes_local@var:industry_token } }
					set_temp_variable = { nopp_invalid_temp = 1 }
				}
				# Refresh attractiveness & cost from current state.
				set_temp_variable = { nopp_type_temp = token:nopp_founding }
				score_national_opportunity = yes
				calc_national_opportunity_cost = yes
			}

			# Stringent: attractiveness must remain positive.
			if = {
				limit = { NOT = { check_variable = { nopp_attr_temp > 0 } } }
				set_temp_variable = { nopp_invalid_temp = 1 }
			}
			set_variable = { nopp_attractiveness@var:nopp_num_temp = nopp_attr_temp }
			set_variable = { nopp_cost@var:nopp_num_temp = nopp_cost_temp }
			set_variable = { nopp_reimburse@var:nopp_num_temp = nopp_reimburse_temp }
		}

		if = {
			limit = { check_variable = { nopp_invalid_temp = 1 } }
			# Refund the pooled capitalist money and any govt subsidy before dropping.
			add_to_variable = { investment_fund = nopp_investment@var:nopp_num_temp }
			add_to_variable = { investment_fund = nopp_subsidy@var:nopp_num_temp }
			remove_national_opportunity = yes
		}
	}

	# Country scope. Input: nopp_num_temp. Carries out a fully-funded opportunity.
	fulfill_national_opportunity = {
		set_temp_variable = { nopp_type_temp = nopp_type@var:nopp_num_temp }
		if = {
			limit = { check_variable = { nopp_type_temp = token:nopp_founding } }

			set_temp_variable = { company_hq_state_temp = nopp_hq_state@var:nopp_num_temp } # HQ = financial hub, not the build state
			set_temp_variable = { company_focus_temp = nopp_focus@var:nopp_num_temp }
			set_temp_variable = { company_type_temp = token:company_type_private }
			found_company = yes # outputs cid

			# Incorporate the seed slice from local ownership (company_add_holding's local clamp is
			# correct here - the slice genuinely comes from owner_sizes_local).
			set_temp_variable = { company_temp = cid }
			set_temp_variable = { holding_state_temp = nopp_state@var:nopp_num_temp }
			set_temp_variable = { holding_building_temp = nopp_focus@var:nopp_num_temp }
			set_temp_variable = { holding_size_temp = nopp_holding_size@var:nopp_num_temp }
			company_add_holding = yes

			# The fund just bought these assets off the locals - that money flows back into the fund (it's
			# circular, which is what makes founding cheap: only the investment portion is truly spent).
			add_to_variable = { investment_fund = nopp_reimburse@var:nopp_num_temp }

			# State-share stub: the govt subsidy should buy equity in the new company. The stock /
			# ownership-split system doesn't exist yet, so nopp_subsidy is just dropped for now.
			# TODO: convert nopp_subsidy@var:nopp_num_temp into state-held shares of cid once stocks exist.

            companies_update = yes
		}
		remove_national_opportunity = yes
	}

	# Country scope. Monthly driver. The actual money pooling lives in investment_funds_monthly (where the
	# fund's spending cap is, so foundings compete with IOs and respect the target spend). This runs AFTER
	# it in the tick and: validates/ticks the clock on active opportunities, fulfils any that got fully
	# funded this tick, and proposes a new founding if none is active.
	# Reverse-walk by index so check/fulfill removals don't shift entries we haven't visited.
	national_opportunities_monthly = {
		set_temp_variable = { nom_idx_temp = national_opportunities^num }
		while_loop_effect = {
			limit = { check_variable = { nom_idx_temp > 0 } }
			subtract_from_temp_variable = { nom_idx_temp = 1 }
			set_temp_variable = { nopp_num_temp = national_opportunities^nom_idx_temp }

			if = {
				# Fully funded -> fulfill BEFORE any cost/clock recalc, so a risen cost (or the clock
				# expiring this very month) can't rob a founding that already raised its money.
				limit = { NOT = { check_variable = { nopp_cost@var:nopp_num_temp > nopp_investment@var:nopp_num_temp } } }
				fulfill_national_opportunity = yes
			}
			else = {
				# Not yet funded: validate, tick the clock, and refresh cost/attractiveness for next month.
				check_national_opportunity = yes # stringent + clock; may drop+refund (removes from array)
			}
		}
        
		if = {
			limit = {
				check_variable = { active_national_opportunity = 0 }
				check_variable = { national_opportunities^num > 0 }
			}
			set_variable = { active_national_opportunity = national_opportunities^0 }
		}

		# Propose a new founding if none is active. After fulfilment, so a fresh proposal gets a full clock.
		national_opportunity_select = yes
	}
