define_ig_egalitarians = {
    # Interest Group: Egalitarians - Egalitarian Interest Group

    # If ideologies conflict with each other, position score gets summed together to determine overall stance.
    clear_array = global.ig_traits@token:interest_group_egalitarians
    add_to_array = { global.ig_traits@token:interest_group_egalitarians = token:trait_abolitionist }
    add_to_array = { global.ig_traits@token:interest_group_egalitarians = token:trait_republican }
}

define_ig_radical_republicans = {
    # Interest Group: Radical Republicans - Egalitarian Interest Group
    set_variable = { ig_name@token:interest_group_egalitarians = token:ig_radical_republicans }

    # If ideologies conflict with each other, position score gets summed together to determine overall stance.
    clear_array = ig_traits@token:interest_group_egalitarians
    for_each_loop = {
        array = global.ig_traits@token:interest_group_egalitarians
        value = trait_temp
        add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_temp }
    }

    add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_abolitionist } # Slavery
    add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_republican } # Government
    add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_protectionist } # Trade
    add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_capitalist } # Economy
    add_to_array = { ig_traits@token:interest_group_egalitarians = token:trait_pro_welfare } # Welfare
}

			}
			if = {
				limit = { autocracy > 0.01 }

				set_variable = { strength@token:party_generic_fundamentalist_party =  party_popularity_100@fundamentalism }
				set_variable = { ideology@token:party_generic_fundamentalist_party = token:ideology_fundamentalist }

				set_variable = { strength@token:party_generic_independent_party =  party_popularity_100@radical_democracy }
				set_variable = { ideology@token:party_generic_independent_party = token:ideology_reactionary }
			}
			for_each_loop = {
				array = parties_temp_array
				value = p
				if = {
					limit = {
						current_party_ideology_group = centrism
						check_variable = { ideology@var:p = token:ideology_liberal }
					}
					add_to_array = { ruling_coalition_array = p }
				}
				else_if = {
					limit = {
						current_party_ideology_group = traditional_conservatism
						check_variable = { ideology@var:p = token:ideology_conservative }
					}
					add_to_array = { ruling_coalition_array = p }
				}
				else_if = {
					limit = {
						current_party_ideology_group = autocracy

					}

					limit = {
						current_party_ideology_group = chauvinist_populism
						check_variable = { ideology@var:p = token:ideology_chauvinist }
					}

					}

				}
				else_if = {
					limit = {
						current_party_ideology_group = radical_democracy
						check_variable = { ideology@var:p = token:ideology_reactionary }
					}
					add_to_array = { ruling_coalition_array = p }
				}
			}
			for_each_loop = {
				array = parties_temp_array
				value = p
				for_each_loop = {
					array = parties_array
					index = ii
					value = pp
					break = bb
					if = {
						limit = { check_variable = { strength@var:p > strength@var:pp } }
						add_to_array = {
							array = parties_array
							value = p
							index = ii
						}
						set_temp_variable = { bb = 1 }
					}
				}
				if = {
					limit = { check_variable = { bb = 0 } }
					add_to_array = { parties_array = p }
				}
			}
		}
}
