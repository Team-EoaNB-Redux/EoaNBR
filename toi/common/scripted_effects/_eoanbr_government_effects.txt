internal_politics_overview_party_interest_groups_toggle = {
	set_temp_variable = { party_temp = v }
	set_temp_variable = { index_temp = i }

	if = {
		limit = { check_variable = { color@var:party_temp > 0 } } # Check if Party
		if = {
			limit = {
				NOT = {
					all_of = {
						array = interest_groups@var:party_temp
						value = vv
						is_in_array = {
							array = politics_overview_array
							value = vv
						}
					}
				}
			}
			set_temp_variable = { ii = index_temp }
			add_to_temp_variable = { ii = 1 }
			for_each_loop = {
				array = interest_groups@var:party_temp
				value = vv
				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = politics_overview_array
								value = vv
							}
						}
					}
					add_to_array = {
						array = politics_overview_array
						value = vv
						index = ii
					}
				}
			}
		}
		else = {
			for_each_loop = {
				array = interest_groups@var:party_temp
				value = vv
				remove_from_array = {
					array = politics_overview_array
					value = vv
				}
			}
		}
	}
}

politics_overview_reset = {
	clear_array = politics_overview_array
	set_variable = { political_overview_open@token:ruling_coalition = 0 }
	set_variable = { political_overview_open@token:oppositional_parties = 0 }
	set_variable = { political_overview_open@token:ousted_parties = 0 }
	if = {
		limit = {
			check_variable = { parties@token:ruling_coalition^num > 0 }
		}
		set_variable = { political_overview_open@token:ruling_coalition = 1 }
		add_to_array = {
			array = politics_overview_array
			value = token:ruling_coalition
		}
		set_variable = { strength@token:ruling_coalition = 0 }
		for_each_loop = {
			array = parties@token:ruling_coalition
			value = p
			add_to_array = {
				array = politics_overview_array
				value = p
			}
