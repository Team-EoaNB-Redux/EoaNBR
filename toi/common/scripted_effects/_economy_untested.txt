change_interest_rate = {
    #input: ir_change_temp

    if = {
        limit = {
            check_variable = { ir_change_temp > 0 }
        }
        add_to_variable = { interest_rate = ir_change_temp }
        
        if = {
            limit = {
                check_variable = { interest_rate > 10 }
            }
            set_variable = { interest_rate = 10 }
        }
    }
    else_if = {
        limit = {
            check_variable = { ir_change_temp < 0 }
        }
        add_to_variable = { interest_rate = ir_change_temp }

        if = {
            limit = {
                check_variable = { interest_rate < 1 }
            }
            set_variable = { interest_rate = 1 }
        }
    }
}

calc_trade_rework = {
	# FulL Reset
	for_each_scope_loop = {
		array = global.tr_arr
	
		set_variable = { tr_supply = tr_production }
		set_variable = { tr_demand = tr_consumption }

		set_variable = {
			tr_price = {
				value = tr_demand
				divide = tr_supply
				pow = 2
				min = 4
				max = 0.25
			}
		}
	}


	# Adjust Trade Volumes
	for_each_scope_loop = {
		array = global.tr_arr

		# Add last month trade volume
		if = {
			limit = { check_variable = { tr_total_trade > 0 } }
			add_to_variable = { tr_supply = tr_total_trade }
		}
		else = {
			add_to_variable = { tr_demand = tr_total_trade }
		}

		for_each_loop = {
			array = tr_arr
			value = country_temp
			if = {
				limit = { check_variable = { tr_trade@var:country_temp > 0 } }
				add_to_variable = { country_temp:tr_supply = tr_trade@var:country_temp }
			}
			else = {
				add_to_variable = { country_temp:tr_demand = tr_trade@var:country_temp }
			}
			var:country_temp = {
				set_variable = {
					tr_price = {
						value = tr_demand
						divide = tr_supply
						pow = 2
						min = 4
						max = 0.25
					}
				}
			}
		}

		set_variable = { tr_total_trade = 0 }
		set_variable = { tr_profits = 0 }

		for_each_loop = {
			array = tr_arr
			value = country_temp

			# Export = S_production * sqrt( global_price ) - D_consumption

			set_temp_variable = {
				export_margin_temp = {
					value = country_temp:tr_price
					root = 2
					multiply = tr_supply
					subtract = tr_demand
				}
				import_margin_temp = {
					value = country_temp:tr_price
					root = 2
					multiply = tr_demand
					subtract = tr_supply
				}
			}
			if = {
				limit = {
					check_variable = { margin_temp > 0.1 }
				}
				add_to_variable = { tr_trade@var:country_temp = 1 }
			}
			else = {
				subtract_from_variable = { tr_trade@var:country_temp = 1 }
			}
			add_to_variable = { tr_total_trade = tr_trade@var:country_temp }
			multiply_temp_variable = { margin_temp = tr_trade@var:country_temp }
			add_to_variable = { tr_profits = margin_temp }
		}

		
		
		log = "[THIS.GetName]: [?tr_supply] | tr_profits"
	}
}
