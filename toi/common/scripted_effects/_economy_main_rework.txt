
#Pops
	pops_update = {
		pops_reset_core_variable = yes
		pops_prepare_contextual_multipliers = yes

		for_each_loop = {
			array = global.pops_array
			value = pop_update_token

			if = {
				limit = {
					check_variable = { pops_workforce@var:pop_update_token > 0 }
				}

				set_temp_variable = { current_pop_token = pop_update_token }

				pops_cache_pop_context = yes

				pops_process_workforce_status = yes

				if = { # Slaves have no income and no needs (Q12)
					limit = { check_variable = { current_pop_token = token:slaves } }
					pops_commit_slaves = yes
				}
				else = {
					pops_calculate_earnings_and_investment = yes
					pops_calculate_needs_spending = yes
					pops_calculate_living_standards = yes
					pops_calculate_taxation = yes

					pops_calculate_debt_servicing = yes
					pops_calculate_credit_metrics = yes

					pops_update_confidence = yes
					pops_calculate_spending_targets = yes
					pops_calculate_savings_targets = yes

					pops_add_to_state_totals = yes

					pops_commit_pop_context = yes
				}
			}
			else = {
				pops_clear_inactive_pop_data = yes
			}
		}

		pops_compute_summary_metrics = yes
	}

	pops_cache_pop_context = {
		set_temp_variable = { p_pops_workforce = pops_workforce@var:current_pop_token }
		set_temp_variable = { p_pops_employed = pops_employed@var:current_pop_token }
		set_temp_variable = { p_pops_dividends = pops_dividends@var:current_pop_token }
		set_temp_variable = { p_bank_lent_to_pops_k = bank_lent_to_pops_k@var:current_pop_token }
		set_temp_variable = { p_pops_savings = pops_savings@var:current_pop_token }
	}
	pops_commit_pop_context = {
		# Process Workforce
		set_variable = { pops@var:current_pop_token = p_pops }
		set_variable = { pops_percentage@var:current_pop_token = p_pops_percentage }
		set_variable = { pops_unemployed@var:current_pop_token = p_pops_unemployed }

		# Calculate Earnings & Investment
		set_variable = { pops_earnings@var:current_pop_token = p_pops_earnings }
		set_variable = { pops_investment@var:current_pop_token = p_pops_investment }

		# Calculate Needs Spending
		set_variable = { pops_needs_spending@var:current_pop_token = p_pops_needs_spending }
		set_variable = { pops_spending@var:current_pop_token = p_pops_spending }

		# Calculate Living Standards
		set_variable = { pops_col@var:current_pop_token = p_pops_col }

		# Calculate Taxation
		set_variable = { pops_income_tax_spending@var:current_pop_token = p_pops_income_tax_spending }
		set_variable = { pops_sales_tax_spending@var:current_pop_token = p_pops_sales_tax_spending }
		set_variable = { pops_investment_tax@var:current_pop_token = p_pops_investment_tax }

		# Calculate Debt Servicing
		set_variable = { pops_debt_interest_spending@var:current_pop_token = p_pops_debt_interest_spending }
		set_variable = { pops_debt_repayment_spending@var:current_pop_token = p_pops_debt_repayment_spending }
		
		# Calculate Credit Metrics
		set_variable = { pops_debt_spending@var:current_pop_token = p_pops_debt_spending }
		set_variable = { pops_debt_spending_ratio@var:current_pop_token = p_pops_debt_spending_ratio }
		set_variable = { pops_debt_spending_limit@var:current_pop_token = p_pops_debt_spending_limit }

		# Update Confidence
		set_variable = { pops_consumer_confidence@var:current_pop_token = p_pops_consumer_confidence }
		
		# Calc Targets
		set_variable = { pops_spending_target@var:current_pop_token = p_pops_spending_target }
		set_variable = { pops_savings_target@var:current_pop_token = p_pops_savings_target }
	}
	pops_commit_slaves = {
		set_variable = { pops@var:current_pop_token = p_pops }
		set_variable = { pops_percentage@var:current_pop_token = p_pops_percentage }
		set_variable = { pops_unemployed@var:current_pop_token = p_pops_unemployed }
	}

	log_pop_vals = {
		set_temp_variable = { current_pop_token = token:working_class }
		log = "Pop Values for [?current_pop_token.GetTokenKey]:"
		log = "pops: [?pops@var:current_pop_token]"
		log = "pops_percentage: [?pops_percentage@var:current_pop_token]"
		log = "pops_unemployed: [?pops_unemployed@var:current_pop_token]"
		log = "pops_earnings: [?pops_earnings@var:current_pop_token]"
		log = "pops_investment: [?pops_investment@var:current_pop_token]"
		log = "pops_col: [?pops_col@var:current_pop_token]"
		log = "pops_spending: [?pops_spending@var:current_pop_token]"
		log = "pops_income_tax_spending: [?pops_income_tax_spending@var:current_pop_token]"
		log = "pops_sales_tax_spending: [?pops_sales_tax_spending@var:current_pop_token]"
		log = "pops_investment_tax: [?pops_investment_tax@var:current_pop_token]"
		log = "pops_debt_interest_spending: [?pops_debt_interest_spending@var:current_pop_token]"
		log = "pops_debt_repayment_spending: [?pops_debt_repayment_spending@var:current_pop_token]"
		log = "pops_debt_spending: [?pops_debt_spending@var:current_pop_token]"
		log = "pops_debt_spending_ratio: [?pops_debt_spending_ratio@var:current_pop_token]"
		log = "pops_debt_spending_limit: [?pops_debt_spending_limit@var:current_pop_token]"
		log = "pops_consumer_confidence: [?pops_consumer_confidence@var:current_pop_token]"
		log = "pops_spending_target: [?pops_spending_target@var:current_pop_token]"
		log = "pops_savings_target: [?pops_savings_target@var:current_pop_token]"
	}

	pops_reset_core_variable = {
		set_variable = { pops_workforce_percentage = global.pops_workforce_participation }
		
		set_variable = { pops = pops_workforce }
		divide_variable = { pops = global.pops_workforce_participation }

		for_each_loop = {
			array = global.pop_needs_resources
			value = pop_r_token

			set_variable = { pops_need@var:pop_r_token = 0 }
		}

		set_variable = { pops_sol = 0 }
		set_variable = { pops_needs_spending = 0 }

		set_variable = { pops_earnings = 0 }
		set_variable = { pops_dividends = 0 }
		set_variable = { pops_investment = 0 }
		set_variable = { pops_spending_target = 0 }
		set_variable = { pops_savings_target = 0 }
		set_variable = { pops_creditworthiness = 0 }

		set_variable = { pops_income_tax_spending = 0 }
		#set_variable = { pops_corporate_tax_spending = 0 }
		set_variable = { pops_sales_tax_spending = 0 }
		set_variable = { pops_debt_interest_spending = 0 }
		set_variable = { pops_debt_repayment_spending = 0 }
	}
	pops_prepare_contextual_multipliers = {
		# Consumer Confidence Factor
		set_temp_variable = { cons_conf_temp = controller:modifier@consumer_confidence_modifier }
		divide_temp_variable = { cons_conf_temp = 100 }
		clamp_temp_variable = { var = cons_conf_temp min = 0.2 max = 2 }
		# Banking Confidence Factor
		set_temp_variable = { bank_conf_temp = controller:modifier@banking_confidence_modifier }
		divide_temp_variable = { bank_conf_temp = 100 }
		clamp_temp_variable = { var = bank_conf_temp min = 0.0 max = 1 }

		# Earnings Limit
		set_temp_variable = { upper_earnings_limit = controller:pops_earnings_avg }
		multiply_temp_variable = { upper_earnings_limit = 3 }
		set_temp_variable = { lower_earnings_limit = controller:pops_earnings_avg }
		
		# Tax Brackets
		set_temp_variable = { upper_income_multiplier = controller:income_tax_upper_strata_percentage_actual }
		divide_temp_variable = { upper_income_multiplier = 100 }

		set_temp_variable = { middle_income_multiplier = controller:income_tax_middle_strata_percentage_actual }
		divide_temp_variable = { middle_income_multiplier = 100 }

		set_temp_variable = { lower_income_multiplier = controller:income_tax_lower_strata_percentage_actual }
		divide_temp_variable = { lower_income_multiplier = 100 }

		set_temp_variable = { sales_tax_multiplier = controller:sales_tax_percentage_actual }
		divide_temp_variable = { sales_tax_multiplier = 100 }

		set_temp_variable = { corporate_tax_multiplier = controller:corporate_tax_percentage_actual }
		divide_temp_variable = { corporate_tax_multiplier = 100 }
	}
	pops_compute_summary_metrics = {
		set_variable = { pops_employed_percentage = pops_employed }
		divide_variable = { pops_employed_percentage = pops_workforce }

		set_variable = { pops_unemployed = pops_workforce }
		subtract_from_variable = { pops_unemployed = pops_employed }
		subtract_from_variable = { pops_unemployed = pops_workforce@token:peasants }
		subtract_from_variable = { pops_unemployed = active_manpower }
		clamp_variable = { var = pops_unemployed min = 0 max = pops_workforce }

		set_variable = { pops_unemployed_percentage = pops_unemployed }
		divide_variable = { pops_unemployed_percentage = pops_workforce }

		divide_variable = { pops_creditworthiness = pops_workforce }
		multiply_variable = { pops_creditworthiness = 100 }

		set_variable = { pops_spending = pops_needs_spending }
		add_to_variable = { pops_spending = pops_income_tax_spending }
		add_to_variable = { pops_spending = pops_sales_tax_spending }
		add_to_variable = { pops_spending = pops_debt_interest_spending }
		add_to_variable = { pops_spending = pops_debt_repayment_spending }

		divide_variable = { pops_sol = pops_workforce }
	}

	pops_process_workforce_status = {
		# current_pop_token 

		# Total Amount
		set_temp_variable = { p_pops = p_pops_workforce }
		divide_temp_variable = { p_pops = pops_workforce_percentage }

		# Percentage
		set_temp_variable = { p_pops_percentage = p_pops }
		divide_temp_variable = { p_pops_percentage = pops }

		# Employment
		set_temp_variable = { p_pops_unemployed = p_pops_workforce }
		subtract_from_temp_variable = { p_pops_unemployed = p_pops_employed }

	}
	pops_calculate_earnings_and_investment = {
		# current_pop_token 

		# Base Earnings
		set_temp_variable = { p_pops_earnings = pops@var:current_pop_token }
		multiply_temp_variable = { p_pops_earnings = global.base_pop_earnings } # 0.1
		
		# Earnings from Wages
		if = { 
			limit = { check_variable = { current_pop_token = token:peasants } }
			add_to_temp_variable = { p_pops_earnings = expenses_payroll@token:subsistence_farm }
		}
		else_if = { # The upper class is paid only for senior government jobs. Owners live on dividends
			limit = { check_variable = { current_pop_token = token:upper_class } }
			set_temp_variable = { p_pops_wage_earnings = bureaucrats_employed_upper }
			multiply_temp_variable = { p_pops_wage_earnings = global.wage_multiplier@token:upper_class }
			multiply_temp_variable = { p_pops_wage_earnings = base_wage }
			add_to_temp_variable = { p_pops_earnings = p_pops_wage_earnings }
		}
		else = {
			set_temp_variable = { p_pops_wage_earnings = p_pops_employed }
			multiply_temp_variable = { p_pops_wage_earnings = global.wage_multiplier@var:current_pop_token }
			multiply_temp_variable = { p_pops_wage_earnings = base_wage }
			add_to_temp_variable = { p_pops_earnings = p_pops_wage_earnings }
		}

		# Earnings from Bonuses
		add_to_temp_variable = { p_pops_earnings = pops_bonuses@var:current_pop_token }

		# Earnings from Dividends
		set_temp_variable = { p_pops_investment = 0 }
		if = {
			limit = { check_variable = { current_pop_token = token:upper_class } }
			add_to_temp_variable = { p_pops_earnings = p_pops_dividends }

			if = {
				limit = { check_variable = { p_pops_earnings > 0 } }
				set_temp_variable = { p_pops_investment = controller:modifier@upper_class_investment_contribution_factor }
				clamp_temp_variable = { var = p_pops_investment max = 0.8 min = 0 }
				multiply_temp_variable = { p_pops_investment = p_pops_earnings }

				# Scale the diverted share by investment appetite: structural incentives x the best
				# available opportunity's attractiveness (sorted list top; up to 1 month stale - fine).
				set_temp_variable = { incentives_temp = controller:modifier@incentives_modifier }
				if = {
					limit = { check_variable = { controller:active_investment_opportunities^num > 0 } }
					set_temp_variable = { top_io_id_temp = controller:active_investment_opportunities^0 }
					set_temp_variable = { top_attractiveness_temp = controller:investment_opportunity_attractiveness@var:top_io_id_temp }
				}
				else = {
					set_temp_variable = { top_attractiveness_temp = 100 }
				}
				calculate_investment_appetite = yes
				multiply_temp_variable = { p_pops_investment = investment_appetite_temp }
				#log = "[?current_pop_token.GetTokenKey] Investment: [?p_pops_investment] / [?p_pops_earnings] | [?investment_appetite_temp]"

				subtract_from_temp_variable = { p_pops_earnings = p_pops_investment }
			}
		}

		# Other Earnings (TODO: Corruption for Bureaucrats)

	}
	pops_calculate_needs_spending = {
		# current_pop_token
		
		set_temp_variable = { sol_temp = pops_sol@var:current_pop_token }
		set_temp_variable = { pop_temp = p_pops }
		set_temp_variable = { magnitude_temp = 1 }
		get_pop_needs = yes
		set_temp_variable = { p_pops_needs_spending = needs_spending_temp }

		if = { # After-the-fact consumption modifier on goods total. E.g. peasants consuming less market goods due to subsistence farming
			limit = {
				check_variable = { current_pop_token = token:peasants }
			}
			set_temp_variable = { consumption_modifier = global.peasant_market_interaction }
		}
		else = {
			set_temp_variable = { consumption_modifier = 1 }
		}

		for_each_loop = {
			array = global.pop_needs_resources
			value = pop_r_token

			set_temp_variable = { pops_need_t = pops_need_t@var:pop_r_token }
			multiply_temp_variable = { pops_need_t = consumption_modifier }
			add_to_variable = { pops_need@var:pop_r_token = pops_need_t }
		}

		set_temp_variable = { p_pops_spending = p_pops_needs_spending }
	}
	pops_calculate_living_standards = {
		# current_pop_token 

		# Cost of Living
		set_temp_variable = { p_pops_col = p_pops_needs_spending }
		divide_temp_variable = { p_pops_col = p_pops_workforce }
	}
	pops_calculate_taxation = {
		# current_pop_token 
		
		# Income Tax
		set_temp_variable = { avg_earnings_temp = p_pops_earnings }
		divide_temp_variable = { avg_earnings_temp = p_pops }
		set_temp_variable = { p_pops_income_tax_spending = p_pops_earnings }
		if = {
			limit = {
				check_variable = { avg_earnings_temp < lower_earnings_limit }
			}
			multiply_temp_variable = { p_pops_income_tax_spending = lower_income_multiplier }
		}
		else_if = {
			limit = {
				check_variable = { avg_earnings_temp < upper_earnings_limit }
			}
			multiply_temp_variable = { p_pops_income_tax_spending = middle_income_multiplier }
		}
		else = {
			multiply_temp_variable = { p_pops_income_tax_spending = upper_income_multiplier }
		}
		add_to_temp_variable = { p_pops_spending = p_pops_income_tax_spending }
		
		#Sales Tax
		set_temp_variable = { p_pops_sales_tax_spending = p_pops_needs_spending }
		multiply_temp_variable = { p_pops_sales_tax_spending = sales_tax_multiplier }
		add_to_temp_variable = { p_pops_spending = p_pops_sales_tax_spending }
		
		#Corporate Tax
		if = {
			limit = { check_variable = { p_pops_dividends > 0 } }

			set_temp_variable = { p_pops_investment_tax = p_pops_dividends }
			multiply_temp_variable = { p_pops_investment_tax = corporate_tax_multiplier }
			if = { limit = { check_variable = { p_pops_investment_tax > p_pops_dividends } } set_temp_variable = { p_pops_investment_tax = p_pops_dividends } }
			if = { limit = { check_variable = { p_pops_investment_tax < 0 } } set_temp_variable = { p_pops_investment_tax = 0 } }
		}
	}
	pops_calculate_debt_servicing = {
		# current_pop_token 

		#Interest
		set_temp_variable = { p_pops_debt_interest_spending = controller:bank_interest_rate }
		# add_to_temp_variable = { p_pops_debt_interest_spending = pops_credit_rating_interest@var:current_pop_token }
		multiply_temp_variable = { p_pops_debt_interest_spending = p_bank_lent_to_pops_k }
		divide_temp_variable = { p_pops_debt_interest_spending = 1.2 }
		add_to_temp_variable = { p_pops_spending = p_pops_debt_interest_spending }

		# Debt Repayment
		set_temp_variable = { p_pops_debt_repayment_spending = 4 }
		multiply_temp_variable = { p_pops_debt_repayment_spending = p_bank_lent_to_pops_k }
		divide_temp_variable = { p_pops_debt_repayment_spending = 1.2 }
		add_to_temp_variable = { p_pops_spending = p_pops_debt_repayment_spending }
	}
	pops_calculate_credit_metrics = {
		# current_pop_token
		set_temp_variable = { p_pops_debt_spending = p_pops_debt_interest_spending }
		add_to_temp_variable = { p_pops_debt_spending = p_pops_debt_repayment_spending }

		set_temp_variable = { p_pops_debt_spending_ratio = p_pops_debt_spending }
		divide_temp_variable = { p_pops_debt_spending_ratio = p_pops_earnings }

		# Use global banking_confidence & consumer_confidence as temp vars if needed
		set_temp_variable = { p_pops_debt_spending_limit = 0.1 }

		set_temp_variable = { debt_limit_temp = bank_conf_temp }
		multiply_temp_variable = { debt_limit_temp = 0.05 }
		add_to_temp_variable = { p_pops_debt_spending_limit = limit_temp }

		set_temp_variable = { debt_limit_temp = cons_conf_temp }
		multiply_temp_variable = { debt_limit_temp = 0.025 }
		add_to_temp_variable = { p_pops_debt_spending_limit = debt_limit_temp }

		# Creditworthiness
		set_temp_variable = { income_temp = p_pops_earnings }
		set_temp_variable = { debt_temp = p_bank_lent_to_pops_k }
		set_temp_variable = { debt_spending_temp = p_pops_debt_spending }
		set_temp_variable = { savings_temp = p_pops_savings }
		calculate_creditworthiness = yes
		set_variable = { pops_creditworthiness@var:current_pop_token = creditworthiness_temp }

		set_temp_variable = { weighted_credit = creditworthiness_temp }
		divide_temp_variable = { weighted_credit = 100 }
		multiply_temp_variable = { weighted_credit = p_pops_workforce }
		add_to_variable = { pops_creditworthiness = weighted_credit }
	}
	pops_update_confidence = {
		# Base from macro conditions (From 0.2 to 2.0)
		set_temp_variable = { p_pops_consumer_confidence = cons_conf_temp }

		# Debt Spending Penalty (Up to -1.0)
		subtract_from_temp_variable = { p_pops_consumer_confidence = p_pops_debt_spending_ratio }

		# Savings Bonus (Up to +0.12)
		set_temp_variable = { temp_conf_delta = p_pops_savings }
		divide_temp_variable = { temp_conf_delta = p_pops_earnings }
		multiply_temp_variable = { temp_conf_delta = 0.01 }
		add_to_temp_variable = { p_pops_consumer_confidence = temp_conf_delta }

		# Debt Panic Penalty (Limitless, but 100% excess -> -0.55)
		set_temp_variable = { excess_ratio = p_pops_debt_spending_ratio }
		divide_temp_variable = { excess_ratio = p_pops_debt_spending_limit }
		if = {
			limit = { check_variable = { excess_ratio > 1 } }
			set_temp_variable = { temp_penalty = excess_ratio }
			subtract_from_temp_variable = { temp_penalty = 1 }
			multiply_temp_variable = { temp_penalty = 0.5 }
			add_to_temp_variable = { temp_penalty = 0.05 }
			subtract_from_temp_variable = { p_pops_consumer_confidence = temp_penalty }
		}

		# Interest Rate Shock (every 12% excess interest -> -0.25)
		set_temp_variable = { temp_interest = bank_interest_rate }
		subtract_from_temp_variable = { temp_interest = 12 }  # 1% monthly
		if = {
			limit = { check_variable = { temp_interest > 0 } }
			set_temp_variable = { temp_penalty = temp_interest }
			divide_temp_variable = { temp_penalty = 12 }
			multiply_temp_variable = { temp_penalty = 0.25 }
			subtract_from_temp_variable = { p_pops_consumer_confidence = temp_penalty }
		}

		# Clamp 0.2–2
		clamp_temp_variable = {
			var = p_pops_consumer_confidence
			min = 0.2
			max = 2
		}
	}
	pops_calculate_spending_targets = {
		# current_pop_token 
		
		#Spending Target
		set_temp_variable = { p_pops_spending_target = p_pops_earnings }
		multiply_temp_variable = { p_pops_spending_target = p_pops_consumer_confidence }
	}
	pops_calculate_savings_targets = {
		# current_pop_token
		set_temp_variable = { p_pops_savings_target = p_pops_earnings }
		multiply_temp_variable = { p_pops_savings_target = bank_conf_temp }
		multiply_temp_variable = { p_pops_savings_target = 12 }
	}
	pops_add_to_state_totals = {
		# current_pop_token 
		
		set_temp_variable = { sol_temp = pops_sol@var:current_pop_token }
		multiply_temp_variable = { sol_temp = p_pops_workforce }
		add_to_variable = { pops_sol = sol_temp }
		add_to_variable = { pops_needs_spending = p_pops_needs_spending }
		
		add_to_variable = { pops_earnings = p_pops_earnings }
		add_to_variable = { pops_investment = p_pops_investment }
		add_to_variable = { pops_spending_target = p_pops_spending_target }
		add_to_variable = { pops_savings_target = p_pops_savings_target }

	employment_update = {
		# Updates industry jobs, job fulfillment, and pop and industry employment
		# Jobs per class: each industry's Wrk and Mid shares go to the working and middle classes, and its owner places
		# (about 2% of its jobs, owner_jobs@) to the upper class (Q11). Bureaucratic jobs (bureaucracy_and_manpower_update)
		# are middle class, plus senior ones in the capital for the upper class (Q13, Q58). Slaves fill working-class jobs
		# before free workers (Q12).

		## JOBS
		set_temp_variable = { e_working_jobs = 0 }
		set_temp_variable = { e_working_jobs_available = 0 }

		set_temp_variable = { e_middle_jobs = bureaucratic_jobs }
		set_temp_variable = { e_middle_jobs_available = bureaucratic_jobs }

		set_temp_variable = { e_upper_jobs = bureaucratic_jobs_upper }

		# Industries
		for_each_loop = {
			array = industries
			value = bu_token

			if = {
				limit = { NOT = { check_variable = { bu_token = token:subsistence_farm } } }
				industry_update_size_jobs = yes

				add_to_temp_variable = { e_working_jobs = i_working_jobs }
				add_to_temp_variable = { e_working_jobs_available = i_working_jobs_available }

				add_to_temp_variable = { e_middle_jobs = i_middle_jobs }
				add_to_temp_variable = { e_middle_jobs_available = i_middle_jobs_available }
			}

			add_to_temp_variable = { e_upper_jobs = owner_jobs@var:bu_token } # Owner places already follow the employment target
		}

		# Pops. Slaves take working-class jobs first, and free workers get the rest
		set_variable = { pops_jobs@token:slaves = { value = pops_workforce@token:slaves min = e_working_jobs } }
		set_variable = { pops_jobs_available@token:slaves = { value = pops_workforce@token:slaves min = e_working_jobs_available } }
		set_variable = { pops_jobs@token:working_class = { value = e_working_jobs subtract = pops_jobs@token:slaves } }
		set_variable = { pops_jobs_available@token:working_class = { value = e_working_jobs_available subtract = pops_jobs_available@token:slaves } }
		set_variable = { pops_jobs@token:middle_class = e_middle_jobs }
		set_variable = { pops_jobs_available@token:middle_class = e_middle_jobs_available }
		set_variable = { pops_jobs@token:upper_class = e_upper_jobs }
		set_variable = { pops_jobs_available@token:upper_class = e_upper_jobs }

		## JOB FULFILLMENT
		set_variable = { pops_jobs_available = 0 }
		set_variable = { pops_jobs = 0 }
		set_variable = { pops_employed = 0 }
		for_each_loop = {
			array = global.pops_array
			value = pop_token

			if = {
				limit = { NOT = { check_variable = { pop_token = token:peasants } } }

				set_temp_variable = { e_pops_jobs_available = pops_jobs_available@var:pop_token }

				set_variable = { pops_workforce_balance@var:pop_token = { value = pops_workforce@var:pop_token subtract = e_pops_jobs_available } }
				set_variable = { pops_employed@var:pop_token = { value = pops_workforce@var:pop_token min = e_pops_jobs_available } }
				if = {
					limit = { check_variable = { e_pops_jobs_available > 0 } }
					set_variable = { job_fulfillment@var:pop_token = { value = pops_employed@var:pop_token multiply = 100 divide = e_pops_jobs_available } }
				}
				else = {
					set_variable = { job_fulfillment@var:pop_token = 0 }
				}

				add_to_variable = { pops_jobs_available = e_pops_jobs_available }
				add_to_variable = { pops_jobs = pops_jobs@var:pop_token }
				add_to_variable = { pops_employed = pops_employed@var:pop_token }
			}
		}

		# Peasants all work, on subsistence land or marginal land
		set_variable = { pops_employed@token:peasants = pops_workforce@token:peasants }
		set_variable = { pops_unemployed@token:peasants = 0 }

		set_variable = { available_workforce = { value = pops_workforce subtract = active_manpower subtract = pops_jobs_available } }

		# Working-class jobs as a whole, slaves and free workers together, for industry employment
		set_temp_variable = { e_working_employed = { value = pops_employed@token:slaves add = pops_employed@token:working_class } }
		if = {
			limit = { check_variable = { e_working_jobs_available > 0 } }
			set_variable = { working_job_fulfillment = { value = e_working_employed multiply = 100 divide = e_working_jobs_available } }
		}
		else = {
			set_variable = { working_job_fulfillment = 0 }
		}

		# The enslaved share of working-class labour. Employers pay no wage for it (industry_update)
		if = {
			limit = { check_variable = { pops_employed@token:slaves > 0 } }
			set_variable = { slave_labor_share = { value = pops_employed@token:slaves divide = e_working_employed } }
		}
		else = {
			set_variable = { slave_labor_share = 0 }
		}

		# Bureaucrats. The government pays them at the state's wage, through administration spending (budget_update)
		set_variable = { bureaucrats_employed = { value = bureaucratic_jobs multiply = job_fulfillment@token:middle_class divide = 100 } }
		set_variable = { bureaucrats_employed_upper = { value = bureaucratic_jobs_upper multiply = job_fulfillment@token:upper_class divide = 100 } }
		set_variable = {
			bureaucrats_payroll = {
				value = bureaucrats_employed
				multiply = global.wage_multiplier@token:middle_class
				add = {
					value = bureaucrats_employed_upper
					multiply = global.wage_multiplier@token:upper_class
				}
				multiply = base_wage
			}
		}

		## LABOR UTILIZATION (drives floating wages in base_wage_update)
		# Slaves, the working class and the middle class, against private jobs. Slaves count in the supply (Q62), so plentiful
		# slaves hold free wages down. Bureaucrats are left out on both sides: the government pays the going wage and doesn't
		# bid it up, and states with only bureaucrats keep a steady wage, as they did before M02.
		set_variable = {
			industrial_workforce = {
				value = pops_workforce@token:slaves
				add = pops_workforce@token:working_class
				add = pops_workforce@token:middle_class
				subtract = bureaucrats_employed
			}
		}
		set_variable = {
			industrial_jobs_available = {
				value = e_working_jobs_available
				add = e_middle_jobs_available
				subtract = bureaucratic_jobs
			}
		}

		# Utilization = available jobs / workforce (effective employment/availability ratio).
		# Neutral 0.97 (mid stable band) when there is no industrial workforce.
		if = {
			limit = { check_variable = { industrial_workforce > 0 } }
			set_variable = { labor_utilization = industrial_jobs_available }
			divide_variable = { labor_utilization = industrial_workforce }
		}
		else = {
			set_variable = { labor_utilization = 0.97 }
		}

		## INDUSTRY EMPLOYMENT
		for_each_loop = {
			array = industries
			value = bu_token

			if = {
				limit = { NOT = { check_variable = { bu_token = token:subsistence_farm } } }
				industry_update_employment = yes
			}
		}

		set_variable = { total_industry_size = 0 }
		for_each_loop = {
			array = buildings
			value = pops_bld_token
			add_to_variable = { total_industry_size = size@var:pops_bld_token }
		}
		add_to_variable = { total_industry_size = size@token:railway }
		add_to_variable = { total_industry_size = size@token:port }
		add_to_variable = { total_industry_size = size@token:industrial_complex }
		add_to_variable = { total_industry_size = size@token:arms_factory }
		add_to_variable = { total_industry_size = size@token:dockyard }
	}

	pops_balance = {
		# Moves pops between the working, middle and upper classes toward the jobs each offers (Q10). Total population never
		# changes here. Slaves never move, and peasants join or leave the working class in pops_peasant_job_migration.
		# Classes short of workers take them from classes with too many, in proportion. Moving up into the middle and upper
		# classes needs free qualifications (Q57), and the whole move is scaled down to fit them. Moving down loses none (Q74).

		set_temp_variable = { change_working = { value = pops_workforce_balance@token:working_class multiply = -1 } }
		set_temp_variable = { change_middle = { value = pops_workforce_balance@token:middle_class multiply = -1 } }
		set_temp_variable = { change_upper = { value = pops_workforce_balance@token:upper_class multiply = -1 } }

		# Positive changes are shortfalls, negative ones surpluses. The number who move is the smaller total of the two
		set_temp_variable = { shortfall_temp = { value = change_working max = 0 add = { value = change_middle max = 0 } add = { value = change_upper max = 0 } } }
		set_temp_variable = { surplus_temp = { value = change_working min = 0 add = { value = change_middle min = 0 } add = { value = change_upper min = 0 } multiply = -1 } }

		if = {
			limit = {
				check_variable = { shortfall_temp > 0 }
				check_variable = { surplus_temp > 0 }
			}
			set_temp_variable = { shortfall_scale_temp = { value = shortfall_temp min = surplus_temp divide = shortfall_temp } }
			set_temp_variable = { surplus_scale_temp = { value = shortfall_temp min = surplus_temp divide = surplus_temp } }

			if = { limit = { check_variable = { change_working > 0 } } multiply_temp_variable = { change_working = shortfall_scale_temp } }
			else = { multiply_temp_variable = { change_working = surplus_scale_temp } }
			if = { limit = { check_variable = { change_middle > 0 } } multiply_temp_variable = { change_middle = shortfall_scale_temp } }
			else = { multiply_temp_variable = { change_middle = surplus_scale_temp } }
			if = { limit = { check_variable = { change_upper > 0 } } multiply_temp_variable = { change_upper = shortfall_scale_temp } }
			else = { multiply_temp_variable = { change_upper = surplus_scale_temp } }

			# Qualifications the move needs, against those the middle and upper classes don't already hold
			set_temp_variable = {
				qualifications_change_temp = {
					value = change_middle
					multiply = global.qualifications_need@token:middle_class
					add = {
						value = change_upper
						multiply = global.qualifications_need@token:upper_class
					}
				}
			}
			if = {
				limit = { check_variable = { qualifications_change_temp > 0 } }
				set_temp_variable = {
					qualifications_free_temp = {
						value = qualifications
						subtract = { value = pops_workforce@token:middle_class multiply = global.qualifications_need@token:middle_class }
						subtract = { value = pops_workforce@token:upper_class multiply = global.qualifications_need@token:upper_class }
						max = 0
					}
				}
				if = {
					limit = { check_variable = { qualifications_change_temp > qualifications_free_temp } }
					set_temp_variable = { qualifications_scale_temp = { value = qualifications_free_temp divide = qualifications_change_temp } }
					multiply_temp_variable = { change_middle = qualifications_scale_temp }
					multiply_temp_variable = { change_upper = qualifications_scale_temp }
				}
			}

			# The working class takes the other side of every move, so the total is exact
			set_temp_variable = { change_working = { value = change_middle add = change_upper multiply = -1 } }

			add_to_variable = { pops_workforce@token:working_class = change_working }
			add_to_variable = { pops_workforce@token:middle_class = change_middle }
			add_to_variable = { pops_workforce@token:upper_class = change_upper }
		}
	}

	bureaucracy_and_manpower_update = {
		# Country scope, before its states update. Sets each controlled state's bureaucratic jobs, and the active manpower
		# per head of population that pops_active_manpower_monthly draws from the states.
		# Bureaucratic jobs (Q13, Q58): population x global.bureaucrats_per_pop x (1 + bureaucratic_strain_factor), with
		# non-core states at global.bureaucrats_non_core_weight. The capital adds senior jobs for the upper class.
		set_temp_variable = { bureaucrats_per_pop_temp = { value = 1 add = modifier@bureaucratic_strain_factor multiply = global.bureaucrats_per_pop } }
		set_temp_variable = { population_temp = 0 }
		set_variable = { bureaucratic_jobs_required = 0 }
		every_controlled_state = {
			set_variable = { bureaucratic_jobs = state_population_k }
			multiply_variable = { bureaucratic_jobs = bureaucrats_per_pop_temp }
			if = {
				limit = { NOT = { is_core_of = PREV } }
				multiply_variable = { bureaucratic_jobs = global.bureaucrats_non_core_weight }
			}
			clear_variable = bureaucratic_jobs_upper

			add_to_variable = { PREV.bureaucratic_jobs_required = bureaucratic_jobs }
			add_to_temp_variable = { population_temp = state_population_k }
		}
		set_temp_variable = { bureaucrats_upper_temp = { value = bureaucratic_jobs_required multiply = global.bureaucrats_upper_share } }
		capital_scope = {
			if = {
				limit = { is_controlled_by = PREV }
				set_variable = { bureaucratic_jobs_upper = bureaucrats_upper_temp }
				add_to_variable = { PREV.bureaucratic_jobs_required = bureaucrats_upper_temp }
			}
		}

		# Active manpower (Q14): HOI4's fielded and training manpower. The deployment queue is counted in people, not thousands
		set_variable = { active_manpower_k = deployed_army_manpower_k }
		add_to_variable = { active_manpower_k = deployed_navy_manpower_k }
		add_to_variable = { active_manpower_k = deployed_airforce_manpower_k }
		set_temp_variable = { training_temp = amount_manpower_in_deployment_queue }
		divide_temp_variable = { training_temp = 1000 }
		add_to_variable = { active_manpower_k = training_temp }

		set_variable = { active_manpower_per_pop = active_manpower_k }
		divide_variable = { active_manpower_per_pop = population_temp }
	}

	pops_slaves_monthly = {
		# State scope. Frees slaves above the cap: the state's share in marginalised cultures, or none where the controller
		# bans slavery (slavery_banned, Q61). Freed slaves join the working class. Nobody is newly enslaved (Q73).
		# Input: marginalized_share_temp (pops_culture_shares_calc)
		if = {
			limit = { check_variable = { pops_workforce@token:slaves > 0 } }

			set_temp_variable = { slaves_cap_temp = { value = pops_workforce multiply = marginalized_share_temp } }
			if = {
				limit = { check_variable = { controller:modifier@slavery_banned > 0 } }
				set_temp_variable = { slaves_cap_temp = 0 }
			}
			if = {
				limit = { check_variable = { pops_workforce@token:slaves > slaves_cap_temp } }
				set_temp_variable = { freed_temp = { value = pops_workforce@token:slaves subtract = slaves_cap_temp } }
				subtract_from_variable = { pops_workforce@token:slaves = freed_temp }
				add_to_variable = { pops_workforce@token:working_class = freed_temp }
				add_to_variable = { pops_workforce_balance@token:working_class = freed_temp }
			}
		}

	pops_active_manpower_monthly = {
		# State scope. Moves pops between the civilian classes and active manpower (Q14): the state's share of its controller's
		# active manpower, by population, drawn from peasants and the working class in proportion to their numbers.
		# Demobilised soldiers go back the same way. Soldiers' pay and upkeep stay on the military budget lines.
		set_temp_variable = { draft_pool_temp = { value = pops_workforce@token:peasants add = pops_workforce@token:working_class } }

		set_temp_variable = { draft_temp = state_population_k }
		multiply_temp_variable = { draft_temp = controller:active_manpower_per_pop }
		subtract_from_temp_variable = { draft_temp = active_manpower }
		if = { limit = { check_variable = { draft_temp > draft_pool_temp } } set_temp_variable = { draft_temp = draft_pool_temp } }

		if = {
			limit = { NOT = { check_variable = { draft_temp = 0 } } }

			if = {
				limit = { check_variable = { draft_pool_temp > 0 } }
				set_temp_variable = { draft_working_temp = { value = draft_temp multiply = pops_workforce@token:working_class divide = draft_pool_temp } }
			}
			else = {
				set_temp_variable = { draft_working_temp = 0 } # Nobody left to draw from: demobilised soldiers all return as peasants
			}
			subtract_from_variable = { pops_workforce@token:working_class = draft_working_temp }
			subtract_from_variable = { pops_workforce_balance@token:working_class = draft_working_temp }
			subtract_from_variable = { pops_workforce@token:peasants = draft_temp }
			add_to_variable = { pops_workforce@token:peasants = draft_working_temp }
			add_to_variable = { active_manpower = draft_temp }
		}
	}
	pops_qualifications_monthly = {
		# State scope. Advances the state's qualifications stock, in K qualified people (Q57). The middle and upper classes
		# need global.qualifications_need@<class> per head, and pops_balance moves people up only into what's free.
		# Gains a month per K pops, from Strat's notepad: (literacy - 10%) x 0.3, (schooling - 5%) x 0.5, education x 0.3,
		# (wealth - 10) x 0.005, and the middle and upper classes' share of the workforce x 0.2, all x global.qualifications_gain_scale.
		# They're cut for marginalised and non-accepted cultures (Q76), and for peasants by serfdom laws. Slaves gain nothing.
		# Losses: global.qualifications_loss_rate of the stock a month, for deaths and decay.
		# Input: marginalized_share_temp, non_accepted_share_temp (pops_culture_shares_calc)
		if = {
			limit = { check_variable = { pops_workforce > 0 } }

			# Schooling and the workforce mix are the same for every class
			set_temp_variable = {
				qualifications_base_temp = {
					value = pops_literacy_percentage
					subtract = 10
					max = 0
					multiply = 0.003
					add = { value = pops_schooling_percentage subtract = 5 max = 0 multiply = 0.005 }
					add = { value = pops_education_percentage multiply = 0.003 }
					add = { value = pops_workforce@token:middle_class add = pops_workforce@token:upper_class divide = pops_workforce multiply = 0.2 }
				}
			}

			set_temp_variable = { qualifications_gain_temp = 0 }
			for_each_loop = {
				array = global.pops_array
				value = pop_token

				if = {
					limit = {
						check_variable = { pops_workforce@var:pop_token > 0 }
						NOT = { check_variable = { pop_token = token:slaves } }
					}
					set_temp_variable = {
						class_gain_temp = {
							value = pops_sol@var:pop_token
							subtract = 10
							max = 0
							multiply = 0.005
							add = qualifications_base_temp
							multiply = pops_workforce@var:pop_token
						}
					}
					if = {
						limit = { check_variable = { pop_token = token:peasants } }
						set_temp_variable = { serfdom_temp = { value = 1 add = controller:modifier@peasant_qualifications_factor max = 0 } }
						multiply_temp_variable = { class_gain_temp = serfdom_temp }
					}
					add_to_temp_variable = { qualifications_gain_temp = class_gain_temp }
				}
			}

			set_variable = {
				qualifications = {
					value = qualifications
					multiply = { value = 1 subtract = global.qualifications_loss_rate }
					add = {
						value = qualifications_gain_temp
						multiply = global.qualifications_gain_scale
						multiply = {
							value = 1
							add = {
								value = global.qualifications_marginalized_factor
								add = controller:modifier@marginalized_culture_qualifications_factor
								clamp = { min = -1 max = 0 }
								multiply = marginalized_share_temp
							}
							add = {
								value = global.qualifications_non_accepted_factor
								add = controller:modifier@non_accepted_culture_qualifications_factor
								clamp = { min = -1 max = 0 }
								multiply = non_accepted_share_temp
							}
						}
					}
				}
			}
		}
	}

	create_pie_chart_pop_type = {
		# this creates a pie chart of popularities - can be easily adapted to show anything that adds up to 100%
		# This code is much different from Yard's, but I'll credit him anyway
		# State Scope

		clear_array = pie_chart_pop_type
		resize_array = {
			array = pie_chart_pop_type
			value = 0 # Not set yet. Classes count from 1 (slaves), so 0 can't be mistaken for one
			size = 100 # 100 pieces, each percent is one piece
		}

		clear_temp_array = pop_types_percentage
		add_to_temp_array = { pop_types_percentage = 0 }
		for_each_loop = {
			array = global.pops_array
			value = pop_token
			add_to_temp_array = { pop_types_percentage = pops_percentage@var:pop_token }
		}

		set_temp_variable = { total_temp = 0 }
		set_temp_variable = { largest = 0 }
		for_each_loop = {
			array = pop_types_percentage
			index = pie_ind

			multiply_temp_variable = { pop_types_percentage^pie_ind = 100 }
			round_temp_variable = pop_types_percentage^pie_ind
			add_to_temp_variable = { total_temp = pop_types_percentage^pie_ind }

			# The rounding remainder goes to the largest class. Both sides are in pieces by now
			if = {
				limit = {
					check_variable = { pop_types_percentage^pie_ind > pop_types_percentage^largest }
				}
				set_temp_variable = { largest = pie_ind }
			}
		}

		if = {
			limit = {
				NOT = { check_variable = { total_temp = 100 } }
			}
			add_to_temp_variable = { pop_types_percentage^largest = 100 }
			subtract_from_temp_variable = { pop_types_percentage^largest = total_temp }
		}

		set_temp_variable = { pie_pop = 0 }
		for_each_loop = {
			array = pop_types_percentage
			index = pie_ind
			value = pie_v


			set_variable = { last_idx = pie_pop }
			set_temp_variable = { pie_pop = pie_v }

			add_to_temp_variable = { pie_pop = last_idx }
			#subtract_from_temp_variable = { pie_pop = 1 }
			if = {
				limit = {
					check_variable = { pie_pop > 100 }
				}
				set_temp_variable = { pie_pop = 100 }
			}
			if = {
				limit = {
					check_variable = { last_idx < 100 }
				}
				set_variable = { pie_chart_pop_type^last_idx = pie_ind }
			}
			set_variable = { last_idx = pie_pop }
		}
		clear_variable = last_idx
		clear_variable = pie_chart_v

		# Paradox bugginess forced me to make this a separate loop rather than put it in the above loop
		# Each piece not set yet takes the class of the piece before it. Piece 0 always holds the first class
		set_temp_variable = { pie_chart_idx = 1 }
		while_loop_effect = {
			limit = {
				check_variable = { pie_chart_idx < 100 }
			}
			if = {
				limit = {
					check_variable = { pie_chart_pop_type^pie_chart_idx = 0 }
				}
				set_temp_variable = { prev_idx = pie_chart_idx }
				subtract_from_temp_variable = { prev_idx = 1 }
				set_variable = { pie_chart_pop_type^pie_chart_idx = pie_chart_pop_type^prev_idx }
			}
			add_to_temp_variable = { pie_chart_idx = 1 }
		}

		add_to_variable = { update_state_econ_window = 1 }
	}

	remove_debt_internal = {
		# Takes in repay_temp_k and country_temp and removes that amount of money from that.
		# Outputs repay_temp_k which is the amount of debt actually repaid

		if = {
			limit = {
				check_variable = { repay_temp_k > debt_to_bank_k@var:country_temp }
			}
			set_temp_variable = { repay_temp_k = debt_to_bank_k@var:country_temp }
		}
		#log = "Repay Amount: [?repay_temp_k]K"
		
		if = {
			limit = {
				check_variable = { repay_temp_k < 0 }
			}
			set_temp_variable = { repay_temp_k = 0 }
		}
		else = {
			if = {
				limit = {
					check_variable = { repay_temp_k < debt_to_bank_k@var:country_temp }
				}
				subtract_from_variable = { debt_to_bank_k@var:country_temp = repay_temp_k }
				subtract_from_variable = { country_temp:money_lent_k@THIS = repay_temp_k }
			}
			else = {
				remove_from_array = { array = debt_to_bank_country value = country_temp }
				remove_from_array = { array = country_temp:money_lent_country value = THIS }
				clear_variable = debt_to_bank_k@var:country_temp
				clear_variable = country_temp:money_lent_k@THIS
			}
			#log = "Debt to Bank K ([THIS.GetName]): [?debt_to_bank_k@var:country_temp]K"
			#log = "Money Lent K ([?country_temp.GetName]): [?country_temp:money_lent_k@THIS]K"

			if = {
				limit = {
					NOT = { check_variable = { country_temp = THIS } }
				}
				var:country_temp = {
					national_debt_update = yes
				}
			}
		}
	}

	gdp_update = {
		#State
		set_variable = { gdp_amount = 0 }
		for_each_loop = {
			array = global.resource_token_array
			value = r_token

			set_temp_variable = { gdp_temp = production@var:r_token }
			subtract_from_temp_variable = { gdp_temp = demand_input@var:r_token }
			multiply_temp_variable = { gdp_temp = controller:price@var:r_token }
			add_to_variable = { gdp_amount = gdp_temp }
		}
		# Subsistence Farming
		for_each_loop = {
			array = global.building_base_production@token:subsistence_farm
			value = r_token
			index = sub_i_temp

			set_temp_variable = { sub_temp = production@token:subsistence_farm^sub_i_temp }
			divide_temp_variable = { sub_temp = global.peasant_market_interaction }

			set_temp_variable = { market_reach_temp = 1 }
			subtract_from_temp_variable = { market_reach_temp = global.peasant_market_interaction }
			multiply_temp_variable = { sub_temp = market_reach_temp }

			multiply_temp_variable = { sub_temp = global.base_price@var:r_token }
			add_to_variable = { gdp_amount = sub_temp }
		}

		# Railways & ports: nobody buys transport, so what they're paid counts as value added
		add_to_variable = { gdp_amount = revenue_industrial_capacity@railway }
		add_to_variable = { gdp_amount = revenue_shipping_capacity@port }

		# Special
		set_temp_variable = { gdp_temp = productivity@industrial_complex }
		multiply_temp_variable = { gdp_temp = payroll_multiplier@industrial_complex }
		multiply_temp_variable = { gdp_temp = production_percentage@industrial_complex }
		divide_temp_variable = { gdp_temp = 100 }
		multiply_temp_variable = { gdp_temp = size@industrial_complex }
		add_to_variable = { gdp_amount = gdp_temp }
		
		set_temp_variable = { gdp_temp = productivity@arms_factory }
		multiply_temp_variable = { gdp_temp = payroll_multiplier@arms_factory }
		multiply_temp_variable = { gdp_temp = production_percentage@arms_factory }
		divide_temp_variable = { gdp_temp = 100 }
		multiply_temp_variable = { gdp_temp = size@arms_factory }
		add_to_variable = { gdp_amount = gdp_temp }
		
		set_temp_variable = { gdp_temp = productivity@dockyard }
		multiply_temp_variable = { gdp_temp = payroll_multiplier@dockyard }
		multiply_temp_variable = { gdp_temp = production_percentage@dockyard }
		divide_temp_variable = { gdp_temp = 100 }
		multiply_temp_variable = { gdp_temp = size@dockyard }
		add_to_variable = { gdp_amount = gdp_temp }


		# GDP per Capita
		set_variable = { gdp_per_capita = gdp_amount }
		divide_variable = { gdp_per_capita = pops }
		if = {
			limit = {
				check_variable = { gdp_per_capita > global.highest_state_gdp_c }
			}
			set_variable = { global.highest_state_gdp_c = gdp_per_capita }
		}
		if = {
			limit = {
				check_variable = { gdp_amount > global.highest_state_gdp }
			}
			set_variable = { global.highest_state_gdp = gdp_amount }
		}
	}
	country_calc_gdp = {
		set_variable = { gdp_amount = 0 }

		every_owned_state = {
			gdp_update = yes

			add_to_variable = { PREV.gdp_amount = gdp_amount }
		}

		divide_variable = { gdp_per_capita = pops }
	}
	market_calc_gdp = {
		if = {
			limit = {
				check_variable = { market_leader = THIS }
				check_variable = { market_countries^num > 1 }
			}
			set_variable = { market_gdp = 0 }
			set_variable = { market_pops = 0 }
			set_temp_variable = { home_temp = THIS }
			for_each_scope_loop = {
				array = market_countries
				country_calc_gdp = yes
				add_to_variable = { home_temp:market_gdp = gdp_amount }
				add_to_variable = { home_temp:market_pops = pops }
			}
			set_variable = { market_gdp_per_capita = market_gdp }
			divide_variable = { market_gdp_per_capita = market_pops }
		}
		else_if = {
			limit = {
				NOT = { check_variable = { market_leader = THIS } }
			}
			set_variable = { market_gdp = market_leader:market_gdp }
			set_variable = { market_pops = market_leader:market_pops }
			set_variable = { market_gdp_per_capita = market_leader:market_gdp_per_capita }
		}
		else = {
			country_calc_gdp = yes
			set_variable = { market_gdp = gdp_amount }
			set_variable = { market_pops = pops }
			set_variable = { market_gdp_per_capita = gdp_per_capita }
		}
	}

	econ_rerender_gdp_graph = {
		# Code courtesy of TNO (totally not stolen. Sorry, couldn't figure out the graph line thing on my own. I can make complicated economy systems but graphs? Nah).
		# Find the maximum and minimum GDP values for the graph
		set_temp_variable = { min = gdp_amount_array^0 }
		set_temp_variable = { max = gdp_amount_array^0 }
		for_loop_effect = {
			start = 0
			end = gdp_amount_array^num
			value = i

			if = {
				limit = { check_variable = { gdp_amount_array^i > max } }
				set_temp_variable = { max = gdp_amount_array^i }
			}
			if = {
				limit = { check_variable = { gdp_amount_array^i < min } }
				set_temp_variable = { min = gdp_amount_array^i }
			}
		}

		# Our range is (min value / 1.1) - (max value * 1.1)
		multiply_temp_variable = { max = 1.05 }
		divide_temp_variable = { min = 1.05 }


		set_temp_variable = { range = max }
		subtract_from_temp_variable = { range = min }
		divide_temp_variable = { range = 100 }

		# Get all segments Progress Bar values
		clear_array = gdp_graph_frames
		for_loop_effect = {
			start = 1
			end = gdp_amount_array^num
			value = i

			set_temp_variable = { j = i }
			add_to_temp_variable = { j = -1 }

			set_temp_variable = { hi = gdp_amount_array^i }
			set_temp_variable = { lo = gdp_amount_array^j }


			subtract_from_temp_variable = { hi = min }
			subtract_from_temp_variable = { lo = min }
			divide_temp_variable = { hi = range }
			divide_temp_variable = { lo = range }

			round_temp_variable = hi
			round_temp_variable = lo

			set_temp_variable = { endpoints = lo }
			multiply_temp_variable = { endpoints = 1000 }
			add_to_temp_variable = { endpoints = hi }
			add_to_array = { gdp_graph_frames = endpoints }
		}
		
		#Correcting the max/min
		set_variable = { gdp_graph_min = min }
		set_variable = { gdp_graph_max = max }
		if = {
			limit = {
				check_variable = { gdp_graph_max > 1000 } #gdp_graph_min should always be less than max, so there's no point to checking it
			}
			divide_variable = { gdp_graph_min = 1000 }
			divide_variable = { gdp_graph_max = 1000 }
			set_state_flag = gdp_graph_thousands
		}
		else_if = {
			limit = {
				check_variable = { gdp_graph_max > 100 }
			}
			divide_variable = { gdp_graph_min = 100 } #This gets rid of some decimals so there's only up to 4 digits
			divide_variable = { gdp_graph_max = 100 }
			multiply_variable = { gdp_graph_min = 100 }
			multiply_variable = { gdp_graph_max = 100 }
		}
		else_if = {
			limit = {
				check_variable = { gdp_graph_max > 10 }
			}
			divide_variable = { gdp_graph_min = 10 } #This gets rid of some decimals so there's only up to 4 digits
			divide_variable = { gdp_graph_max = 10 }
			multiply_variable = { gdp_graph_min = 10 }
			multiply_variable = { gdp_graph_max = 10 }
		}

		add_to_variable = { update_state_econ_window = 1 }
	}

		# Creates a list of buildings under construction in your country, sorted by completion time.

		clear_array = under_construction_list_state

				array = buildings
				value = building_token

						check_variable = { construction_speed@var:building_token > 0 }

					set_temp_variable = { months_temp = construction_amount@var:building_token }

						limit = {
							check_variable = { months_temp > 0 }

						array = PREV.under_construction_list_time
						index = ii
						value = vv
						break = bb

						if = {
							limit = {
								check_variable = { vv > months_temp }
							}
							add_to_array = { 
								array = PREV.under_construction_list_time 
								value = months_temp
								index = ii
							}
							add_to_array = { 
								array = PREV.under_construction_list_type 
								value = building_token
								index = ii
							}
							add_to_array = { 
								array = PREV.under_construction_list_building
								value = building_token
								index = ii
							}
							add_to_array = { 
								array = PREV.under_construction_list_state 
								value = THIS
								index = ii
							}
							set_temp_variable = { bb = 1 }
						}
					}
					if = {
						limit = {
							check_variable = { bb = 0 }
						}
						add_to_array = { PREV.under_construction_list_state = THIS }
						add_to_array = { PREV.under_construction_list_type = building_token }
						add_to_array = { PREV.under_construction_list_building = building_token }
						add_to_array = { PREV.under_construction_list_time = months_temp }
					}
				}
			}
