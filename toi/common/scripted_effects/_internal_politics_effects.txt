	# party_temp - <party_token> (Optional) Token for Party to be created.
	# name_override_temp - <party_token> (Optional) Token for an override party name. Party value is still party_temp
	# color_temp - <color_token> (Optional) Token for color used for party's symbol
	# type_temp - <party_range_token> - (Optional) Token for party type
	# leader_temp - <character_token> - (Optional) Token for party
	# marginalized_temp - <boolean> - (Optional) Boolean for if party is 'marginalized'
	# interest_groups_temp - list of <ig_token> - (Required)
	# ig_override_temp - <boolean> (Optional) - (1) will cause the party to 'steal' ig, if there are overlaps with current parties
	# treasury_temp - <float> (Optional)
add_party_to_country = {
	if = {
		limit = {
			check_variable = { party_temp = 0 }
		}
		get_party_name_from_interest_groups = yes
	}
	if = {
		limit = {
			check_variable = { name_override_temp > 0 }
		}
		set_variable = { name_override@var:party_temp = name_override_temp }
	}
	if = {
		limit = {
			check_variable = { color_temp = 0 }
		}
		get_party_color_from_interest_groups = yes
	}
	set_variable = { color@var:party_temp = color_temp }
	if = {
		limit = {
			check_variable = { type_temp = 0 }
		}
		get_party_type_from_interest_groups = yes
	}
	set_variable = { pol_range@var:party_temp = type_temp }
	var:leader_temp = {
		set_variable = { PREV.leader@var:party_temp = THIS }
	}
	set_variable = { marginalized@var:party_temp = marginalized_temp }
	set_variable = { treasury@var:party_temp = treasury_temp }

	if = {
		limit = {
			check_variable = { ig_override_temp > 0 }
		}
		for_each_loop = {
			array = interest_groups_temp
			value = ig_temp
			add_to_array = {
				array = interest_groups@var:party_temp
				value = ig_temp
			}
			if = {
				limit = {
					any_of = {
						array = parties_array
						value = p
						is_in_array = {
							array = interest_groups@var:p
							value = ig_temp
						}
					}
				}
				for_each_loop = {
					array = parties_array
					value = p
					remove_from_array = {
						array = interest_groups@var:p
						value = ig_temp
					}
				}
			}
		}
		else = {
			for_each_loop = {
				array = interest_groups_temp
				value = ig_temp
				if = {
					limit = {
						NOT = {
							any_of = {
								array = parties_array
								value = p
								is_in_array = {
									array = interest_groups@var:p
									value = ig_temp
								}
							}
						}
					}
					add_to_array = {
						array = interest_groups@var:party_temp
						value = ig_temp
					}
				}
			}
		}
	}
	# Log
		log = "Forming [?party_temp.GetTokenLocalizedKey] in [THIS.GetTag]."
		log = " [?party_temp.GetTokenLocalizedKey] is using [?color_temp.GetTokenLocalizedKey].."
		log = " [?party_temp.GetTokenLocalizedKey] is [?type_temp.GetTokenLocalizedKey]..."
		log = " [?party_temp.GetTokenLocalizedKey] has selected [?leader_temp.GetTokenLocalizedKey] as their leader...."
		for_each_loop = {
			array = interest_groups_temp
			value = ig_temp
			log = " Recruiting [?ig_temp_GetTokenLocalizedKey] Voters to [?party_temp.GetTokenLocalizedKey]"
		}
	#
	# Clear Values
		set_temp_variable = { party_temp = 0 }
		set_temp_variable = { name_override_temp = 0 }
		set_temp_variable = { color_temp = 0 }
		set_temp_variable = { type_temp = 0 }
		set_temp_variable = { leader_temp = 0 }
		set_temp_variable = { marginalized_temp = 0 }
		set_temp_variable = { treasury_temp = 0 }
		clear_temp_array = interest_groups_temp
		set_temp_variable = { ig_override_temp = 0 }
	#
}

	get_party_name_from_interest_groups = {
		# interest_groups_temp - list of <ig_token> - (Required)
		# type_temp - <party_range_token> - (Optional) Token for party type

		if = {
			limit = {

		}

				modifier = {
					check_variable = { interest_groups_temp^1 = token:ig_communists }
					add = 2
				}
				modifier = {
					check_variable = { interest_groups_temp^2 = token:ig_communists }
					add = 1
				}
				modifier = {
					check_variable = { type_temp = token:pr_ideological }
					factor = 2
				}
				modifier = {
					NOT = { has_tech = tech_humanities_communism }
					factor = 0
				}
				modifier = {
					is_in_array = {
						array = parties_array
						value = token:party_communist_party
					}
					factor = 0
				}
			}
		}
		log = "[?party_temp.GetTokenLocalizedKey] selected from interest_groups_temp"
	}
	get_party_type_from_interest_groups = {
		# interest_groups_temp - list of <ig_token> - (Required)

		set_temp_variable = { ig_institutional_count = 0 }
		set_temp_variable = { ig_ideological_count = 0 }
		set_temp_variable = { ig_establishment_count = 0 }
		set_temp_variable = { ig_oppositional_count = 0 }
		for_each_loop = {
			array = interest_groups_temp
			value = ig_temp
			if = {
				limit = {
					is_in_array = {
						array = global.ig_institutional_array
						value = ig_temp
					}
				}
				add_to_temp_variable = { ig_institutional_count = 1 }
			}
			if = {
				limit = {
					is_in_array = {
						array = global.ig_ideological_array
						value = ig_temp
					}
				}
				add_to_temp_variable = { ig_ideological_count = 1 }
			}
			if = {
				limit = {
					check_variable = { strength_perc@var:ig_temp > 0.075 }
				}
				add_to_temp_variable = { ig_establishment_count = strength@var:ig_temp }
			}
			else = {
				add_to_temp_variable = { ig_oppositional_count = strength@var:ig_temp }
			}
		}
		set_temp_variable = { total_temp = ig_institutional_count }
		add_to_temp_variable = { total_temp = ig_ideological_count }
		set_temp_variable = { divide_temp = ig_institutional_count }
		set_temp_variable = { divide_temp = total_temp }
		if = {
			limit = {
				check_variable = { divide_temp > 0.65 }
			}
			set_temp_variable = { type_temp = token:pr_institutional }
		}
		else_if = {
			limit = {
				check_variable = { divide_temp < 0.35 }
			}
			set_temp_variable = { type_temp = token:pr_institutional }
		}
		else_if = {
			limit = {
				check_variable = { interest_groups_temp^num > 4 }
			}
			set_temp_variable = { type_temp = token:pr_big_tent }
		}
		else_if = {
			limit = {
				check_variable = { ig_establishment_count > ig_oppositional_count }
			}
			set_temp_variable = { type_temp = token:pr_establishment }
		}
		else = {
			set_temp_variable = { type_temp = token:pr_oppositional}
		}
		
		log = "[?type_temp.GetTokenLocalizedKey] selected from interest_groups_temp"
	}

					array = political_overview_array
