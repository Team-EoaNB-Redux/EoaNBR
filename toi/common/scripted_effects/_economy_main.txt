# Main Economy File

# This includes the master script used to tick by every month, _advance_time, and the related weekly trade calculation, _weekly_trade
# The scripts that are used at the start of the game to setup the game are in _economy_setup.txt

# The _advance_time script consistes of two main portions:
#   Enact changes from the previous month
#   Update everything for the next month

_advance_time = {
    # Create econ order array 
    random_country = {
        if = {
            limit = {
                check_variable = { global.month > 11 }
            }
            set_variable = { global.month = 0 }
            add_to_variable = { global.years_passed = 1 }
        }

        create_econ_order_array = yes

        #Reset
            set_variable = { global.highest_state_gdp = 0 }
            set_variable = { global.highest_state_gdp_c = 0 }
        #
    }

    global_economy_update = yes
    for_each_scope_loop = {
        array = global.econ_order_array

        _country_advance_time = yes
    }
}
_country_advance_time = {

    ## ENACTING CHANGES ##

    # If you don't have eco modifiers, initialize your eco
    if = {
        limit = {
            NOT = {
                OR = {
                    has_dynamic_modifier = {
                        modifier = resource_trade
                    }
                    has_dynamic_modifier = {
                        modifier = budgetary_spending
                    }
                }
            }
        }
        country_eco_setup = yes
    }

    # Budget
    national_debt_update = yes
    set_temp_variable = { x_temp = net_budget }
    add_treasury = yes

    if = { limit = { is_ai = yes } ai_budget = yes }
    spending_effects_update = yes

    # Projects

    # 1.2 secs (calc: 1.4)
    update_national_production_methods = yes
    r_update_modifiers_country_effects = yes
    #buildings_update_base_latest_vals = yes
    every_controlled_state = {
        r_update_modifiers_state_effects = yes

        bank_reset_month = yes

        on_monthly_industries = yes # Money Accumulation & Factory Construction
        on_monthly_pops = yes # Pop Finances & Growth

        industry_productivity_sort = yes
        base_wage_calculation = yes
        on_monthly_industry_ai = yes # Adjusts employment_target

        # TEMPORARY - prospecting
        for_each_loop = {
            array = rgos
            value = r_temp

            set_temp_variable = { r_token = r_temp }
            if = {
                limit = {
                    check_variable = { prospecting@var:r_token = 1 }
                }
                set_temp_variable = { roll_temp = random }

                if = {
                    limit = { check_variable = { roll_temp < discovery_chance@var:r_token } }
                    rgo_discovery = yes
                }
                else_if = {
                    limit = {
                        subtract_from_temp_variable = { roll_temp = discovery_chance@var:r_token }
                        check_variable = { roll_temp < major_discovery_chance@var:r_token }
                    }
                    rgo_major_discovery = yes
                }
            }

            set_temp_variable = { r_token = r_temp }
            if = {
                limit = {
                    check_variable = { known_reserves@var:r_token > 0 }
                }
                set_temp_variable = { known_temp = known_reserves@var:r_token }
                set_temp_variable = { deposit_temp = known_temp }
                multiply_temp_variable = { known_temp = -1 }
                add_rgo_internal = yes
            }
        }

        pops_balance = yes
        fix_negative_workforce = yes

        ## UPDATING EVERYTHING ##
        update_base_resource_count = yes
        reload_industries_array = yes

        subsist_and_arable_update = yes
		employment_update = yes
        govt_building_update = yes
        industries_update_construction_retooling = yes
        industries_update = yes
        pops_update = yes
        
        set_variable = { prev_gdp_amount = gdp_amount }
    }

    # Investment
    investment_opportunities_update = yes
    investment_fund_update = yes
    investment_funds_monthly = yes
    investment_opportunities_monthly = yes

    ## UPDATING EVERYTHING ##
    update_country_pop_stats = yes

    # 0.2 secs
    market_leader_check = yes
    if = {
        limit = {
            check_variable = { market_leader = THIS }
        }
        market_update = yes
        market_calc_gdp = yes
    }

    # 0.1 secs
    budget_update = yes
    national_debt_update = yes

    # Statistics - 0.2 to 1.2 secs
    every_controlled_state = {
        refresh_construction_list = yes

        # nil secs
        add_to_array = {
            array = gdp_amount_array
            value = gdp_amount
        }
        if = {
            limit = {
                check_variable = { gdp_amount_array^num > 37 }
            }
            remove_from_array = { array = gdp_amount_array index = 0 }
        }

		# Average GDP over the last 3 years
		set_variable = { gdp_average = 0 }
		for_each_loop = {
			array = gdp_amount_array
			value = gdp_amount_temp
			add_to_variable = { gdp_average = gdp_amount_temp }
		}
		divide_variable = { gdp_average = gdp_amount_array^num }

        # nil secs
        set_temp_variable = { num_temp = 12 }
        if = {
            limit = {
                check_variable = { gdp_amount_array^num < num_temp }
            }
            set_temp_variable = { num_temp = gdp_amount_array^num }
        }
        subtract_from_temp_variable = { num_temp = 1 }

        # 0.1 secs
        set_variable = { gdp_growth = gdp_amount }
        if = {
            limit = {
                OR = {
                    check_variable = { gdp_amount > 1000 }
                    check_variable = { gdp_amount_array^num_temp > 1000 }
                }
            }
            set_temp_variable = { gdp_temp = gdp_amount_array^num_temp }
            divide_temp_variable = { gdp_temp = 100 }
            divide_variable = { gdp_growth = gdp_temp }
        }
        else = {
            set_variable = { gdp_growth = gdp_amount }
            multiply_variable = { gdp_growth = 100 }
            divide_variable = { gdp_growth = gdp_amount_array^num_temp }
        }
        subtract_from_variable = { gdp_growth = 100 }

        econ_rerender_gdp_graph = yes
        if = {
            limit = {
                check_variable = { global.month = 0 }
            }
            create_pie_chart_pop_type = yes # 0.2 secs
        }
    }

    #National Statistics - nil secs
        # GDP
        set_temp_variable = { gdp_amount_temp = gdp_amount }
        divide_temp_variable = { gdp_amount_temp = 1000 }

        add_to_array = { gdp_amount_monthly = gdp_amount_temp }
        if = {
            limit = {
                check_variable = { gdp_amount_monthly^num > 2 }
            }
            set_temp_variable = { gdp_amount_temp = gdp_amount_monthly^0 }
            add_to_temp_variable = { gdp_amount_temp = gdp_amount_monthly^1 }
            add_to_temp_variable = { gdp_amount_temp = gdp_amount_monthly^2 }
            add_to_array = {
                array = gdp_amount_quarterly
                value = gdp_amount_temp
            }
            if = {
                limit = {
                    check_variable = { gdp_amount_quarterly^num > 41 }
                }
                remove_from_array = { array = gdp_amount_quarterly index = 0 }
            }
            clear_array = gdp_amount_monthly
        }

        if = {
            limit = {
                check_variable = { global.month = 0 }
                check_variable = { global.years_passed > 0 }
            }
            set_temp_variable = { gdp_amount_temp = gdp_amount_quarterly^0 }
            add_to_temp_variable = { gdp_amount_temp = gdp_amount_quarterly^1 }
            add_to_temp_variable = { gdp_amount_temp = gdp_amount_quarterly^2 }
            add_to_temp_variable = { gdp_amount_temp = gdp_amount_quarterly^3 }
            add_to_array = {
                array = gdp_amount_yearly
                value = gdp_amount_temp
            }
        }

        # Budget
        add_to_array = {
            array = net_budget_array
            value = net_budget
        }
        if = {
            limit = {
                check_variable = { net_budget_array^num > 32 }
            }
            remove_from_array = { array = net_budget_array index = 0 }
        }

        #Create the Graphs
        econ_rerender_country_gdp_graph = yes
        econ_rerender_country_budget_graph = yes
    #

	if = {
		limit = {
			check_variable = { global.years_passed > 0 }
		}
		check_economic_cycle = yes
	}
}

r_weekly_trade = {
    # Effect that runs weekly 
    random_country = {
        create_econ_order_array = yes
    }

    for_each_scope_loop = {
        array = global.econ_order_array
        if = {
            limit = {
                check_variable = { market_leader = THIS }
            }
            market_update = yes
        }
    }

    global_market_update = yes

    for_each_scope_loop = {
        array = global.econ_order_array
        if = {
            limit = {
                OR = {
                    has_idea = export_focus
                    has_idea = free_trade
                }
            }
            for_each_loop = {
                array = global.resource_token_array
                value = resource_token
                
                clear_variable = trade_ai@var:resource_token
            }
        }

        if = {
            limit = {
                OR = {
                    has_idea = closed_economy
                    NOT = { check_variable = { market_leader = THIS } }
                }
            }
            clear_trade = yes
        }
        else_if = {
            limit = {
                check_variable = { market_leader = THIS }
            }
            trade_ai_calculation = yes
        }
    }
    
    for_each_scope_loop = {
        array = global.econ_order_array
        if = {
            limit = {
                check_variable = { market_leader = THIS }
            }
            trade_update = yes
        }
    }
}


on_monthly_industries = {
    for_each_loop = { # Building Construction & Retooling
        array = industries
        value = om_building

        if = {
            limit = {
                check_variable = { construction@var:om_building > 0 }
            }
            set_temp_variable = { const_speed = construction@var:om_building }
            set_temp_variable = { construction_speed_temp = 1 }
            add_to_temp_variable = { construction_speed_temp = industry_construction_speed_factor }
            multiply_temp_variable = { const_speed = construction_speed_temp }

            add_to_variable = { construction_progress@var:om_building = const_speed }
            subtract_from_variable = { construction_time_left@var:om_building = construction_speed_temp }
            if = {
                limit = {
                    check_variable = { construction_time_left@var:om_building < 0.001 }
                }
                add_to_variable = { size@var:om_building = construction@var:om_building }
                if = {
                    limit = {
                        check_variable = { om_building = token:arms_factory }
                    }
                    set_temp_variable = { const_amount = construction@var:om_building }
                    add_building_construction = {
                        type = arms_factory
                        level = var:const_amount
                        instant_build = yes
                    }
                }
                else_if = {
                    limit = {
                        check_variable = { om_building = token:industrial_complex }
                    }
                    set_temp_variable = { const_amount = construction@var:om_building }
                    add_building_construction = {
                        type = industrial_complex
                        level = var:const_amount
                        instant_build = yes
                    }
                }
                else_if = {
                    limit = {
                        check_variable = { om_building = token:dockyard }
                    }
                    set_temp_variable = { const_amount = construction@var:om_building }
                    add_building_construction = {
                        type = dockyard
                        level = var:const_amount
                        instant_build = yes
                    }
                }
                set_variable = { construction@var:om_building = 0 }
            }
        }
        if = {
            limit = {
                NOT = { check_variable = { retooling_pm@var:om_building = 0 } }
            }
            set_temp_variable = { const_speed = 1 }
            add_to_temp_variable = { const_speed = building_retooling_speed_factor }

            add_to_variable = { retooling_progress@var:om_building = const_speed }
            subtract_from_variable = { retooling_time_left@var:om_building = const_speed }
            if = {
                limit = {
                    check_variable = { retooling_time_left@var:om_building < 0.001 }
                }
                if = { limit = { is_in_array = { global.building_production_methods@var:om_building = retooling_pm@var:om_building }}
                    set_variable = { production_method@var:om_building = retooling_pm@var:om_building }
                }
                else = {
                    set_variable = { automation_method@var:om_building = retooling_pm@var:om_building }
                }
                set_variable = { retooling_pm@var:om_building = 0 }
            }
        }
    }
    for_each_loop = { # Money Accumulation
        array = industries
        value = om_building

        add_to_variable = { money@var:om_building = money_accumulation@var:om_building }
        if = { #Excess goes to owners
            limit = {
                check_variable = { money@var:om_building > money_limit@var:om_building }
            }
            set_variable = { money@var:om_building = money_limit@var:om_building }
        }
        if = {
            limit = {
                check_variable = { money@var:om_building < 0 }
            }
            set_variable = { money@var:om_building = 0 }
        }
    }
}
on_monthly_pops = {
    # Finances
	for_each_loop = {
		array = global.pops_array
		value = pop_token

		if = { limit = { check_variable = { pops_workforce@var:pop_token > 0 } }

			set_temp_variable = { current_pop_token = pop_token }

			# Calculates net balance after planned spending
			pops_financial_surplus_calc = yes

			# Automatic debt repayment
			pops_debt_monthly_repayment = yes

			# Deposits or Withdraws from Savings depending on Savings Target
			pops_handle_savings = yes

			if = { limit = { check_variable = { net_money_temp > 0 } } 
				# Use Surplus to Repay Debt
				pops_handle_surplus = yes
			}
			else_if = { limit = { check_variable = { net_money_temp < 0 } }
				# Attempts to Borrow the difference. 
				# Anything not covered will lead to a reduction in spending.
				pops_handle_deficit = yes
			}

			# Update SoL based on net position after financial adjustments
			pops_update_standard_of_living = yes
		}
		else = {
			set_variable = { pops_sol_progress@var:pop_token = 0 }
		}
	}
    
	# Update bank after doing pops
	bank_update = yes

	# Demographic Changes
    pops_population_growth = yes

	# Structural Pop Changes
    pops_peasant_job_migration = yes
}

pops_population_growth = {
	# Adds additional population from between months
	set_temp_variable = { pop_growth_temp = state_population_k }
	subtract_from_temp_variable = { pop_growth_temp = pops }
	multiply_temp_variable = { pop_growth_temp = pops_workforce_participation }

	add_to_variable = { pops_workforce = pop_growth_temp }
	for_each_loop = {
		array = global.pops_array
		value = pop_token
		set_temp_variable = { pops_amt_temp = pops_percentage@var:pop_token }
		multiply_temp_variable = { pops_amt_temp = pop_growth_temp }
		add_to_variable = { pops_workforce@var:pop_token = pops_amt_temp }
		add_to_variable = { pops_workforce_balance@var:pop_token = pops_amt_temp }
	}
}
pops_peasant_job_migration = {
	# If there's not enough jobs available, then laborers will become peasants
		# If there aren't enough laborers, nothing happens
	# If there's not enough workers for the jobs available, then peasants will become laborers

	set_temp_variable = { pops_change_temp = arable_land }
	divide_temp_variable = { pops_change_temp = 10 }
	set_temp_variable = { base_wage_temp = base_wage }
	divide_temp_variable = { base_wage_temp = 2 }

	set_temp_variable = { workforce_balance_temp = pops_workforce_balance@token:laborers }
	add_to_temp_variable = { workforce_balance_temp = pops_workforce_balance@token:workers }
	add_to_temp_variable = { workforce_balance_temp = pops_workforce_balance@token:professionals }
	add_to_temp_variable = { workforce_balance_temp = pops_workforce_balance@token:specialists }
	
	if = {
		limit = {
			check_variable = { workforce_balance_temp < 0 }
			check_variable = { pops_workforce@token:peasants > 0 }
		}
		clamp_temp_variable = { var = pops_change_temp min = 0 max = pops_workforce@token:peasants }
		subtract_from_variable = { pops_workforce@token:peasants = pops_change_temp }
		add_to_variable = { pops_workforce@token:laborers = pops_change_temp }
	}
	else_if = {
		limit = {
			check_variable = { workforce_balance_temp > 0 }
			check_variable = { base_wage_temp < productivity@token:subsistence_farm }
			OR = {
				check_variable = { peasants_marginal_land = 0 }
				check_variable = { pops_unemployed@token:laborers > 0 }
			}
		}
		clamp_temp_variable = { var = pops_change_temp min = 0 max = workforce_balance_temp }
		add_to_variable = { pops_workforce@token:peasants = pops_change_temp }
		subtract_from_variable = { pops_workforce@token:laborers = pops_change_temp }
	}
}

pops_financial_surplus_calc = {
	# current_pop_token
	# Outputs net_money_temp

	set_temp_variable = { net_money_temp = pops_earnings@var:current_pop_token }
	subtract_from_temp_variable = { net_money_temp = pops_spending@var:current_pop_token }
}
pops_debt_monthly_repayment = {
	# current_pop_token
	
	# The normal monthly repayment
	set_temp_variable = { x_temp = pops_debt_repayment_spending@var:current_pop_token }
	multiply_temp_variable = { x_temp = 0.5 }

	# Make repayment
	divide_temp_variable = { x_temp = 1000 }
	set_temp_variable = { target_type = 1 }
	set_temp_variable = { target_temp = pop_token }
	bank_repay_loan = yes
	
	# Amount is already accounted for in pops_spending, so no need for additional accounting
}
pops_handle_savings = {
	# current_pop_token
	# net_money_temp
	
	if = {
		limit = { check_variable = { pops_savings@var:current_pop_token > pops_savings_target@var:current_pop_token } }
		# Remove savings down to target
		set_temp_variable = { x_temp = pops_savings@var:current_pop_token }
		subtract_from_temp_variable = { x_temp = pops_savings_target@var:current_pop_token }
		
		set_temp_variable = { target_type = 1 }
		set_temp_variable = { target_temp = current_pop_token }
		set_temp_variable = { no_update_temp = 1 }
		bank_withdraw_money = yes

		add_to_temp_variable = { net_money_temp = withdraw_temp }
	}
	else = {
		# Add savings up to target
		set_temp_variable = { x_temp = pops_savings_target@var:current_pop_token }
		subtract_from_temp_variable = { x_temp = pops_savings@var:current_pop_token }
		clamp_temp_variable = { var = x_temp max = net_money_temp min = 0 }
		subtract_from_temp_variable = { net_money_temp = x_temp }
		
		set_temp_variable = { target_type = 1 }
		set_temp_variable = { target_temp = current_pop_token }
		set_temp_variable = { no_update_temp = 1 }
		bank_store_money = yes
	}
}
pops_handle_surplus = {
	# current_pop_token
	# net_money_temp

	# Modified value
	set_temp_variable = { x_temp = net_money_temp }
	multiply_temp_variable = { x_temp = 0.5 }
	
	# Make repayment
	divide_temp_variable = { x_temp = 1000 }
	set_temp_variable = { target_type = 1 }
	set_temp_variable = { target_temp = current_pop_token }
	bank_repay_loan = yes

	# Remove amount repaid to net_money_temp for bookkeeping
	set_temp_variable = { money_temp = repay_temp_k }
	multiply_temp_variable = { money_temp = 1000 }
	subtract_from_temp_variable = { net_money_temp = money_temp }
}
pops_handle_deficit = {
	# current_pop_token
	# net_money_temp

	# Borrows money depending on the pop's debt spending ratio and debt spending limit
	# Higher spending ratio compared to limit -> less borrowing 
	set_temp_variable = { x_temp = net_money_temp }
	divide_temp_variable = { x_temp = -1000 }

	set_temp_variable = { excess_ratio = pops_debt_spending_ratio@var:current_pop_token }
	divide_temp_variable = { excess_ratio = pops_debt_spending_limit@var:current_pop_token }
	if = { limit = { check_variable = { excess_ratio > 4 } }
		multiply_temp_variable = { x_temp = 0 }
	}
	else_if = { limit = { check_variable = { excess_ratio > 2 } }
		multiply_temp_variable = { x_temp = 0.25 }
	}
	else_if = { limit = { check_variable = { excess_ratio > 1 } }
		multiply_temp_variable = { x_temp = 0.5 }
	}

	# Borrow the money
	set_temp_variable = { target_type = 1 }
	set_temp_variable = { target_temp = current_pop_token }
	set_temp_variable = { no_update_temp = 1 }
	bank_loan_money = yes

	# Add the actual borrowed amount back for accounting
	set_temp_variable = { money_temp = borrow_temp_k }
	multiply_temp_variable = { money_temp = 1000 }
	add_to_temp_variable = { net_money_temp = money_temp }
}
pops_update_standard_of_living = {
	# current_pop_token
	# net_money_temp

	# Standard of Living Progress

	# Target Change = 3 * (Spending Target - Spending + Net Money) / Goods Spending
	set_temp_variable = { net_target_temp = pops_spending_target@var:current_pop_token }
	subtract_from_temp_variable = { net_target_temp = pops_spending@var:current_pop_token }
	add_to_temp_variable = { net_target_temp = net_money_temp }
	divide_temp_variable = { net_target_temp = pops_needs_spending@var:current_pop_token }
	multiply_temp_variable = { net_target_temp = 3 }

	set_temp_variable = { pops_sol_progress_temp = pops_sol_progress@var:current_pop_token }
	set_temp_variable = { pops_sol_temp = pops_sol@var:current_pop_token }

	if = {
		limit = {
			check_variable = { net_target_temp > 0.33 }
		}
		add_to_temp_variable = { pops_sol_progress_temp = net_target_temp }
	}
	else_if = {
		limit = {
			check_variable = { net_target_temp < -0.33 }
		}
		divide_temp_variable = { net_target_temp = 2 }
		add_to_temp_variable = { pops_sol_progress_temp = net_target_temp }
	}
	if = {
		limit = {
			check_variable = { pops_sol_temp = 1 }
			check_variable = { pops_sol_progress_temp < 0 }
		}
		set_temp_variable = { pops_sol_progress_temp = 0 }
	}
	clamp_temp_variable = { var = pops_sol_progress_temp min = -5 max = 5 }

	#Standard of Living Changes
	while_loop_effect = {
		limit = {
			check_variable = { pops_sol_progress_temp > 0.999 }
		}
		subtract_from_temp_variable = { pops_sol_progress_temp = 1 }
		add_to_temp_variable = { pops_sol_temp = 1 }
	}
	while_loop_effect = {
		limit = {
			check_variable = { pops_sol_progress_temp < -0.999 }
		}
		add_to_temp_variable = { pops_sol_progress_temp = 1 }
		subtract_from_temp_variable = { pops_sol_temp = 1 }
	}

	clamp_temp_variable = { var = pops_sol_temp min = 1 max = 200 }

	set_variable = { pops_sol_progress@var:current_pop_token = pops_sol_progress_temp }
	set_variable = { pops_sol@var:current_pop_token = pops_sol_temp }
}

check_economic_cycle = {
	set_temp_variable = { total_temp = 0 }
	set_temp_variable = { number_of_prospering_states = 0 } # >3% above average GDP over past 3 yrs
	set_temp_variable = { number_of_growing_states = 0 } # above average
	set_temp_variable = { number_of_declining_states = 0 } # below average
	set_temp_variable = { number_of_falling_states = 0 } # >3% below avg
	every_controlled_state = {
		set_temp_variable = { ratio_temp = gdp_amount }
		divide_temp_variable = { ratio_temp = gdp_average }

		set_temp_variable = { share_temp = gdp_amount }
		divide_temp_variable = { share_temp = 1000 }

		if = { limit = { check_variable = { ratio_temp > 1.03 } }
			add_to_temp_variable = { number_of_prospering_states = share_temp }
		}
		else_if = { limit = { check_variable = { ratio_temp < 0.97 } }
			add_to_temp_variable = { number_of_falling_states = share_temp }
		}
		else_if = { limit = { check_variable = { ratio_temp < 1 } }
			add_to_temp_variable = { number_of_declining_states = share_temp }
		}
		else = {
			add_to_temp_variable = { number_of_growing_states = share_temp }
		}

		add_to_temp_variable = { total_temp = share_temp }
	}

	divide_temp_variable = { number_of_prospering_states = total_temp }
	divide_temp_variable = { number_of_growing_states = total_temp }
	divide_temp_variable = { number_of_declining_states = total_temp }
	divide_temp_variable = { number_of_falling_states = total_temp }

	set_temp_variable = { growing_portion = number_of_growing_states }
	add_to_temp_variable = { growing_portion = number_of_prospering_states }
	
	set_temp_variable = { declining_portion = number_of_declining_states }
	add_to_temp_variable = { declining_portion = number_of_falling_states }

	# Eco statuses:
	# -2 - Recession
	# -1 - Downturn
	# 0 - Recovery
	# 1 - Downturn (from Boom)
	# 2 - Boom

	if = {
		limit = {
			NOT = { check_variable = { current_economic_status = -2 } }
			check_variable = { number_of_falling_states > 0.5 }
		}

		# log = "RECESSION"
		set_variable = { current_economic_status = -2 }
		if = {
			limit = {
				is_ai = no
			}
			country_event = { id = economy.140 } # Recession
		}
		else = {
			major_recession_effect = yes
		}
	}
	else_if = {
		limit = {
			check_variable = { current_economic_status > -1 }
			check_variable = { declining_portion > 0.5 }
		}

		# log = "DOWNTURN - SEVERE"
		set_variable = { current_economic_status = -1 }
		if = {
			limit = {
				is_ai = no
			}
			country_event = { id = economy.130 } # Downturn
		}
		else = {
			worsen_economic_situation = yes
		}
	}
	else_if = {
		limit = {
			check_variable = { current_economic_status < 0 }
			check_variable = { growing_portion > declining_portion }
		}

		# log = "RECOVERY"
		set_variable = { current_economic_status = 0 }
		if = {
			limit = {
				is_ai = no
			}
			country_event = { id = economy.110 } # Recovery
		}
		else = {
			improve_economic_situation = yes
		}
	}
	else_if = {
		limit = {
			check_variable = { current_economic_status = 2 }
			check_variable = { number_of_prospering_states < declining_portion }
		}

		# log = "DOWNTURN - MODERATE"
		set_variable = { current_economic_status = 1 }
		if = {
			limit = {
				is_ai = no
			}
			country_event = { id = economy.130 } # Downturn
		}
		else = {
			worsen_economic_situation = yes
		}
	}
	else_if = {
		limit = {
			NOT = { check_variable = { current_economic_status = -2 } }
			NOT = { check_variable = { current_economic_status = 2 } }
			check_variable = { number_of_prospering_states > 0.5 }
		}

		# log = "ECONOMIC BOOM"
		set_variable = { current_economic_status = 2 }
		if = {
			limit = {
				is_ai = no
			}
			country_event = { id = economy.120 } # Boom
		}
		else = {
			major_economic_boom_effect = yes
		}
	}

	# log = "Prospering/Growing | Declining/Falling: [?number_of_prospering_states]/[?number_of_growing_states] | [?number_of_declining_states]/[?number_of_falling_states]"
}
major_recession_effect = {
	if = {
		limit = {
			has_idea = depression
		}
		set_temp_variable = { situation_temp = token:free_fall }
	}
	else_if = {
		limit = {
			OR = {
				has_idea = recesion
				has_idea = stagnation
			}
		}
		set_temp_variable = { situation_temp = token:depression }
	}
	else = {
		set_temp_variable = { situation_temp = token:recession }
	}

	set_economic_situation = yes

	add_stability = -0.05
	decrease_ps_by_one = yes
}
major_economic_boom_effect = {
	if = {
		limit = {
			OR = {
				has_idea = steady_growth
				has_idea = fast_growth
				has_idea = economic_boom
			}
		}
		set_temp_variable = { situation_temp = token:economic_boom }
	}
	else_if = {
		limit = {
			has_idea = stagnation
		}
		set_temp_variable = { situation_temp = token:fast_growth }
	}
	else = {
		set_temp_variable = { situation_temp = token:steady_growth }
	}

	set_economic_situation = yes

	add_stability = 0.05
}
