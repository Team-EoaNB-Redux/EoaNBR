log_economic_cycle = {
	set_temp_variable = { total_temp = 0 }
	set_temp_variable = { number_of_prospering_states = 0 } # >3% above average GDP over past 3 yrs
	set_temp_variable = { number_of_growing_states = 0 } # above average
	set_temp_variable = { number_of_declining_states = 0 } # below average
	set_temp_variable = { number_of_falling_states = 0 } # >3% below avg
	every_controlled_state = {
		set_temp_variable = { ratio_temp = gdp_amount }
		divide_temp_variable = { ratio_temp = gdp_average }

		set_temp_variable = { share_temp = gdp_amount }
		divide_temp_variable = { share_temp = 1000 }

		if = { limit = { check_variable = { ratio_temp > 1.03 } }
			add_to_temp_variable = { number_of_prospering_states = share_temp }
		}
		else_if = { limit = { check_variable = { ratio_temp < 0.97 } }
			add_to_temp_variable = { number_of_falling_states = share_temp }
		}
		else_if = { limit = { check_variable = { ratio_temp < 1 } }
			add_to_temp_variable = { number_of_declining_states = share_temp }
		}
		else = {
			add_to_temp_variable = { number_of_growing_states = share_temp }
		}

		add_to_temp_variable = { total_temp = share_temp }
	}

	divide_temp_variable = { number_of_prospering_states = total_temp }
	divide_temp_variable = { number_of_growing_states = total_temp }
	divide_temp_variable = { number_of_declining_states = total_temp }
	divide_temp_variable = { number_of_falling_states = total_temp }
	
	log = "Prospering/Growing | Declining/Falling: [?number_of_prospering_states]/[?number_of_growing_states] | [?number_of_declining_states]/[?number_of_falling_states]"
}

calculate_state_deaths = {
	# pops

	# Pops die when:
	#	- State controller does not fully control state (contested)
	#	- State has damaged buildings
	#	- State has resistance

	# Applies weekly
	set_temp_variable = { deaths_base_temp = pops }
	multiply_temp_variable = { deaths_base_temp = 0.001 }

	set_temp_variable = { weekly_pop_deaths = 0 }

	log = "[THIS.GetName]"
	# Contested State
	if = {
		limit = {
			NOT = { is_fully_controlled_by = var:controller }
		}
		log = "contested"
		add_to_temp_variable = { weekly_pop_deaths = deaths_base_temp }
	}
	else = {
		log = "not contested"
	}

	# Damaged Buildings
	if = {
		limit = {
			has_damaged_buildings = yes
		}
		set_temp_variable = { total_levels = building_level@industrial_complex }
		add_to_temp_variable = { total_levels = building_level@arms_factory }
		add_to_temp_variable = { total_levels = building_level@dockyard }
		add_to_temp_variable = { total_levels = building_level@infrastructure }

		set_temp_variable = { damaged_levels = damaged_building_level@industrial_complex }
		add_to_temp_variable = { damaged_levels = damaged_building_level@arms_factory }
		add_to_temp_variable = { damaged_levels = damaged_building_level@dockyard }
		add_to_temp_variable = { damaged_levels = damaged_building_level@infrastructure }

		if = {
			limit = { check_variable = { total_levels > 0 } }
			set_temp_variable = { deaths_temp = damaged_levels }
			divide_temp_variable = { deaths_temp = total_levels }
			multiply_temp_variable = { deaths_temp = deaths_base_temp }
			add_to_temp_variable = { weekly_pop_deaths = deaths_base_temp }
		}
	}

	# Resistance
	if = {
		limit = {
			has_resistance = yes
		}
		set_temp_variable = { deaths_temp = resistance }
		divide_temp_variable = { deaths_temp = 100 }
		multiply_temp_variable = { deaths_temp = deaths_base_temp }
		add_to_temp_variable = { weekly_pop_deaths = deaths_base_temp }
	}

	log = "Weekly Deaths: [?weekly_pop_deaths]"
}
