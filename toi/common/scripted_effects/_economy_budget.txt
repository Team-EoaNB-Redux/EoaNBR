taxes_update = {
	#Country Scope

        set_temp_variable = { corruption_temp = modifier@corruption_factor }
        if = {
            limit = {
                check_variable = { corruption_temp > 1 }
            }
            set_temp_variable = { corruption_temp = 1 }
        }
        else_if = {
            limit = {
                check_variable = { corruption_temp < 0 }
            }
            set_temp_variable = { corruption_temp = 0 }
        }

        # Monthly cached free-class bases; the household formula is the same linear combination.
		set_variable = { income_tax_revenue = 0 }
		set_variable = { asset_tax_revenue = 0 }
		for_each_loop = { 
			array = global.pops_array value = tax_class index = tax_class_index
			if = { 
				limit = { NOT = { check_variable = { tax_class = token:slaves } } }
				set_variable = { income_tax_revenue@var:tax_class = { 
					value = tax_income_base@var:tax_class 
					multiply = income_tax_rate@var:tax_class 
					add = { value = tax_heads_base@var:tax_class multiply = income_tax_per_head } 
					add = { value = tax_low_base@var:tax_class multiply = income_tax_low_rate } 
					add = { value = tax_middle_base@var:tax_class multiply = income_tax_middle_rate } 
					add = { value = tax_high_base@var:tax_class multiply = income_tax_high_rate } } 
				}
				set_variable = { asset_tax_revenue@var:tax_class = { 
					value = tax_land_base@var:tax_class 
					multiply = asset_tax_land_rate 
					add = { value = tax_property_base@var:tax_class multiply = asset_tax_property_rate } 
					add = { value = tax_yield_base@var:tax_class multiply = asset_tax_yield_rate } 
					add = { value = tax_capital_base@var:tax_class multiply = asset_tax_capital_rate } 
				}}
				add_to_variable = { income_tax_revenue = income_tax_revenue@var:tax_class }
				add_to_variable = { asset_tax_revenue = asset_tax_revenue@var:tax_class }
			} 
		}
		set_variable = { sales_tax_revenue = { value = 0 } }
		for_each_loop = { array = global.resource_token_array value = tax_good index = tax_good_index
			if = { 
				limit = { NOT = { check_variable = { sales_tax_exempt@var:tax_good = 1 } } }
				add_to_variable = { sales_tax_revenue = { 
					value = household_goods@var:tax_good 
					multiply = price@var:tax_good 
					multiply = sales_tax_percentage_actual 
					divide = 100 
				}}
			}
		}
		sales_tax_goods_menu_update = yes
		set_variable = { income_tax_corruption = { value = income_tax_revenue multiply = corruption_temp } }
		set_variable = { sales_tax_corruption = { value = sales_tax_revenue multiply = corruption_temp } }
		set_variable = { asset_tax_corruption = { value = asset_tax_revenue multiply = corruption_temp } }
		subtract_from_variable = { income_tax_revenue = income_tax_corruption }
		subtract_from_variable = { sales_tax_revenue = sales_tax_corruption }
		subtract_from_variable = { asset_tax_revenue = asset_tax_corruption }
        #Corporate. The sectors' generic owners and the companies pay it on their positive operating profit (sectors_update, Q86)
            set_variable = { corporate_tax_revenue = 0 }
            for_each_loop = { array = sectors value = tax_sector index = tax_sector_index
                add_to_variable = { corporate_tax_revenue = { value = sector_generic_profit@var:tax_sector max = 0 multiply = corporate_tax_percentage_actual divide = 100 } }
            }
            for_each_loop = { array = companies value = tax_company index = tax_company_index
                add_to_variable = { corporate_tax_revenue = { value = global.company_profit@var:tax_company max = 0 multiply = corporate_tax_percentage_actual divide = 100 } }
            }

            set_variable = { corporate_tax_corruption = corporate_tax_revenue }
            multiply_variable = { corporate_tax_corruption = corruption_temp }
            subtract_from_variable = { corporate_tax_revenue = corporate_tax_corruption }
        #

        #Tariffs
            set_variable = { export_tariffs = 0 }
            set_variable = { import_tariffs = 0 }

            for_each_loop = {
                array = trade_routes
                value = r_token

                if = { #Import
                    limit = {
                        check_variable = { trade@var:r_token > 0 }
                    }
                    add_to_variable = { import_tariffs = trade_tariffs@var:r_token }
                }
                else_if = { #Export
                    limit = {
                        check_variable = { trade@var:r_token < 0 }
                    }
                    subtract_from_variable = { export_tariffs = trade_tariffs@var:r_token }
                }
            }

            if = {
                limit = {
                    check_variable = { export_tariffs < 0 }
                }
                divide_variable = { export_tariffs = modifier@bureaucratic_efficiency_factor }
            }
            if = {
                limit = {
                    check_variable = { import_tariffs < 0 }
                }
                divide_variable = { import_tariffs = modifier@bureaucratic_efficiency_factor }
            }

            set_variable = { net_tariffs = import_tariffs }
            add_to_variable = { net_tariffs = export_tariffs }

            set_variable = { tariff_revenue = 0 }
            if = {
                limit = {
                    check_variable = { import_tariffs > 0 }
                }
                add_to_variable = { tariff_revenue = import_tariffs }

    change_manufacturing_split = {
        #Scope: Country
        #Moves points of the contracted factories' split (M08, Q107) to or from one factory type. The military share balances
        #civilian and naval changes, and a military change is balanced by the civilian share, or by the naval share once the
        #civilian share has run out. It changes the targets: HOI4's factories follow month by month (Q115)
        #INPUT:
            #f_token - the factory type: token:industrial_complex, token:arms_factory or token:dockyard
            #percent_change_temp - Points to add to it, 1.0 = 1%

        set_temp_variable = { balance_token_temp = token:arms_factory }
        if = {
            limit = { check_variable = { f_token = token:arms_factory } }
            set_temp_variable = { balance_token_temp = token:industrial_complex }
            if = {
                limit = {
                    check_variable = { percent_change_temp > 0 }
                    check_variable = { manufacturing_split@token:industrial_complex < percent_change_temp }
                }
                set_temp_variable = { balance_token_temp = token:dockyard }
            }
        }
        # No share goes below 0
        if = {
            limit = { check_variable = { manufacturing_split@var:balance_token_temp < percent_change_temp } }
            set_temp_variable = { percent_change_temp = manufacturing_split@var:balance_token_temp }
        }
        set_temp_variable = { lowest_change_temp = manufacturing_split@var:f_token }
        multiply_temp_variable = { lowest_change_temp = -1 }
        if = {
            limit = { check_variable = { percent_change_temp < lowest_change_temp } }
            set_temp_variable = { percent_change_temp = lowest_change_temp }
        }

        add_to_variable = { manufacturing_split@var:f_token = percent_change_temp }
        subtract_from_variable = { manufacturing_split@var:balance_token_temp = percent_change_temp }
        manufacturing_base_update = yes
    }

    set_temp_variable = { subsidy_size = 0 }

    for_each_loop = { array = sectors value = grant_sector index = grant_index
        if = { limit = { check_variable = { sector_subsidized@var:grant_sector = 1 } } add_to_temp_variable = { subsidy_size = sector_size@var:grant_sector } }

    }
    set_temp_variable = { industrial_grant_budget = industrial_programme_spending }
    set_temp_variable = { transport_grant_budget = { value = infrastructure_programme_spending multiply = 0.8 } }
    set_temp_variable = { infrastructure_baseline_cost = { value = infrastructure_programme_spending subtract = transport_grant_budget } }
    set_variable = { industrial_programme_spending = 0 }
    set_variable = { infrastructure_programme_spending = infrastructure_baseline_cost }
    for_each_loop = { array = sectors value = grant_sector index = grant_index

        }
        if = { limit = { check_variable = { transport_size > 0 } OR = { check_variable = { grant_sector = token:railway } check_variable = { grant_sector = token:port } } }
            set_variable = { sector_transport_grant_per_size@var:grant_sector = { value = transport_grant_budget divide = transport_size } }
        }
        set_variable = { sector_government_grant@var:grant_sector = { value = sector_grant_per_size@var:grant_sector multiply = sector_generic_size@var:grant_sector } }
        add_to_variable = { industrial_programme_spending = sector_government_grant@var:grant_sector }

        add_to_variable = { infrastructure_programme_spending = transport_grant }
        add_to_variable = { sector_government_grant@var:grant_sector = transport_grant }

            }

government_budget_commit = {
    # Monthly closing quote. UI changes forecasts; they cannot change cash already promised to recipients.
    # Tax receipts use the actual deductions in the households' next monthly financial settlement.
    set_variable = { government_primary_balance_due = { value = net_budget add = national_debt_interest subtract = income_tax_revenue subtract = asset_tax_revenue subtract = sales_tax_revenue subtract = corporate_tax_revenue add = { value = sectors_profit_tax add = companies_profit_tax multiply = { value = 1 subtract = modifier@corruption_factor clamp = { min = 0 max = 1 } } } add = { value = household_income_tax_due add = household_asset_tax_due add = household_sales_tax_due multiply = { value = 1 subtract = modifier@corruption_factor clamp = { min = 0 max = 1 } } } add = welfare_programme_spending subtract = household_welfare_due add = patronage_programme_spending subtract = household_patronage_due } }
    set_variable = { development_fund_due = development_fund_spending }
    for_each_loop = { array = sectors value = grant_sector index = grant_index
        set_variable = { sector_government_grant_due@var:grant_sector = sector_government_grant@var:grant_sector }
    }
    for_each_loop = { array = companies value = grant_company index = grant_company_index
        set_variable = { company_government_grant_due@var:grant_company = company_government_grant@var:grant_company }
    }
}

government_construction_update = {
    # Refresh once monthly, outside UI quotations. The state construction records already include financing.
        set_variable = { construction_spending = 0 }
        set_variable = { projects_spending = 0 }
        set_variable = { industrial_subsidy = 0 }
        set_variable = { infrastructure_subsidy = 0 }

        set_variable = { administration_spending = 0 }
        set_variable = { military_payroll_spending = 0 }
        every_controlled_state = {
            for_each_loop = {
                array = industries
                index = sp_i
                value = sp_token

                #Construction
                if = {
                    limit = {
                        check_variable = { construction@var:sp_token > 0 }
                        check_variable = { construction_financier@var:sp_token = PREV }
                    }
                    add_to_variable = { PREV.construction_spending = construction_cost@var:sp_token }
                }
                if = {
                    limit = {
                        NOT = { check_variable = { retooling_pm@var:sp_token = 0 } }
                        check_variable = { retooling_financier@var:sp_token = PREV }
                    }
                    add_to_variable = { PREV.construction_spending = retooling_cost@var:sp_token }
                }

                # Prospecting
                if = {
                    limit = {
                        check_variable = { prospecting@var:sp_token = 1 }
                        check_variable = { prospecting_financier@var:sp_token = ROOT }
                    }
                    add_to_variable = { PREV.construction_spending = prospecting_cost@var:sp_token }
                }

            }
        }

    set_variable = { government_construction_cost = construction_spending }
}
