#Market Updates
	join_market = {
		if = {
			limit = { NOT = { check_variable = { market_leader = x_temp } } }
			set_variable = { market_leader = x_temp }
			market_update = yes
			if = {
				limit = {
					is_in_array = {
						array = global.markets
						value = THIS
					}
				}
				remove_from_array = {
					array = global.markets
					value = THIS
				}
			}
		}
		custom_effect_tooltip = join_market_tooltip
	}
	market_update = { # The master market update script. Adds up all values from every state up to the market leader 
		# Scope: Market Leader
		# Make sure current scope is a market leader by doing check_variable = { market_leader = THIS }
		
		# The following two inputs allow market_update to run without running more strenuous operations. For performance
		# ignore_state_market_temp - Uses each member's goods totals from its last monthly tick (national_market_states_update)
		#   instead of adding them up again. The monthly and weekly ticks set it: industries and pops only change once a month
		# ignore_national_market_temp - Does not recalc state OR national markets if non-zero

		set_temp_variable = { market_leader_temp = THIS }

        hidden_effect = {

            # Runs the market_setup command again to ensure that all countries are in your market that SHOULD be
            market_setup = yes

            for_each_loop = { # Reset to 0
                array = global.resource_token_array
                value = r_type
                
                set_variable = { need@var:r_type = 0 }
                set_variable = { production@var:r_type = 0 }
            }

            for_each_scope_loop = { # Add up each country
                array = market_countries

                if = {
                    limit = {
                        check_variable = { ignore_national_market_temp = 0 }
                    }
                    national_market_update = yes
                }

                
                for_each_loop = {
                    array = global.resource_token_array
                    value = r_type
                    add_to_variable = { market_leader_temp:need@var:r_type = local_need@var:r_type }
                    add_to_variable = { market_leader_temp:production@var:r_type = local_production@var:r_type }
                }
            }

            market_calculations = yes

            set_temp_variable = { ignore_state_market_temp = 0 }
            set_temp_variable = { ignore_national_market_temp = 0 }
        }
	}
	market_calculations = { # Does supply/demand calculations for the market and adjusts prices. Included in market_update & trade_update_quantity. 
		# Scope: Market Leader
		# Does all the market calculations after having gotten all the stuff from states. Mostly for trade

		set_temp_variable = { market_leader_temp = THIS }

		for_each_loop = { # Market Calculations
			array = global.resource_token_array
			value = r_type
			
			set_variable = { supply@var:r_type = production@var:r_type }
			set_variable = { demand@var:r_type = need@var:r_type }

			if = { # Adding positive Trade to supply
				limit = {
					check_variable = { trade@var:r_type > 0 }
				}
				add_to_variable = { supply@var:r_type = trade@var:r_type }
			}
			if = { # Adding negative Trade to demand
				limit = {
					check_variable = { trade@var:r_type < 0 }
				}
				subtract_from_variable = { demand@var:r_type = trade@var:r_type }
			}

			# if = { limit = { check_variable = { r_type = token:wood } } add_to_variable = { supply@token:wood = 3000 } }
			# if = { limit = { check_variable = { r_type = token:fiber } } add_to_variable = { supply@token:fiber = 400 } }
			# if = { limit = { check_variable = { r_type = token:luxuries } } add_to_variable = { supply@token:luxuries = 600 } }
			# if = { limit = { check_variable = { r_type = token:textiles } } add_to_variable = { demand@token:textiles = 1200 } }
			# if = { limit = { check_variable = { r_type = token:consumer_goods } } add_to_variable = { demand@token:consumer_goods = 600 } }
			# if = { limit = { check_variable = { r_type = token:machinery } } add_to_variable = { demand@token:machinery = 100 } }
			# if = { limit = { check_variable = { r_type = token:steel } } add_to_variable = { demand@token:steel = 300 } }
			# if = { limit = { check_variable = { r_type = token:iron } } add_to_variable = { demand@token:iron = 500 } }
			# if = { limit = { check_variable = { r_type = token:coal } } add_to_variable = { demand@token:coal = 500 } }
			# if = { limit = { check_variable = { r_type = token:machinery } } add_to_variable = { demand@token:machinery = 100 } }

			# National Balance
			set_variable = { balance@var:r_type = supply@var:r_type }
			subtract_from_variable = { balance@var:r_type = demand@var:r_type }

			# Target price. The actual price (price@) only moves toward it once a month, in market_prices_monthly
			set_temp_variable = { s_temp = supply@var:r_type }
			set_temp_variable = { d_temp = demand@var:r_type }
			get_price = yes
			set_variable = { target_price@var:r_type = p_return_temp }
			if = { # While the economy is being set up (prices_snap_temp), and for a good with no price yet, the price starts at its target
				limit = {
					OR = {
						check_variable = { prices_snap_temp = 1 }
						NOT = { check_variable = { price@var:r_type > 0 } }
					}
				}
				set_variable = { price@var:r_type = p_return_temp }
			}
		}
		if = {
			limit = { check_variable = { prices_snap_temp = 1 } }
			market_prices_update = yes
		}

		for_each_scope_loop = { # Apply to each country
			array = market_countries

			for_each_loop = {
				array = global.resource_token_array
				value = r_type
				set_variable = { supply@var:r_type = market_leader_temp:supply@var:r_type }
				set_variable = { demand@var:r_type = market_leader_temp:demand@var:r_type }
				set_variable = { balance@var:r_type = market_leader_temp:balance@var:r_type }
				# The price only moves monthly (market_prices_update copies it then). Copying it here too gives a country that just joined the market its prices within the week
				set_variable = { price@var:r_type = market_leader_temp:price@var:r_type }
				set_variable = { price_compared_to_base@var:r_type = market_leader_temp:price_compared_to_base@var:r_type }
				set_variable = { price_shipping_capacity = market_leader_temp:price_shipping_capacity }
			}
		}

        for_each_scope_loop = { # Each country gets their own production + trade (multiplied by portion of GDP in market - only if it's imports)
			array = market_countries
			for_each_loop = {
				array = global.resource_token_array
				value = r_type
				set_variable = { resource_cost@var:r_type = local_production@var:r_type }

				if = { # Trade
					limit = {
						check_variable = { market_leader:trade@var:r_type > 0 }
					}
					set_temp_variable = { trade_amount_temp = gdp_amount }
					divide_temp_variable = { trade_amount_temp = market_leader:market_gdp }
					multiply_temp_variable = { trade_amount_temp = market_leader:trade@var:r_type }
					add_to_variable = { resource_cost@var:r_type = trade_amount_temp }
				}

				multiply_variable = { resource_cost@var:r_type = -1 }
				round_variable = resource_cost@var:r_type
			}
        }
		for_each_loop = { # Market leader gets a duplicate of all resources in the market
			array = global.resource_token_array
			value = r_type
			# Trade
			set_variable = { resource_cost@var:r_type = local_production@var:r_type }
			add_to_variable = { resource_cost@var:r_type = production@var:r_type }
			add_to_variable = { resource_cost@var:r_type = trade@var:r_type }
			multiply_variable = { resource_cost@var:r_type = -1 }
			round_variable = resource_cost@var:r_type
		}
		
		
	}
	market_statistics_update = {
		# Scope: Market Leader. The market screen's biggest producing and consuming countries for each good. States keep no
		# per-good totals: each country lists its sectors' biggest states instead (sector_states@, sectors_update).
		# Monthly and at startup: production and demand only change once a month
		set_temp_variable = { market_leader_temp = THIS }

		for_each_loop = { # Market Statistics
			array = global.resource_token_array
			value = r_type

			clear_array = market_production_countries@var:r_type
			clear_array = market_consumption_countries@var:r_type
			
			for_each_scope_loop = {
				array = market_countries
				
				# Add production from this country to market production countries array, sorted
				for_each_loop = {
					array = market_leader_temp:market_production_countries@var:r_type
					index = ii value = vv break = bb

					if = { limit = { check_variable = { vv:local_production@var:r_type < local_production@var:r_type } }
						add_to_array = { array = market_leader_temp:market_production_countries@var:r_type value = THIS index = ii }
						set_temp_variable = { bb = 1 }
					}
				}
				if = { limit = { check_variable = { bb = 0 } } add_to_array = { market_leader_temp:market_production_countries@var:r_type = THIS } }
				
				# Add consumption from this country to market consumption countries array, sorted
				for_each_loop = {
					array = market_leader_temp:market_consumption_countries@var:r_type
					index = ii value = vv break = bb

					if = { limit = { check_variable = { vv:local_need@var:r_type < local_need@var:r_type } }
						add_to_array = { array = market_leader_temp:market_consumption_countries@var:r_type value = THIS index = ii }
						set_temp_variable = { bb = 1 }
					}
				}
				if = { limit = { check_variable = { bb = 0 } } add_to_array = { market_leader_temp:market_consumption_countries@var:r_type = THIS } }
			}
		}
	}
	market_prices_monthly = {
		# Scope: Market Leader
		# Runs once a month, at the start of the tick, so revenue, wages, pop spending and GDP all use the same prices.
		# Moves the market price (price@) and the world price (global_market_price@) part of the way toward the targets
		# the weekly trade tick last set (target_price@, global_market_target_price@). Trade itself works off the targets.

		for_each_loop = {
			array = global.resource_token_array
			value = r_type

			set_temp_variable = { p_return_temp = price@var:r_type }
			set_temp_variable = { target_temp = target_price@var:r_type }
			price_step = yes
			set_variable = { price@var:r_type = p_return_temp }

			set_temp_variable = { p_return_temp = global_market_price@var:r_type }
			set_temp_variable = { target_temp = global_market_target_price@var:r_type }
			price_step = yes
			set_variable = { global_market_price@var:r_type = p_return_temp }
		}

		market_prices_update = yes
	}
	market_prices_update = {
		# Scope: Market Leader
		# Copies the actual prices to every country in the market and to their states, after they move (and at startup)

		set_temp_variable = { market_leader_temp = THIS }

		for_each_loop = {
			array = global.resource_token_array
			value = r_type

			set_variable = { price_compared_to_base@var:r_type = { value = price@var:r_type divide = global.base_price@var:r_type subtract = 1 } }
			set_variable = { global_market_price_compared_to_base@var:r_type = { value = global_market_price@var:r_type divide = global.base_price@var:r_type subtract = 1 } }
		}

		for_each_scope_loop = {
			array = market_countries

			for_each_loop = {
				array = global.resource_token_array
				value = r_type
				set_variable = { price@var:r_type = market_leader_temp:price@var:r_type }
				set_variable = { price_compared_to_base@var:r_type = market_leader_temp:price_compared_to_base@var:r_type }
			}
			every_controlled_state = {
				for_each_loop = {
					array = global.resource_token_array
					value = r_type
					set_variable = { price@var:r_type = PREV.price@var:r_type }
					set_variable = { price_compared_to_base@var:r_type = PREV.price_compared_to_base@var:r_type }
				}
			}
		}

		market_cache_update = yes
	}
	market_cache_update = {
		# Scope: Market Leader. Runs from market_prices_update, once the members have the new prices
		# The market layer of the PM/AM cache (see cache_register_industry): each sector's output value and each method's input and
		# construction cost, at the market's prices. Worked out once per market for every sector and method a member uses, and
		# copied to each member that uses it. The leader keeps the market's copy, stamped with the price update it belongs to.
		add_to_variable = { market_cache_stamp = 1 }
		set_temp_variable = { market_leader_temp = THIS }

		for_each_scope_loop = {
			array = market_countries

			for_each_loop = {
				array = sectors
				value = cache_token
				if = {
					limit = { check_variable = { market_leader_temp:market_output_stamp@var:cache_token = market_leader_temp:market_cache_stamp } }
					set_variable = { market_output_value@var:cache_token = market_leader_temp:market_output_value@var:cache_token }
				}
				else = { # Each member has the market's prices, so the first member to need it works it out for the market
					cache_sector_calc = yes
					set_variable = { market_leader_temp:market_output_value@var:cache_token = market_output_value@var:cache_token }
					set_variable = { market_leader_temp:market_output_stamp@var:cache_token = market_leader_temp:market_cache_stamp }
				}
			}
			for_each_loop = {
				array = methods_in_use
				value = cache_token
				if = {
					limit = { check_variable = { market_leader_temp:method_cost_stamp@var:cache_token = market_leader_temp:market_cache_stamp } }
					set_variable = { method_input_cost@var:cache_token = market_leader_temp:method_input_cost@var:cache_token }
					set_variable = { method_construction_cost@var:cache_token = market_leader_temp:method_construction_cost@var:cache_token }
				}
				else = {
					cache_method_calc = yes
					set_variable = { market_leader_temp:method_input_cost@var:cache_token = method_input_cost@var:cache_token }
					set_variable = { market_leader_temp:method_construction_cost@var:cache_token = method_construction_cost@var:cache_token }
					set_variable = { market_leader_temp:method_cost_stamp@var:cache_token = market_leader_temp:market_cache_stamp }
				}
			}
		}
	}
	national_market_update = {
		# The country's goods demand and supply for its market: its industries' and pops' totals (states_need@,
		# states_production@), plus military demand, which changes daily. ignore_state_market_temp keeps the totals from the
		# country's last monthly tick. Otherwise they're added up again
		if = {
			limit = { check_variable = { ignore_state_market_temp = 0 } }
			national_market_states_update = yes
		}

		calculate_military_demand = yes

		for_each_loop = {
			array = global.resource_token_array
			value = r_type

			set_variable = { local_need@var:r_type = { value = military_demand@var:r_type add = states_need@var:r_type } }
			set_variable = { local_production@var:r_type = states_production@var:r_type }
		}

		set_temp_variable = { modifier_temp = modifier@military_production_cost_factor }
		add_to_temp_variable = { modifier_temp = 1 }

		for_each_loop = { # Military Resources
			array = global.resource_token_array
			value = r_type

			set_temp_variable = { amount_temp = resource_consumed@var:r_type }
			if = {
				limit = {
					check_variable = { resource@var:r_type < 0 }
				}
				subtract_from_temp_variable = { amount_temp = resource@var:r_type }
			}
			multiply_temp_variable = { amount_temp = modifier_temp }
			add_to_variable = { local_need@var:r_type = amount_temp }
			#set_variable = { balance@var:r_type = resource@var:r_type }
		}
	}
	national_market_states_update = {
		# Country scope. The goods its states make and use, for its market: states_production@<good> and states_need@<good>.
		# No state keeps per-good totals. Industries are added up by sector, and those are turned into goods once per country:
		#   output@<sector> - the bundles of base production its state-industries made
		#   sector_pm_scale@<sector>, sector_am_scale@<sector> - the input_scale@ of its state-industries on each PM and AM, in
		#     arrays parallel to the building's method lists (global.building_production_methods@, building_automation_methods@)
		#   sector_pm_size@<sector>, sector_am_size@<sector> - the method mix: the sector's size on each PM and AM, in the same
		#     arrays. Not needed for the goods, but this pass finds each method's place in the lists anyway
		# Subsistence farms (not a sector) add their goods, and each state its pops' demand (pops_need@<good>, get_pop_needs),
		# construction and retooling. Runs once a month after the country's states update, and whenever market_update asks
		# for fresh totals
		for_each_loop = {
			array = global.resource_token_array
			value = r_type
			set_variable = { states_need@var:r_type = 0 }
			set_variable = { states_production@var:r_type = 0 }
		}
		for_each_loop = {
			array = sectors
			value = g_token
			set_variable = { output@var:g_token = 0 }
			set_temp_variable = { g_n_temp = global.building_production_methods@var:g_token^num }
			clear_array = sector_pm_size@var:g_token
			clear_array = sector_pm_scale@var:g_token
			resize_array = { array = sector_pm_size@var:g_token value = 0 size = g_n_temp }
			resize_array = { array = sector_pm_scale@var:g_token value = 0 size = g_n_temp }
			set_temp_variable = { g_n_temp = global.building_automation_methods@var:g_token^num }
			clear_array = sector_am_size@var:g_token
			clear_array = sector_am_scale@var:g_token
			resize_array = { array = sector_am_size@var:g_token value = 0 size = g_n_temp }
			resize_array = { array = sector_am_scale@var:g_token value = 0 size = g_n_temp }
		}

		every_controlled_state = {
			for_each_loop = {
				array = industries
				value = g_token
				if = {
					limit = {
						check_variable = { size@var:g_token > 0 } # Railways and ports at size 0 aren't updated, and keep old figures
						NOT = { check_variable = { g_token = token:subsistence_farm } }
						NOT = { check_variable = { global.building_government_control@var:g_token = 1 } }
					}
					if = {
						limit = { check_variable = { output@var:g_token > 0 } }
						add_to_variable = { PREV.output@var:g_token = output@var:g_token }
					}
					set_temp_variable = { g_size_temp = size@var:g_token }
					set_temp_variable = { g_scale_temp = input_scale@var:g_token }
					set_temp_variable = { g_pm_temp = production_method@var:g_token }
					set_temp_variable = { g_am_temp = automation_method@var:g_token }
					PREV = {
						if = {
							limit = { check_variable = { sector_pm_size@var:g_token^num > 0 } } # Unregistered: a state taken this month
							set_temp_variable = { g_break_temp = 0 }
							for_each_loop = {
								array = global.building_production_methods@var:g_token
								index = g_m value = g_method_temp break = g_break_temp
								if = { limit = { check_variable = { g_method_temp = g_pm_temp } }
									add_to_variable = { sector_pm_size@var:g_token^g_m = g_size_temp }
									add_to_variable = { sector_pm_scale@var:g_token^g_m = g_scale_temp }
									set_temp_variable = { g_break_temp = 1 }
								}
							}
						}
						if = {
							limit = {
								NOT = { check_variable = { g_am_temp = 0 } }
								check_variable = { sector_am_size@var:g_token^num > 0 }
							}
							set_temp_variable = { g_break_temp = 0 }
							for_each_loop = {
								array = global.building_automation_methods@var:g_token
								index = g_m value = g_method_temp break = g_break_temp
								if = { limit = { check_variable = { g_method_temp = g_am_temp } }
									add_to_variable = { sector_am_size@var:g_token^g_m = g_size_temp }
									add_to_variable = { sector_am_scale@var:g_token^g_m = g_scale_temp }
									set_temp_variable = { g_break_temp = 1 }
								}
							}
						}
					}
				}
			}

			# Subsistence farms aren't a sector (Q85), so their goods are added here
			for_each_loop = {
				array = global.building_base_production@token:subsistence_farm
				value = r_type
				index = g_i
				set_temp_variable = { g_amount_temp = global.building_base_production_amount@token:subsistence_farm^g_i }
				multiply_temp_variable = { g_amount_temp = output@token:subsistence_farm }
				add_to_variable = { PREV.states_production@var:r_type = g_amount_temp }
			}

			for_each_loop = {
				array = global.pop_needs_resources
				value = r_type
				add_to_variable = { PREV.states_need@var:r_type = pops_need@var:r_type }
			}

			for_each_loop = { # Construction
				array = industries_construction_array
				value = g_token
				for_each_loop = {
					array = construction_resources@var:g_token
					value = r_type
					index = g_i
					add_to_variable = { PREV.states_need@var:r_type = construction_amount@var:g_token^g_i }
				}
			}
			for_each_loop = { # Retooling
				array = industries_retooling_array
				value = g_token
				for_each_loop = {
					array = retooling_resources@var:g_token
					value = r_type
					index = g_i
					add_to_variable = { PREV.states_need@var:r_type = retooling_amount@var:g_token^g_i }
				}
			}
		}

		for_each_loop = { # What each sector made and used
			array = sectors
			value = g_token
			if = {
				limit = { check_variable = { output@var:g_token > 0 } }
				for_each_loop = {
					array = global.building_base_production@var:g_token
					value = r_type
					index = g_i
					set_temp_variable = { g_amount_temp = global.building_base_production_amount@var:g_token^g_i }
					multiply_temp_variable = { g_amount_temp = output@var:g_token }
					add_to_variable = { states_production@var:r_type = g_amount_temp }
				}
			}
			for_each_loop = {
				array = sector_pm_scale@var:g_token
				value = g_scale_temp
				index = g_m
				if = {
					limit = { check_variable = { g_scale_temp > 0 } }
					set_temp_variable = { g_method_temp = global.building_production_methods@var:g_token^g_m }
					for_each_loop = {
						array = global.pm_input@var:g_method_temp
						value = r_type
						index = g_i
						set_temp_variable = { g_amount_temp = global.pm_input_amount@var:g_method_temp^g_i }
						multiply_temp_variable = { g_amount_temp = g_scale_temp }
						add_to_variable = { states_need@var:r_type = g_amount_temp }
					}
				}
			}
			for_each_loop = {
				array = sector_am_scale@var:g_token
				value = g_scale_temp
				index = g_m
				if = {
					limit = { check_variable = { g_scale_temp > 0 } }
					set_temp_variable = { g_method_temp = global.building_automation_methods@var:g_token^g_m }

	get_price = {
		#Given the resource & supply/demand, returns price
		#Input: r_type (Resource Token), s_temp (supply), d_temp (demand)
		#Return: p_return_temp
		#Any Scope

		set_temp_variable = { p_return_temp = global.base_price@var:r_type }
		if = {
			limit = {
				NOT = { check_variable = { r_type = token:valuables } }
				is_in_array = { global.resource_token_array = r_type }
				#NOT = { check_variable = { r_type = token:resource_finance } }
			}
			if = {
				limit = {
					check_variable = { s_temp > 0 }
					NOT = { check_variable = { d_temp > 0 } }
				}
				multiply_temp_variable = { p_return_temp = 0.25 }
			}
			else_if = {
				limit = {
					NOT = { check_variable = { s_temp > 0 } }
					check_variable = { d_temp > 0 }
				}
				multiply_temp_variable = { p_return_temp = 4.0 }
			}
			else = {
				set_temp_variable = { grp_multiplier = d_temp }
				divide_temp_variable = { grp_multiplier = s_temp }
				if = {
					limit = {
						check_variable = { grp_multiplier > 2 }
					}
					multiply_temp_variable = { p_return_temp = 4.0 }
				}
				else_if = {
					limit = {
						check_variable = { grp_multiplier < 0.5 }
					}
					multiply_temp_variable = { p_return_temp = 0.25 }
				}
				else = {
					multiply_temp_variable = { grp_multiplier = d_temp }
					divide_temp_variable = { grp_multiplier = s_temp }

					multiply_temp_variable = { p_return_temp = grp_multiplier } # P = B * D^2 / S^2
				}
			}
		}
		else_if = {
			limit = {
				check_variable = { r_type = token:valuables }
			}
			multiply_temp_variable = { p_return_temp = 2 }
		}
	}
	price_step = {
		# Moves a price part of the way toward its target. A wider gap closes a bigger share of itself each month,
		# and a gap inside the dead band doesn't move the price at all.
		# The gap is max(target/price, price/target) - 1, so a doubling and a halving count the same.
		#Input: p_return_temp (actual price), target_temp (target price)
		#Return: p_return_temp
		#Any Scope

		if = {
			limit = { # Both are guarded: a math expression doesn't return 0 when dividing by 0
				check_variable = { target_temp > 0 }
				check_variable = { p_return_temp > 0 }
			}
			set_temp_variable = { gap_temp = { value = target_temp divide = p_return_temp max = { value = p_return_temp divide = target_temp } subtract = 1 } }
			if = {
				limit = { check_variable = { gap_temp > global.price_dead_band } }
				add_to_temp_variable = {
					p_return_temp = {
						value = target_temp
						subtract = p_return_temp
						multiply = {
							value = gap_temp
							multiply = global.price_step_slope
							add = global.price_step_base
							min = global.price_step_max
						}
					}
				}
			}
		}
	}

	calc_industrial_capacity = {
		# State Scope
		# Transport supply & demand -> Infrastructure Efficiency, and the rate railways & ports get paid.
		# Runs before this month's industries update, so it works off of last month's demand.
		# Supply from population, after roads, is capped (Q103): roads only carry so much freight

		set_variable = {
			industrial_capacity = {
				value = pops
				multiply = global.base_infrastructure_per_pop
				multiply = {
					value = 1
					add = controller:modifier@base_infrastructure_factor
				}
				min = {
					value = global.base_infrastructure_cap
					add = controller:modifier@base_infrastructure_cap_modifier
					multiply = {
						value = 1
						add = controller:modifier@base_infrastructure_cap_factor
					}
				}
				max = 0
				add = global.base_industrial_capacity
				add = modifier@local_industrial_capacity_modifier
			}
		}
		set_variable = { industrial_capacity_use = modifier@local_industrial_capacity_use_modifier }

		for_each_loop = {
			array = industries
			value = b_type

			add_to_variable = { industrial_capacity_use = industrial_capacity_use@var:b_type }
			add_to_variable = { industrial_capacity = industrial_capacity@var:b_type }
		}

		# Both ratios are guarded: a math expression doesn't return 0 when dividing by 0
		if = {
			limit = {
				check_variable = { industrial_capacity_use > 0 }
				check_variable = { industrial_capacity > 0 }
			}
			if = {
				limit = { check_variable = { industrial_capacity < industrial_capacity_use } }
				set_variable = {
					infrastructure_efficiency = {
						value = industrial_capacity
						divide = industrial_capacity_use
						max = global.infrastructure_efficiency_floor
					}
				}
			}
			else = {
				set_variable = {
					infrastructure_efficiency = {
						value = 1
						subtract = {
							value = industrial_capacity_use
							divide = industrial_capacity
						}
						multiply = global.infrastructure_efficiency_bonus
						add = 1
					}
				}
			}
			set_variable = {
				price_industrial_capacity = {
					value = industrial_capacity_use
					divide = industrial_capacity
					clamp = { min = global.transport_pay_floor max = global.transport_pay_cap }
					multiply = global.base_price_industrial_capacity
				}
			}
		}
		else_if = { # No transport supply at all
			limit = { check_variable = { industrial_capacity_use > 0 } }
			set_variable = { infrastructure_efficiency = global.infrastructure_efficiency_floor }
			set_variable = { price_industrial_capacity = { value = global.transport_pay_cap multiply = global.base_price_industrial_capacity } }
		}
		else = { # No transport demand at all
			set_variable = { infrastructure_efficiency = { value = 1 add = global.infrastructure_efficiency_bonus } }
			set_variable = { price_industrial_capacity = { value = global.transport_pay_floor multiply = global.base_price_industrial_capacity } }
		}
	}

	adjust_trade = { # Adjusts trade offers by given amount
		# resource_temp - resource token
		# amount_temp - Positive values for increasing imports/decreasing exports and Negative values for decreasing imports/increasing exports
		# stop_at_zero_temp - Set to non-zero value to allow reductions in exports/imports to go past 0 and flip to become imports/exports respectively.
		#					 AKA makes it so if you reduce exports by 200 when you're exporting 100, instead of it just being set to 0, it gets set to 100 immports
		# no_update_temp - Set to non-zero to not update trade

		if = {
			limit = {
				check_variable = { market_leader = THIS }
                NOT = { has_idea = closed_economy }
			}

			round_temp_variable = amount_temp

			if = {
				limit = {
					NOT = { check_variable = { amount_temp = 0 } }
				}
				if = { # IMPORTING
					limit = {
						check_variable = { offers@var:resource_temp > 0 }
					}
					add_to_variable = { offers@var:resource_temp = amount_temp }
					if = {
						limit = {
							check_variable = { stop_at_zero_temp = 0 }
							check_variable = { offers@var:resource_temp < 0 }
						}
						set_variable = { offers@var:resource_temp = 0 }
					}
				}
				else_if = { # EXPORTING
					limit = {
						check_variable = { offers@var:resource_temp < 0 }
					}
					add_to_variable = { offers@var:resource_temp = amount_temp }
					if = {
						limit = {
							check_variable = { stop_at_zero_temp = 0 }
							check_variable = { offers@var:resource_temp > 0 }
						}
						set_variable = { offers@var:resource_temp = 0 }
					}
				}
				else = { # NEW TRADE
					add_to_variable = { offers@var:resource_temp = amount_temp }
				}
			}

			set_temp_variable = { amount_temp = offers@var:resource_temp }
			set_trade = yes # for adding trade routes & trade update & making sure stuff is in boundaries
		}
	}
	set_trade = { # Sets trade offers to given amount
		# resource_temp - resource token
		# amount_temp - Positive values for increasing imports/decreasing exports and Negative values for decreasing imports/increasing exports
		# no_update_temp - Set to non-zero to not update trade
		

		if = {
			limit = {
				check_variable = { market_leader = THIS }
                NOT = { has_idea = closed_economy }
			}

			round_temp_variable = amount_temp
			
			if = {
				limit = {
					NOT = { has_global_flag = debug_trade }
				}
				# Max & Min Trade
				set_temp_variable = { trade_max_temp = demand@var:resource_temp } # Imports - Max is from local demand
				multiply_temp_variable = { trade_max_temp = 2.5 }
				subtract_from_temp_variable = { trade_max_temp = supply@var:resource_temp }
				if = {
					limit = {
						check_variable = { trade@var:resource_temp > 0 }
					}
					add_to_temp_variable = { trade_max_temp = trade@var:resource_temp }
				}
				clamp_temp_variable = { var = trade_max_temp min = 0 }

				set_temp_variable = { trade_min_temp = supply@var:resource_temp } # Exports - Min is from local supply
				multiply_temp_variable = { trade_min_temp = -3 }
				add_to_temp_variable = { trade_min_temp = need@var:resource_temp }
				clamp_temp_variable = { var = trade_min_temp max = 0 }

				set_temp_variable = { world_floor_temp = { value = global.base_price@var:resource_temp multiply = 0.25 } }
				if = {
					limit = { # The target world price is above its floor
						check_variable = { global_market_target_price@var:resource_temp > world_floor_temp }
					}
					clamp_temp_variable = { var = amount_temp max = trade_max_temp min = trade_min_temp }
				}
				else = {
					set_temp_variable = { trade_max_temp = need@var:resource_temp }
					multiply_temp_variable = { trade_max_temp = 2.5 }
					subtract_from_temp_variable = { trade_max_temp = production@var:resource_temp }
					clamp_temp_variable = { var = trade_max_temp min = 0 }
					clamp_temp_variable = { var = amount_temp max = trade_max_temp min = trade_min_temp }
				}
			}

			# Set Offers
			set_variable = { offers@var:resource_temp = amount_temp }
			#log = "Min: [?trade_min_temp], Max: [?trade_max_temp]"

			if = {
				limit = {
					NOT = { check_variable = { offers@var:resource_temp = 0 } }
					NOT = { is_in_array = { trade_routes = resource_temp } }
				}
				add_trade = yes
			}
			else_if = {
				limit = {
					check_variable = { no_update_temp = 0 }
				}
				trade_update = yes
			}
		}
	}

			get_price = yes
			set_variable = { global.market_target_price@var:r_type = p_return_temp } # Trade uses each market leader's own view of the world (global_market_price@)
		}
	}

	trade_update = {
		# Normal Trade Update Script

		for_each_loop = {
			array = global.resource_token_array
			value = r_type

			set_variable = { shipping_use@var:r_type = 0 }
			set_variable = { trade@var:r_type = 0 }
			set_variable = { trade_productivity@var:r_type = 0 }
			set_variable = { trade_value@var:r_type = 0 }
			set_variable = { trade_tariffs@var:r_type = 0 }
			set_variable = { trade_profit@var:r_type = 0 }
			set_variable = { trade_jobs@var:r_type = 0 }
		}

		calc_shipping_capacity_use = yes # Would normally be executed on every country before the next two

		local_global_market_update = yes
		trade_update_quantity = yes
		trade_update_value = yes
		local_global_market_calculations = yes

		if = {
			limit = {
				is_ai = no
			}
			country_update_trade_display = yes
		}
	}
	calc_shipping_capacity_use = {
		# Only updates shipping capacity usage

		set_variable = { shipping_capacity_use = 0 }
		# Shipping Capacity
			for_each_loop = {
				array = trade_routes
				value = r_type

				set_variable = { shipping_use@var:r_type = offers@var:r_type }
				if = {
					limit = {
						check_variable = { shipping_use@var:r_type < 0 }
					}
					multiply_variable = { shipping_use@var:r_type = -1 }
				}
				multiply_variable = { shipping_use@var:r_type = global.shipping_cost@var:r_type }
				set_temp_variable = { modifier_temp = modifier@shipping_cost_factor }
				add_to_temp_variable = { modifier_temp = 1 }
				multiply_variable = { shipping_use@var:r_type = modifier_temp }

				add_to_variable = { shipping_capacity_use = shipping_use@var:r_type }
			}

			#Shipping Efficiency
			if = {
				limit = {
					check_variable = { shipping_capacity_use > shipping_capacity }
					NOT = { has_global_flag = debug_trade }
				}
				set_variable = { shipping_efficiency = shipping_capacity }
				divide_variable = { shipping_efficiency = shipping_capacity_use }
			}
			else = {
				set_variable = { shipping_efficiency = 1 }
			}
			
			if = {
				limit = {
					check_variable = { shipping_capacity = 0 }
				}
				set_variable = { shipping_usage = 1 }
			}
			else = {
				set_variable = { shipping_usage = shipping_capacity_use }
				divide_variable = { shipping_usage = shipping_capacity }
			}
		#
	}
	trade_update_quantity = {
		# Only updates trade routes & supply/demand/prices.

		for_each_loop = {  # Actual Trade
			array = trade_routes
			value = r_type

			set_variable = { trade@var:r_type = offers@var:r_type }

			if = {
				limit = {
					check_variable = { offers@var:r_type > 0 } # Importing
					check_variable = { trade_supply@var:r_type < trade_demand@var:r_type } # Not enough countries to buy from
				}
				set_temp_variable = { trade_portion_temp = trade_supply@var:r_type }
				multiply_temp_variable = { trade_portion_temp = 3 }
				divide_temp_variable = { trade_portion_temp = trade_demand@var:r_type }
				if = {
					limit = {
						OR  ={
							check_variable = { trade_portion_temp > 1 }
							has_global_flag = debug_trade
						}
					}
					set_temp_variable = { trade_portion_temp = 1 }
				}
				else_if = {
					limit = { check_variable = { trade_portion_temp < 0 } }
					set_temp_variable = { trade_portion_temp = 0 }
				}
				multiply_variable = { trade@var:r_type = trade_portion_temp }
			}
			else_if = {
				limit = {
					check_variable = { offers@var:r_type < 0 } # Exporting
					check_variable = { trade_demand@var:r_type < trade_supply@var:r_type } # Not enough countries to sell to
				}
				set_temp_variable = { trade_portion_temp = trade_demand@var:r_type }
				multiply_temp_variable = { trade_portion_temp = 2.5 }
				divide_temp_variable = { trade_portion_temp = trade_supply@var:r_type }
				if = {
					limit = {
						OR  ={
							check_variable = { trade_portion_temp > 1 }
							has_global_flag = debug_trade
						}
					}
					set_temp_variable = { trade_portion_temp = 1 }
				}
				else_if = {
					limit = { check_variable = { trade_portion_temp < 0 } }
					set_temp_variable = { trade_portion_temp = 0 }
				}
				multiply_variable = { trade@var:r_type = trade_portion_temp }
			}

			multiply_variable = { trade@var:r_type = shipping_efficiency }
		}

		market_leader_check = yes
		if = {
			limit = {
				check_variable = { market_leader = THIS }
			}
			market_calculations = yes
		}
	}
	trade_update_value = {
		set_variable = { trade_productivity = 0 } # Average trade productivity
		set_variable = { trade_jobs = 0 } # Total trade jobs
		set_variable = { trade_value = 0 } # Total trade value
		set_variable = { trade_profit = 0 } # Total trade profit

		# Trade Value & Tariffs
		for_each_loop = {
			array = global.resource_token_array
			value = r_type

			if = { # Normal Trade
				limit = {
					NOT = { check_variable = { offers@var:r_type = 0 } }
				}
				set_temp_variable = { trade_temp = trade@var:r_type }
			}
			else_if = { # Importing with 1.1k employees
				limit = {
					check_variable = { global_market_target_price@var:r_type < target_price@var:r_type }
				}
				set_temp_variable = { trade_temp = global.shipping_amount@var:r_type }
			}
			else = { # Exporting with 1.1k employees
				set_temp_variable = { trade_temp = global.shipping_amount@var:r_type }
				multiply_temp_variable = { trade_temp = -1 }
			}


			if = { #Import
				limit = {
					check_variable = { trade_temp > 0 }
				}
				# Value
				set_variable = { trade_value@var:r_type = trade_temp }
				multiply_variable = { trade_value@var:r_type = price@var:r_type }

				set_temp_variable = { temp_value = trade_temp }
				multiply_temp_variable = { temp_value = global_market_price@var:r_type }

				# Tariffs
				set_variable = { trade_tariffs@var:r_type = import_tariff_percentage_actual }
				divide_variable = { trade_tariffs@var:r_type = 100 }
				multiply_variable = { trade_tariffs@var:r_type = trade_value@var:r_type }

				# Shipping Cost
				set_variable = { shipping_cost@var:r_type = shipping_use@var:r_type }
				multiply_variable = { shipping_cost@var:r_type = price_shipping_capacity }

				# Adding temp value
				subtract_from_variable = { trade_value@var:r_type = temp_value }

				# Profit
				set_variable = { trade_profit@var:r_type = trade_value@var:r_type }
				subtract_from_variable = { trade_profit@var:r_type = trade_tariffs@var:r_type }
				subtract_from_variable = { trade_profit@var:r_type = shipping_cost@var:r_type }
			}
			else_if = { #Export
				limit = {
					check_variable = { trade_temp < 0 }
				}
				# Value
				set_variable = { trade_value@var:r_type = trade_temp }
				multiply_variable = { trade_value@var:r_type = global_market_price@var:r_type }
				multiply_variable = { trade_value@var:r_type = -1 }

				set_temp_variable = { temp_value = trade_temp }
				multiply_temp_variable = { temp_value = price@var:r_type }

				# Tariffs
				set_variable = { trade_tariffs@var:r_type = export_tariff_percentage_actual }
				divide_variable = { trade_tariffs@var:r_type = -100 }
				multiply_variable = { trade_tariffs@var:r_type = trade_value@var:r_type }

				# Shipping Cost
				set_variable = { shipping_cost@var:r_type = shipping_use@var:r_type }
				multiply_variable = { shipping_cost@var:r_type = price_shipping_capacity }
				
				# Adding temp value
				add_to_variable = { trade_value@var:r_type = temp_value }

				# Profit
				set_variable = { trade_profit@var:r_type = trade_value@var:r_type }
				add_to_variable = { trade_profit@var:r_type = trade_tariffs@var:r_type }
				subtract_from_variable = { trade_profit@var:r_type = shipping_cost@var:r_type }
			}
			else = {
				set_variable = { trade_value@var:r_type = 0 }
				set_variable = { trade_tariffs@var:r_type = 0 }
				set_variable = { shipping_cost@var:r_type = 0 }
				set_variable = { trade_profit@var:r_type = 0 }
			}

			# The money above is at actual prices. The trade AI judges the route at target prices instead,
			# so trade answers a shortage within the week rather than waiting for the monthly price step
			if = {
				limit = { check_variable = { trade_temp > 0 } }
				set_temp_variable = { trade_ai_profit_temp = { value = import_tariff_percentage_actual divide = -100 add = 1 multiply = target_price@var:r_type subtract = global_market_target_price@var:r_type multiply = trade_temp subtract = shipping_cost@var:r_type } }
			}
			else_if = {
				limit = { check_variable = { trade_temp < 0 } }
				set_temp_variable = { trade_ai_profit_temp = { value = export_tariff_percentage_actual divide = -100 add = 1 multiply = global_market_target_price@var:r_type subtract = target_price@var:r_type multiply = trade_temp multiply = -1 subtract = shipping_cost@var:r_type } }
			}
			else = {
				set_temp_variable = { trade_ai_profit_temp = 0 }
			}

			if = { # If offers are set to 0, then act as though you have 1.1k employees, and reset all the variables above
				limit = {
					check_variable = { offers@var:r_type = 0 }
				}
				set_variable = { trade_productivity@var:r_type = trade_ai_profit_temp }
				divide_variable = { trade_productivity@var:r_type = 1.01 }
				
				set_variable = { trade_value@var:r_type = 0 }
				set_variable = { trade_tariffs@var:r_type = 0 }
				set_variable = { trade_profit@var:r_type = 0 }
				set_variable = { trade_jobs@var:r_type = 0 }
			}
			else = { # Otherwise, set employee count & productivity
				set_variable = { trade_jobs@var:r_type = offers@var:r_type }
				divide_variable = { trade_jobs@var:r_type = global.shipping_amount@var:r_type }
				if = {
					limit = {
						check_variable = { trade_jobs@var:r_type < 0 }
					}
					multiply_variable = { trade_jobs@var:r_type = -1 }
				}
				add_to_variable = { trade_jobs@var:r_type = 0.01 }

				set_variable = { trade_productivity@var:r_type = trade_ai_profit_temp }
				divide_variable = { trade_productivity@var:r_type = trade_jobs@var:r_type }

				add_to_variable = { trade_jobs = trade_jobs@var:r_type }
				add_to_variable = { trade_value = trade_value@var:r_type }
				add_to_variable = { trade_profit = trade_profit@var:r_type }
			}
		}
		set_variable = { trade_productivity = trade_profit }
		if = {
			limit = {
				check_variable = { trade_jobs > 0 }
			}
			divide_variable = { trade_productivity = trade_jobs }
		}
	}
	local_global_market_update = { # Updates global market-related variables for a country
		# Country Scope
		for_each_loop = {
			array = global.resource_token_array
			value = r_type
			set_variable = { global_market_supply@var:r_type = global.market_supply@var:r_type }
			set_variable = { global_market_demand@var:r_type = global.market_demand@var:r_type }
			set_variable = { trade_supply@var:r_type = global.trade_supply@var:r_type }
			set_variable = { trade_demand@var:r_type = global.trade_demand@var:r_type }
		}
		
		set_temp_variable = { country_temp = THIS }
		for_each_scope_loop = {
			array = global.markets
			# Remove countries you cannot trade with
			if = {
				limit = {
					NOT = { check_variable = { country_temp = THIS } }
					can_trade_with = no
				}
				for_each_loop = {
					array = global.resource_token_array
					value = r_type
					set_temp_variable = { offers_temp = offers@var:r_type }
					multiply_temp_variable = { offers_temp = shipping_efficiency }

					subtract_from_variable = { PREV.global_market_supply@var:r_type = production@var:r_type }
					subtract_from_variable = { PREV.global_market_demand@var:r_type = need@var:r_type }
					if = { # This country is importing
						limit = {
							check_variable = { offers@var:r_type > 0 }
						}
						subtract_from_variable = { PREV.trade_demand@var:r_type = offers_temp }
					}
					else = { # This country is exporting
						add_to_variable = { PREV.trade_supply@var:r_type = offers_temp }
					}
				}
			}
		}

		local_global_market_calculations = yes
	}

			array = global.resource_token_array

			subtract_from_variable = { global_market_balance@var:r_type = global_market_demand@var:r_type }

			# Target world price. The actual one (global_market_price@) moves toward it once a month, in market_prices_monthly
			set_temp_variable = { s_temp = global_market_supply@var:r_type }
			set_temp_variable = { d_temp = global_market_demand@var:r_type }
			get_price = yes
			set_variable = { global_market_target_price@var:r_type = p_return_temp }
			if = { # While the economy is being set up (prices_snap_temp), and for a good with no price yet, the price starts at its target
				limit = {
					OR = {
						check_variable = { prices_snap_temp = 1 }
						NOT = { check_variable = { global_market_price@var:r_type > 0 } }
					}
				}
				set_variable = { global_market_price@var:r_type = p_return_temp }
				set_variable = { global_market_price_compared_to_base@var:r_type = { value = p_return_temp divide = global.base_price@var:r_type subtract = 1 } }
			}
		}
	}

	country_update_trade_display = {
		#Trade Routes
		clear_array = show_trade_routes
		
		for_each_loop = {
			array = trade_routes
			index = tr_i
			value = tr_token
			break = tr_b

			for_each_loop = {
				array = show_trade_routes
				index = ii
				value = vv
				break = bb

				if = {
					limit = {
						check_variable = { vv > tr_token }
					}
					add_to_array = { 
						array = show_trade_routes 
						value = tr_token
						index = ii
					}
					set_temp_variable = { bb = 1 }
				}
			}
			if = {
				limit = {
					check_variable = { bb = 0 }
				}
				add_to_array = { show_trade_routes = tr_token }
			}
		}

		#Balance Array
		clear_array = high_balance_array
		clear_array = high_balance_amount_array
		clear_array = high_balance_price_array
		clear_array = low_balance_array
		clear_array = low_balance_amount_array
		clear_array = low_balance_price_array

		for_each_loop = {
			array = global.resource_token_array
			index = tr_i
			value = tr_token
			break = tr_b

			if = {
				limit = {
					NOT = { 
						check_variable = { supply@var:tr_token = 0 } 
						check_variable = { demand@var:tr_token = 0 } 
					}
				}
				set_temp_variable = { net_amount_temp = supply@var:tr_token }
				subtract_from_temp_variable = { net_amount_temp = demand@var:tr_token }
				
				for_each_loop = {
					array = high_balance_amount_array
					index = ii
					value = vv
					break = bb

					if = {
						limit = {
							check_variable = { net_amount_temp > vv }
						}
						add_to_array = { 
							array = high_balance_array 
							value = tr_token
							index = ii
						}
						add_to_array = { 
							array = high_balance_amount_array 
							value = net_amount_temp
							index = ii
						}
						set_temp_variable = { bb = 1 }
					}
				}
				if = {
					limit = {
						check_variable = { bb = 0 }
					}
					add_to_array = { high_balance_array = tr_token }
					add_to_array = { high_balance_amount_array = net_amount_temp }
				}
				add_to_array = { high_balance_price_array = 0 }
			}

		}

		for_loop_effect = {
			start = 0
			end = high_balance_array^num
			value = vv
			break = bb

			set_temp_variable = { ii = high_balance_array^num }
			subtract_from_temp_variable = { ii = vv }
			subtract_from_temp_variable = { ii = 1 }

			add_to_array = { 
				array = low_balance_array 
				value = high_balance_array^ii
			}
			add_to_array = { 
				array = low_balance_amount_array 
				value = high_balance_amount_array^ii
			}
			add_to_array = { low_balance_price_array = 0 }
		}


		for_each_loop = {
			array = high_balance_array
			value = tr_token
			index = tr_i

			set_variable = { high_balance_price_array^tr_i = price@var:tr_token }
			divide_variable = { high_balance_price_array^tr_i = global.base_price@var:tr_token }
			subtract_from_variable = { high_balance_price_array^tr_i = 1 }
		}
		for_each_loop = {
			array = low_balance_array
			value = tr_token
			index = tr_i

			set_variable = { low_balance_price_array^tr_i = price@var:tr_token }
			divide_variable = { low_balance_price_array^tr_i = global.base_price@var:tr_token }
			subtract_from_variable = { low_balance_price_array^tr_i = 1 }
		}

		#Resource List
		clear_array = resource_list_token_array
		clear_array = resource_list_left_price_bar_array
		clear_array = resource_list_right_price_bar_array
		clear_array = resource_list_left_balance_bar_array
		clear_array = resource_list_right_balance_bar_array

		set_temp_variable = { highest_balance_token = 0 }
		set_temp_variable = { highest_balance_temp = 0 }
		set_temp_variable = { lowest_balance_token = 0 }
		set_temp_variable = { lowest_balance_temp = 0 }

		#Default sorted by price (%), can also be sorted by balance, supply, demand
		# resource_list_natural_resources_filter
		# resource_list_produced_resources_filter
		# resource_list_service_resources_filter
		# resource_list_military_resources_filter
		# resource_list_invert_sort
		# resource_list_balance_sort
		# resource_list_supply_sort
		# resource_list_demand_sort
		# resource_list_resource_sort
		for_each_loop = {
			array = global.resource_token_array
			index = tr_i
			value = tr_token
			break = tr_b
			
			# GLOBAL MARKET
			if = {
				limit = {
					check_variable = { selected_market = 0 }
				}
				if = {
					limit = {
						OR = {
							check_variable = { global_market_supply@var:tr_token > 0 }
							check_variable = { global_market_demand@var:tr_token > 0 }
						}
						OR = {
							AND = {
								NOT = { has_country_flag = resource_list_natural_resources_filter }
								NOT = { has_country_flag = resource_list_produced_resources_filter }
								NOT = { has_country_flag = resource_list_service_resources_filter }
								NOT = { has_country_flag = resource_list_military_resources_filter }
							}
							AND = {
								has_country_flag = resource_list_natural_resources_filter
								is_in_array = { global.natural_resources_array = tr_token }
							}
							AND = {
								has_country_flag = resource_list_produced_resources_filter
								is_in_array = { global.produced_resources_array = tr_token }
							}
							AND = {
								has_country_flag = resource_list_service_resources_filter
								is_in_array = { global.service_resources_array = tr_token }
							}
							AND = {
								has_country_flag = resource_list_military_resources_filter
								is_in_array = { global.military_resources_array = tr_token }
							}
						}
					}
					if = {
						limit = {
							check_variable = { global_market_balance@var:tr_token > highest_balance_temp }
						}
						set_temp_variable = { highest_balance_token = tr_token }
						set_temp_variable = { highest_balance_temp = global_market_balance@var:tr_token }
					}
					if = {
						limit = {
							check_variable = { global_market_balance@var:tr_token < lowest_balance_temp }
						}
						set_temp_variable = { lowest_balance_token = tr_token }
						set_temp_variable = { lowest_balance_temp = global_market_balance@var:tr_token }
					}

					if = {
						limit = {
							has_country_flag = resource_list_balance_sort
						}
						set_temp_variable = { new_element_value = global_market_balance@var:tr_token }
					}
					else_if = {
						limit = {
							has_country_flag = resource_list_supply_sort
						}
						set_temp_variable = { new_element_value = global_market_supply@var:tr_token }
					}
					else_if = {
						limit = {
							has_country_flag = resource_list_demand_sort
						}
						set_temp_variable = { new_element_value = global_market_demand@var:tr_token }
					}
					else_if = {
						limit = {
							has_country_flag = resource_list_resource_sort
						}
						set_temp_variable = { new_element_value = global.resources_token_value@var:tr_token }
					}
					else = { #Price sort
						set_temp_variable = { new_element_value = global_market_price_compared_to_base@var:tr_token }
					}
					
					for_each_loop = {
						array = resource_list_token_array
						index = ii
						value = vv
						break = bb

						if = {
							limit = {
								has_country_flag = resource_list_balance_sort
							}
							set_temp_variable = { current_element_value = global_market_balance@var:vv }
						}
						else_if = {
							limit = {
								has_country_flag = resource_list_supply_sort
							}
							set_temp_variable = { current_element_value = global_market_supply@var:vv }
						}
						else_if = {
							limit = {
								has_country_flag = resource_list_demand_sort
							}
							set_temp_variable = { current_element_value = global_market_demand@var:vv }
						}
						else_if = {
							limit = {

	approximate_square_root = {
		# Given sqrt_temp, approximates the square root of the value using Heron's method. Outputs using the same variable

		# Sets initial estimate
		set_temp_variable = { sqrt_approx_temp = 1 }

			set_temp_variable = { sqrt_approx_temp = 10 }
			divide_temp_variable = { sqrt_check = 100 }
		}
		
		if = {
			limit = {
				check_variable = { sqrt_check < 25 }
			}
			if = {
				limit = {
					check_variable = { sqrt_check < 2 }
				}
				multiply_temp_variable = { sqrt_approx_temp = 1 }

					check_variable = { trade_ai@var:resource_temp = 0 }
					NOT = { is_in_array = { global.non_tradeable_resources = resource_temp } }
					
					check_variable = { offers@var:resource_temp = 0 }
					NOT = { is_in_array = { trade_routes = resource_temp } }

					OR = {
						AND = {
							check_variable = { supply@var:resource_temp > 0 }
							check_variable = { demand@var:resource_temp = 0 }
						}
						AND = {
							check_variable = { supply@var:resource_temp = 0 }
							check_variable = { demand@var:resource_temp > 0 }
						}
					}
				}

				set_temp_variable = { no_update_temp = 1 }
				add_trade = yes
			}
		}

		for_each_loop = { # Evaluate Routes
			array = trade_routes
			value = resource_temp

			if = {
				limit = {
					check_variable = { trade_ai@var:resource_temp = 0 }
					NOT = { is_in_array = { global.non_tradeable_resources = resource_temp } }
					OR = {
						check_variable = { offers@var:resource_temp = 0 }
						check_variable = { trade_productivity@var:resource_temp > productivity_upper_bound }
						check_variable = { trade_productivity@var:resource_temp < productivity_lower_bound }
					}
				}

				evaluate_best_trade_quantity = yes
				
				set_temp_variable = { amount_temp = trade_quantity_temp }
				set_temp_variable = { no_update_temp = 1 }
				set_trade = yes
			}

			if = { # Minimum trade - sets trade to resource consumption (for military production)
				limit = {
					check_variable = { trade_ai@var:resource_temp = 0 }
					NOT = { is_in_array = { global.non_tradeable_resources = resource_temp } }
				}
				set_temp_variable = { amount_temp = resource_consumed@var:resource_temp }
				if = { limit = { check_variable = { resource@var:resource_temp < 0 } } subtract_from_temp_variable = { amount_temp = resource_consumed@var:resource_temp } }
				subtract_from_temp_variable = { amount_temp = supply@var:resource_temp }

				if = {
					limit = {
						check_variable = { amount_temp > 0 }
						check_variable = { amount_temp < offers@var:resource_temp }
					}
					set_temp_variable = { no_update_temp = 1 }
					set_trade = yes
				}
			}
			
			# The two minimum trades check the market's own production and need, not supply and demand: those include
			# last week's trade, so a route the evaluation above had just cut to 0 was only restored every other week
			if = { # Minimum trade - sets exports to 1 (for 0 demand but any supply)
				limit = {
					check_variable = { trade_ai@var:resource_temp = 0 }
					NOT = { is_in_array = { global.non_tradeable_resources = resource_temp } }

					check_variable = { offers@var:resource_temp = 0 }
					check_variable = { production@var:resource_temp > 0 }
					check_variable = { need@var:resource_temp = 0 }
				}
				set_temp_variable = { amount_temp = -1 }
				set_temp_variable = { no_update_temp = 1 }
				set_trade = yes
			}
			if = { # Minimum trade - sets imports to 1 (for 0 supply but any demand)
				limit = {
					check_variable = { trade_ai@var:resource_temp = 0 }
					NOT = { is_in_array = { global.non_tradeable_resources = resource_temp } }

					check_variable = { offers@var:resource_temp = 0 }
					check_variable = { production@var:resource_temp = 0 }
					check_variable = { need@var:resource_temp > 0 }
				}
				set_temp_variable = { amount_temp = 1 }
				set_temp_variable = { no_update_temp = 1 }
				set_trade = yes
			}
		}

		for_each_loop = { # Remove Routes
			array = trade_routes
			value = resource_temp
			
			if = {
				limit = {
					check_variable = { trade_ai@var:resource_temp = 0 }
				}
				if = {
					limit = {
						check_variable = { offers@var:resource_temp = 0 }
					}
					set_temp_variable = { no_update_temp = 1 }
					remove_trade = yes
				}
			}
		}
	}
