#
#   effect eco_sim_probes = yes          engine-behaviour probes  -> EOPROBE|name|value lines in logs/game.log
#   effect eco_sim_parity_dump = yes     eco snapshot             -> EOSIM|key|value lines in logs/game.log
#   effect eco_sim_pm_audit = yes        every building's methods -> EOPM lines, see documentation/scripts/verify_pm_audit.py
#
# Then, from redux-eco-sim/:
#   bin\ecosim_cli.exe calibrate        (compares EOPROBE lines with the simulator)
#   bin\ecosim_cli.exe parity           (compares the first EOSIM dump with the simulator's startup)
#   bin\ecosim_cli.exe parity --dump 2 --months 1   (second dump vs. the simulator after one monthly tick)

eco_sim_probes_math = {
	# all_of_sum and array_in_expression read 0 from the console, but the same forms work in real ticks.
	# Each line changes one thing, and `effect set_global_flag = eosim_tick_probes` runs them all on the next monthly tick.
	# Then bin\ecosim_cli.exe calibrate --dump 1 is the console run and --dump 2 the tick run.

	# Is it reading temps, or only array elements?
	set_temp_variable = { eosim_i = 1 }
	set_temp_variable = { a = { value = eosim_i multiply = 10 } }
	log = "EOPROBE|temp_in_expression|[?a]"
	set_variable = { eosim_scoped = 4 }
	set_temp_variable = { a = { value = eosim_scoped multiply = 10 } }
	log = "EOPROBE|scoped_var_in_expression|[?a]"
	clear_variable = eosim_scoped

	# Which array reads?
	set_temp_variable = { a = { value = eosim_loop^1 multiply = 10 } }
	log = "EOPROBE|array_literal_in_expression|[?a]"
	set_temp_variable = { a = { value = eosim_loop^num multiply = 10 } }
	log = "EOPROBE|array_num_in_expression|[?a]"
	set_temp_variable = { ii = 1 }
	set_temp_variable = { a = { value = eosim_loop^ii multiply = 10 } }
	log = "EOPROBE|array_index_ii_in_expression|[?a]"
	set_temp_variable = { a = 0 }
	for_each_loop = {
		array = eosim_loop
		index = eosim_j
		add_to_temp_variable = { a = { value = eosim_loop^eosim_j multiply = 10 } }
	}
	log = "EOPROBE|array_loop_index_in_expression|[?a]"

	# pm_construction_value read 0 for every PM at startup with this form, and it's the only one that did
	set_variable = { eosim_total = 1 }
	add_to_variable = { eosim_total = { value = 2 multiply = 3 } }
	log = "EOPROBE|add_to_variable_math|[?eosim_total]"
	set_variable = { global.eosim_total@token:food = 1 }
	add_to_variable = { global.eosim_total@token:food = { value = 2 multiply = 3 } }
	log = "EOPROBE|add_to_global_token_math|[?global.eosim_total@token:food]"
	set_variable = { global.eosim_total@token:food = 1 }
	for_each_loop = {
		array = eosim_loop
		value = eosim_v
		index = eosim_j
		add_to_variable = { global.eosim_total@token:food = { value = eosim_loop^eosim_j multiply = eosim_v } }
	}
	log = "EOPROBE|add_to_global_array_math|[?global.eosim_total@token:food]"
	clear_variable = eosim_total
	clear_variable = global.eosim_total@token:food

	# Trigger blocks. all_of_staged_read is the form the getters rely on
	set_temp_variable = { eosim_sum = 0 }
	if = {
		limit = {
			all_of = {
				array = eosim_loop
				value = eosim_v
				index = eosim_j
				set_temp_variable = { eosim_t = eosim_loop^eosim_j }
				multiply_temp_variable = { eosim_t = eosim_v }
				add_to_temp_variable = { eosim_sum = eosim_t }
			}
		}
	}
	log = "EOPROBE|all_of_staged_read|[?eosim_sum]"
	set_temp_variable = { eosim_sum = 0 }
	if = {
		limit = {
			all_of = {
				array = eosim_loop
				add_to_temp_variable = { eosim_sum = { value = 2 multiply = 2 } }
			}
		}
	}
	log = "EOPROBE|all_of_literal_math|[?eosim_sum]"
	set_temp_variable = { a = 7 }
	if = { limit = { set_temp_variable = { a = { value = 3 multiply = 2 } } } }
	log = "EOPROBE|math_in_limit_literal|[?a]"
	set_temp_variable = { a = 7 }
	set_temp_variable = { eosim_t = 3 }
	if = { limit = { set_temp_variable = { a = { value = eosim_t multiply = 2 } } } }
	log = "EOPROBE|math_in_limit_temp|[?a]"
}

eco_sim_pm_audit = {
	# Every building + method, per K of base workforce at base prices. verify_pm_audit.py diffs this against the sheet.
	for_each_loop = {
		array = global.industry_token_array
		value = eosim_b

		for_each_loop = {
			array = global.building_production_methods@var:eosim_b
			value = eosim_m
			eco_sim_pm_audit_line = yes
		}
		for_each_loop = {
			array = global.building_automation_methods@var:eosim_b
			value = eosim_m
			eco_sim_pm_audit_line = yes
		}
	}
}

eco_sim_pm_audit_line = {
	set_temp_variable = { eosim_rev = 0 }
	for_each_loop = {
		array = global.building_base_production@var:eosim_b
		value = eosim_g
		index = eosim_i
		set_temp_variable = { eosim_amount = global.building_base_production_amount@var:eosim_b^eosim_i }
		multiply_temp_variable = { eosim_amount = global.building_production_multiplier@var:eosim_b }
		multiply_temp_variable = { eosim_amount = global.pm_output_amount@var:eosim_m }
		multiply_temp_variable = { eosim_amount = global.base_price@var:eosim_g }
		add_to_temp_variable = { eosim_rev = eosim_amount }
	}
	set_temp_variable = { eosim_in = 0 }
	for_each_loop = {
		array = global.pm_input@var:eosim_m
		value = eosim_g
		index = eosim_i
		set_temp_variable = { eosim_amount = global.pm_input_amount@var:eosim_m^eosim_i }
		multiply_temp_variable = { eosim_amount = global.base_price@var:eosim_g }
