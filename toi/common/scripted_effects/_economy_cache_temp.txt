industry_cache_archetype = {
	# Inputs:
	# bu_token
	# pm_temp
	# am_temp

	# Allocate Cache ID
	allocate_cache_number = yes # cache_num
	
	add_to_array = { industry_cache = cache_num }
	set_variable = { c_industry_token@var:cache_num = bu_token }
	set_variable = { c_production_method@var:cache_num = pm_temp }
	set_variable = { c_automation_method@var:cache_num = am_temp }

	## Jobs ##
		# Jobs
		set_temp_variable = { base_jobs_temp = global.building_base_employees@var:bu_token }

		set_temp_variable = { i_laborers_jobs = pm_workforce_laborers@var:pm_temp }
		add_to_temp_variable = { i_laborers_jobs = pm_workforce_laborers@var:am_temp }
		multiply_temp_variable = { i_laborers_jobs = base_jobs_temp }
		clamp_temp_variable = { var = i_laborers_jobs min = 0 max = i_laborers_jobs }

		set_temp_variable = { i_workers_jobs = pm_workforce_workers@var:pm_temp }
		add_to_temp_variable = { i_workers_jobs = pm_workforce_workers@var:am_temp }
		multiply_temp_variable = { i_workers_jobs = base_jobs_temp }
		clamp_temp_variable = { var = i_workers_jobs min = 0 max = i_workers_jobs }

		set_temp_variable = { i_professionals_jobs = pm_workforce_professionals@var:pm_temp }
		add_to_temp_variable = { i_professionals_jobs = pm_workforce_professionals@var:am_temp }
		multiply_temp_variable = { i_professionals_jobs = base_jobs_temp }
		clamp_temp_variable = { var = i_professionals_jobs min = 0 max = i_professionals_jobs }

		set_temp_variable = { i_specialists_jobs = pm_workforce_specialists@var:pm_temp }
		add_to_temp_variable = { i_specialists_jobs = pm_workforce_specialists@var:am_temp }
		multiply_temp_variable = { i_specialists_jobs = base_jobs_temp }
		clamp_temp_variable = { var = i_specialists_jobs min = 0 max = i_specialists_jobs }

		# Payroll Multiplier
		set_temp_variable = { i_payroll_multiplier = 0 }
		set_temp_variable = { payroll_multiplier_temp = i_laborers_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 2 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_workers_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 5 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_professionals_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 10 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_specialists_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 20 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		multiply_temp_variable = { i_payroll_multiplier = base_jobs_temp }

		set_temp_variable = { i_jobs = 0 }
		add_to_temp_variable = { i_jobs = i_laborers_jobs }
		add_to_temp_variable = { i_jobs = i_workers_jobs }
		add_to_temp_variable = { i_jobs = i_professionals_jobs }
		add_to_temp_variable = { i_jobs = i_specialists_jobs }

		set_variable = { c_jobs@var:cache_num = i_jobs }
		set_variable = { c_laborers_jobs@var:cache_num = i_laborers_jobs }
		set_variable = { c_workers_jobs@var:cache_num = i_workers_jobs }
		set_variable = { c_professionals_jobs@var:cache_num = i_professionals_jobs }
		set_variable = { c_specialists_jobs@var:cache_num = i_specialists_jobs }
		set_variable = { c_payroll_multiplier@var:cache_num = i_payroll_multiplier }
	##

	## Production
		set_temp_variable = { i_base_output_value = global.building_base_output_value@var:bu_token }

		# Output/input/throughput modifiers
		set_temp_variable = { output_modifier_temp = controller:industry_output_factor@var:bu_token }
		add_to_temp_variable = { output_modifier_temp = 1 }

		set_temp_variable = { throughput_modifier_temp = controller:industry_throughput_factor@var:bu_token }
		add_to_temp_variable = { throughput_modifier_temp = 1 }

		set_temp_variable = { input_modifier_temp = controller:industry_input_factor@var:bu_token }
		add_to_temp_variable = { input_modifier_temp = 1 }

		# Total Input Modifier
		set_temp_variable = { i_input_modifier = i_base_output_value }
		multiply_temp_variable = { i_input_modifier = input_modifier_temp }
		multiply_temp_variable = { i_input_modifier = throughput_modifier_temp }

		# Total Ouput Modifier
		set_temp_variable = { i_output_modifier = output_modifier_temp }
		multiply_temp_variable = { i_output_modifier = throughput_modifier_temp }

		# Industrial Capacity Use
		set_temp_variable = { i_industrial_capacity_use = controller:pm_industrial_capacity_use@var:pm_temp }
		add_to_temp_variable = { i_industrial_capacity_use = controller:pm_industrial_capacity_use@var:am_temp }
		multiply_temp_variable = { i_industrial_capacity_use = i_input_modifier }
		divide_temp_variable = { i_industrial_capacity_use = global.base_price_industrial_capacity }

		# Industrial Capacity
		set_temp_variable = { i_industrial_capacity = controller:pm_industrial_capacity@var:pm_temp }
		add_to_temp_variable = { i_industrial_capacity = controller:pm_industrial_capacity@var:am_temp }
		multiply_temp_variable = { i_industrial_capacity = i_base_output_value }
		multiply_temp_variable = { i_industrial_capacity = i_output_modifier }
		divide_temp_variable = { i_industrial_capacity = global.base_price_industrial_capacity }

		# Shipping Capacity
		set_temp_variable = { i_shipping_capacity = controller:pm_shipping_capacity@var:pm_temp }
		add_to_temp_variable = { i_shipping_capacity = controller:pm_shipping_capacity@var:am_temp }
		multiply_temp_variable = { i_shipping_capacity = i_base_output_value }
		multiply_temp_variable = { i_shipping_capacity = i_output_modifier }
		divide_temp_variable = { i_shipping_capacity = global.base_price_shipping_capacity }

		# Output Production
		clear_array = c_production@var:cache_num
		set_temp_variable = { i_production_output_multiplier = pm_output_multiplier@var:pm_temp }
		for_each_loop = {
			array = global.building_base_production@var:bu_token
			value = br_token
			index = ii

			set_temp_variable = { prod_temp = global.building_base_production_amount@var:bu_token^ii }
			multiply_temp_variable = { prod_temp = i_production_output_multiplier }
			multiply_temp_variable = { prod_temp = i_output_modifier }
			add_to_array = { c_production@var:cache_num = prod_temp }
		}

		# Input Requirements
		clear_array = c_input@var:cache_num
		clear_array = c_input_amount@var:cache_num
		for_each_loop = {
			array = pm_input@var:pm_temp
			value = br_token
			index = bu_i
			add_to_array = { c_input@var:cache_num = br_token }
			add_to_array = { c_input_amount@var:cache_num = pm_input_amount@var:pm_temp^bu_i }
		}
		for_each_loop = {
			array = pm_input@var:am_temp
			value = br_token
			index = bu_i

			if = {
				limit = { is_in_array = { c_input@var:cache_num = br_token } }

				for_each_loop = {
					array = c_input@var:cache_num
					value = br_token2
					index = bu_ii
					break = bu_brk

					if = {
						limit = { check_variable = { br_token2 = br_token } }
						add_to_variable = { c_input_amount@var:cache_num^bu_ii = pm_input_amount@var:am_temp^bu_i }
						clamp_variable = { var = c_input_amount@var:cache_num^bu_ii min = 0 max = 10000 }

						set_temp_variable = { bu_brk = 1 }
					}
				}
			}
			else = {
				add_to_array = { c_input@var:cache_num = br_token }
				add_to_array = { c_input_amount@var:cache_num = pm_input_amount@var:am_temp^bu_i }
			}
		}

		# Calculating Inputs
		for_each_loop = {
			array = c_input@var:cache_num
			value = br_token
			index = ii

			multiply_variable = { c_input_amount@var:cache_num^ii = i_input_modifier }
			divide_variable = { c_input_amount@var:cache_num^ii = global.base_price@var:br_token }
		}


		set_variable = { c_industrial_capacity_use@var:cache_num = i_industrial_capacity_use }
		set_variable = { c_industrial_capacity@var:cache_num = i_industrial_capacity }
		set_variable = { c_shipping_capacity@var:cache_num = i_shipping_capacity }
	##

	## Accounting
		# Output Finances
		set_temp_variable = { i_money_limit = 0 }
		set_temp_variable = { i_revenue_goods = 0 }
		for_each_loop = {
			array = global.building_base_production@var:bu_token
			value = br_token
			index = ii

			set_temp_variable = { money_temp = c_production@var:cache_num^ii }
			set_temp_variable = { money_limit_temp = money_temp }

			multiply_temp_variable = { money_temp = price@var:br_token }
			add_to_temp_variable = { i_revenue_goods = money_temp }
			
			multiply_temp_variable = { money_limit_temp = global.base_price@var:br_token }
			add_to_temp_variable = { i_money_limit = money_limit_temp }
		}

		# Input Finances
		set_temp_variable = { i_expenses_input = 0 }
		for_each_loop = {
			array = c_input@var:cache_num
			value = br_token
			index = ii
		
			set_temp_variable = { money_temp = c_input_amount@var:cache_num^ii }
			multiply_temp_variable = { money_temp = controller:price@var:br_token }
			add_to_temp_variable = { i_expenses_input = money_temp }
		}

		# Revenue/Expenses/Profit
		set_temp_variable = { i_revenue = i_shipping_capacity }
		add_to_temp_variable = { i_revenue = i_revenue_goods }
		
		set_variable = { c_revenue_goods@var:cache_num = i_revenue_goods }
		set_variable = { c_revenue@var:cache_num = i_revenue_goods }
		set_variable = { c_expenses_input@var:cache_num = i_expenses_input }
		set_variable = { c_expenses_payroll@var:cache_num = i_expenses_input }
		set_variable = { c_money_limit@var:cache_num = i_money_limit }


	##
}

	# state_temp: State to copy to
	# cache_num: Cache ID to copy from

	# am_temp

industry_update_cache = {
	# bu_token
	set_temp_variable = { pm_temp = production_method@var:bu_token }
	set_temp_variable = { am_temp = automation_method@var:bu_token }

	# Check if archetype is in cache
	if = {
		limit = {
			var:controller = {
				any_of = {
					array = industry_cache
					value = cache_num

					check_variable = { c_industry_token@var:cache_num = bu_token }
					check_variable = { c_production_method@var:cache_num = pm_temp }
					check_variable = { c_automation_method@var:cache_num = am_temp }
				}
			}
		}
		# Grab from Cache
	}
	else = {
		# Add to cache, then grab from cache
	}

	## UPDATE SIZE & JOBS
		# Update Industry Size
		if = { # RGOs
			limit = { check_variable = { global.building_rgo@var:bu_token > 0 } }
			set_temp_variable = { size_temp = base_resource@var:bu_token }
			set_variable = { size@var:bu_token = size_temp }
		}
		else_if = { # Infra
			limit ={ check_variable = { bu_token = token:infrastructure } }
			set_temp_variable = { size_temp = pops }
			divide_temp_variable = { size_temp = 100 }
			round_temp_variable = size_temp
			add_to_temp_variable = { size_temp = 1 }

			set_variable = { size@var:bu_token = size_temp }
		}
		else = { set_temp_variable = { size_temp = size@var:bu_token } }

		# Arable Land Usage
		set_variable = { arable_land_use@var:bu_token = size_temp }
		multiply_variable = { arable_land_use@var:bu_token = global.building_arable_land_use@var:bu_token }

		# Jobs Available
		set_temp_variable = { i_jobs_available = jobs@var:bu_token }
		set_temp_variable = { i_laborers_jobs_available = laborers_jobs@var:bu_token }
		set_temp_variable = { i_workers_jobs_available = workers_jobs@var:bu_token }
		set_temp_variable = { i_professionals_jobs_available = professionals_jobs@var:bu_token }
		set_temp_variable = { i_specialists_jobs_available = specialists_jobs@var:bu_token }

		set_temp_variable = { i_employment_target = employment_target@var:bu_token }
		divide_temp_variable = { i_employment_target = 100 }
		multiply_temp_variable = { i_jobs_available = i_employment_target }
		multiply_temp_variable = { i_laborers_jobs_available = i_employment_target }
		multiply_temp_variable = { i_workers_jobs_available = i_employment_target }
		multiply_temp_variable = { i_professionals_jobs_available = i_employment_target }
		multiply_temp_variable = { i_specialists_jobs_available = i_employment_target }

		set_variable = { jobs_available@var:bu_token = i_jobs_available }
		set_variable = { laborers_jobs_available@var:bu_token = i_laborers_jobs_available }
		set_variable = { workers_jobs_available@var:bu_token = i_workers_jobs_available }
		set_variable = { professionals_jobs_available@var:bu_token = i_professionals_jobs_available }
		set_variable = { specialists_jobs_available@var:bu_token = i_specialists_jobs_available }
		
		set_variable = { owner_jobs@var:bu_token = i_jobs_available }
		multiply_variable = { owner_jobs@var:bu_token = 0.02 }
	##

	## Employment
		set_temp_variable = { i_employment_percentage = 0 }
		set_temp_variable = { i_production_percentage = 0 }

		set_temp_variable = { i_laborers_employed = i_laborers_jobs_available }
		multiply_temp_variable = { i_laborers_employed = job_fulfillment@token:laborers }
		add_to_temp_variable = { i_employment_percentage = i_laborers_employed }
		multiply_temp_variable = { i_laborers_employed = 2 }
		add_to_temp_variable = { i_production_percentage = i_laborers_employed }

		set_temp_variable = { i_workers_employed = i_workers_jobs_available }
		multiply_temp_variable = { i_workers_employed = job_fulfillment@token:workers }
		add_to_temp_variable = { i_employment_percentage = i_workers_employed }
		multiply_temp_variable = { i_workers_employed = 5 }
		add_to_temp_variable = { i_production_percentage = i_workers_employed }
		
		set_temp_variable = { i_professionals_employed = i_professionals_jobs_available }
		multiply_temp_variable = { i_professionals_employed = job_fulfillment@token:professionals }
		add_to_temp_variable = { i_employment_percentage = i_professionals_employed }
		multiply_temp_variable = { i_professionals_employed = 10 }
		add_to_temp_variable = { i_production_percentage = i_professionals_employed }
		
		set_temp_variable = { i_specialists_employed = i_specialists_jobs_available }
		multiply_temp_variable = { i_specialists_employed = job_fulfillment@token:specialists }
		add_to_temp_variable = { i_employment_percentage = i_specialists_employed }
		multiply_temp_variable = { i_specialists_employed = 20 }
		add_to_temp_variable = { i_production_percentage = i_specialists_employed }

		divide_temp_variable = { i_production_percentage = payroll_multiplier@var:bu_token }
		divide_temp_variable = { i_production_percentage = size_temp }
		
		divide_temp_variable = { i_employment_percentage = jobs@var:bu_token }

		set_variable = { employment_percentage@var:bu_token = i_employment_percentage }
		set_variable = { production_percentage@var:bu_token = i_production_percentage }
	##

	## Production
		# Economy of scale modifier
		set_temp_variable = { eco_of_scale_modifier_temp = size_temp }
		subtract_from_temp_variable = { eco_of_scale_modifier_temp = 1 }
		clamp_temp_variable = { var = eco_of_scale_modifier_temp min = 0 max = global.buildings_economy_of_scale_max }
		multiply_temp_variable = { eco_of_scale_modifier_temp = global.buildings_economy_of_scale_factor }
		add_to_temp_variable = { eco_of_scale_modifier_temp = 1 }

		# Final Production Modifier
		set_temp_variable = { i_final_prod_modifier = size_temp }
		multiply_temp_variable = { i_final_prod_modifier = eco_of_scale_modifier_temp }
		multiply_temp_variable = { i_final_prod_modifier = i_production_percentage }
		divide_temp_variable = { i_final_prod_modifier = 100 }

		# Industrial Capacity Use
		multiply_variable = { industrial_capacity_use@var:bu_token = i_final_prod_modifier }

		# Industrial Capacity
		multiply_variable = { industrial_capacity@var:bu_token = i_final_prod_modifier }

		# Shipping Capacity
		multiply_variable = { shipping_capacity@var:bu_token = i_final_prod_modifier }

		# Outputs
		for_each_loop = {
			array = global.building_base_production@var:bu_token
			value = br_token
			index = ii

			multiply_variable = { production@var:bu_token^ii = i_final_prod_modifier }
		}

		# Inputs
		for_each_loop = {
			array = input@var:bu_token
			value = br_token
			index = ii

			multiply_variable = { input_amount@var:bu_token^ii = i_final_prod_modifier }
		}
	##

	## Accounting
		# Productivity
		set_temp_variable = { prod_revenue_temp = controller:c_industrial_capacity@var:cache_num }
		multiply_temp_variable = { prod_revenue_temp = price_industrial_capacity }
		set_temp_variable = { prod_expenses_temp = controller:c_industrial_capacity_use@var:cache_num }
		multiply_temp_variable = { prod_expenses_temp = price_industrial_capacity }
		add_to_temp_variable = { prod_revenue_temp = revenue@var:bu_token }
		add_to_temp_variable = { prod_expenses_temp = expenses_input@var:bu_token }
		
		subtract_from_temp_variable = { prod_revenue_temp = prod_expenses_temp }
		divide_temp_variable = { prod_revenue_temp = payroll_multiplier@var:bu_token }
		multiply_temp_variable = { prod_revenue_temp = eco_of_scale_modifier_temp }
		set_variable = { productivity@var:bu_token = prod_revenue_temp }

		# Revenue, Expenses, Profit
		set_temp_variable = { revenue_temp = industrial_capacity@var:bu_token }
		multiply_temp_variable = { revenue_temp = price_industrial_capacity }
		set_temp_variable = { expenses_temp = industrial_capacity_use@var:bu_token }
		multiply_temp_variable = { expenses_temp = price_industrial_capacity }

		multiply_variable = { revenue_goods@var:bu_token = i_final_prod_modifier }
		multiply_variable = { revenue@var:bu_token = i_final_prod_modifier }
		add_to_variable = { revenue@var:bu_token = revenue_temp }
		multiply_variable = { expenses_input@var:bu_token = i_final_prod_modifier }

		divide_temp_variable = { i_final_prod_modifier = eco_of_scale_modifier_temp }
		set_variable = { expenses_payroll@var:bu_token = payroll_multiplier@var:bu_token }
		multiply_variable = { expenses_payroll@var:bu_token = i_final_prod_modifier }
		multiply_variable = { expenses_payroll@var:bu_token = base_wage }

		set_variable = { expenses@var:bu_token = expenses_input@var:bu_token }
		add_to_variable = { expenses@var:bu_token = expenses_payroll@var:bu_token }
		add_to_variable = { expenses@var:bu_token = expenses_temp }

		set_temp_variable = { i_profit = revenue@var:bu_token }
		subtract_from_temp_variable = { i_profit = expenses@var:bu_token }

		# Bonuses
		if = {
			limit = {
				check_variable = { i_profit > 0 }
			}
			set_temp_variable = { bonus_total = i_profit }
