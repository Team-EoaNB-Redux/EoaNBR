	industry_update_size_jobs = {
		# Calculates Industry Size & Job Figures
		set_temp_variable = { production_method_temp = production_method@var:bu_token }
		set_temp_variable = { automation_method_temp = automation_method@var:bu_token }

		# Update Industry Size
		if = { # RGOs
			limit = {
				is_in_array = { global.rgo_token_array = bu_token }
			}
			set_temp_variable = { size_temp = base_resource@var:bu_token }
			set_variable = { size@var:bu_token = size_temp }
		}
		else_if = { # Infra
			limit ={ check_variable = { bu_token = token:infrastructure } }
			set_temp_variable = { size_temp = pops }
			divide_temp_variable = { size_temp = 100 }
			round_temp_variable = size_temp
			add_to_temp_variable = { size_temp = 1 }

			set_variable = { size@var:bu_token = size_temp }
		}
		else = {
			set_temp_variable = { size_temp = size@var:bu_token }
		}

		if = {
			limit = { NOT = { prev_size@var:bu_token = size_temp } }
			set_variable = { industry_dirty_var@var:bu_token = 1 }
		}
		else = {
			set_variable = { prev_size@var:bu_token = size_temp }
		}
		
		# Arable Land Usage
		if = {
			limit = {
				check_variable = { global.building_arable_land_use@var:bu_token > 0 }
			}
			set_variable = { arable_land_use@var:bu_token = size_temp }
			multiply_variable = { arable_land_use@var:bu_token = global.building_arable_land_use@var:bu_token }
		}
		else = {
			clear_variable = arable_land_use@var:bu_token
		}
		
		# Jobs
		set_temp_variable = { base_jobs_temp = global.building_base_employees@var:bu_token }
		multiply_temp_variable = { base_jobs_temp = size_temp }

		set_temp_variable = { i_laborers_jobs = controller:pm_workforce_laborers@var:production_method_temp }
		add_to_temp_variable = { i_laborers_jobs = controller:pm_workforce_laborers@var:automation_method_temp }
		clamp_temp_variable = { var = i_laborers_jobs min = 0 max = i_laborers_jobs }

		set_temp_variable = { i_workers_jobs = controller:pm_workforce_workers@var:production_method_temp }
		add_to_temp_variable = { i_workers_jobs = controller:pm_workforce_workers@var:automation_method_temp }
		clamp_temp_variable = { var = i_workers_jobs min = 0 max = i_workers_jobs }

		set_temp_variable = { i_professionals_jobs = controller:pm_workforce_professionals@var:production_method_temp }
		add_to_temp_variable = { i_professionals_jobs = controller:pm_workforce_professionals@var:automation_method_temp }
		clamp_temp_variable = { var = i_professionals_jobs min = 0 max = i_professionals_jobs }

		set_temp_variable = { i_specialists_jobs = controller:pm_workforce_specialists@var:production_method_temp }
		add_to_temp_variable = { i_specialists_jobs = controller:pm_workforce_specialists@var:automation_method_temp }
		clamp_temp_variable = { var = i_specialists_jobs min = 0 max = i_specialists_jobs }

		# Payroll Multiplier
		set_temp_variable = { i_payroll_multiplier = 0 }
		set_temp_variable = { payroll_multiplier_temp = i_laborers_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 2 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_workers_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 5 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_professionals_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 10 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		set_temp_variable = { payroll_multiplier_temp = i_specialists_jobs }
		multiply_temp_variable = { payroll_multiplier_temp = 20 }
		add_to_temp_variable = { i_payroll_multiplier = payroll_multiplier_temp }
		multiply_temp_variable = { i_payroll_multiplier = global.building_base_employees@var:bu_token }

		set_variable = { payroll_multiplier@var:bu_token = i_payroll_multiplier }

		multiply_temp_variable = { i_laborers_jobs = base_jobs_temp }
		multiply_temp_variable = { i_workers_jobs = base_jobs_temp }
		multiply_temp_variable = { i_professionals_jobs = base_jobs_temp }
		multiply_temp_variable = { i_specialists_jobs = base_jobs_temp }
		
		set_temp_variable = { i_jobs = 0 }
		add_to_temp_variable = { i_jobs = i_laborers_jobs }
		add_to_temp_variable = { i_jobs = i_workers_jobs }
		add_to_temp_variable = { i_jobs = i_professionals_jobs }
		add_to_temp_variable = { i_jobs = i_specialists_jobs }

		# Jobs Available
		set_temp_variable = { i_laborers_jobs_available = i_laborers_jobs }
		set_temp_variable = { i_workers_jobs_available = i_workers_jobs }
		set_temp_variable = { i_professionals_jobs_available = i_professionals_jobs }
		set_temp_variable = { i_specialists_jobs_available = i_specialists_jobs }

		set_temp_variable = { i_owner_jobs = base_jobs_temp }
		multiply_temp_variable = { i_owner_jobs = 0.02 }

		set_temp_variable = { i_employment_target = employment_target@var:bu_token }
		if = {
			limit = { check_variable = { i_jobs > 100 } }
			divide_temp_variable = { i_employment_target = 100 }
			multiply_temp_variable = { i_laborers_jobs_available = i_employment_target }
			multiply_temp_variable = { i_workers_jobs_available = i_employment_target }
			multiply_temp_variable = { i_professionals_jobs_available = i_employment_target }
			multiply_temp_variable = { i_specialists_jobs_available = i_employment_target }
			multiply_temp_variable = { i_owner_jobs = i_employment_target }
		}
		else = {
			multiply_temp_variable = { i_laborers_jobs_available = i_employment_target }
			multiply_temp_variable = { i_workers_jobs_available = i_employment_target }
			multiply_temp_variable = { i_professionals_jobs_available = i_employment_target }
			multiply_temp_variable = { i_specialists_jobs_available = i_employment_target }
			multiply_temp_variable = { i_owner_jobs = i_employment_target }

			divide_temp_variable = { i_laborers_jobs_available = 100 }
			divide_temp_variable = { i_workers_jobs_available = 100 }
			divide_temp_variable = { i_professionals_jobs_available = 100 }
			divide_temp_variable = { i_specialists_jobs_available = 100 }
			divide_temp_variable = { i_owner_jobs = 100 }
		}
		
		set_temp_variable = { i_jobs_available = 0 }
		add_to_temp_variable = { i_jobs_available = i_laborers_jobs_available }
		add_to_temp_variable = { i_jobs_available = i_workers_jobs_available }
		add_to_temp_variable = { i_jobs_available = i_professionals_jobs_available }
		add_to_temp_variable = { i_jobs_available = i_specialists_jobs_available }

		# Decaching
		set_variable = { owner_jobs@var:bu_token = i_owner_jobs }

		set_variable = { jobs@var:bu_token = i_jobs }
		set_variable = { laborers_jobs@var:bu_token = i_laborers_jobs }
		set_variable = { workers_jobs@var:bu_token = i_workers_jobs }
		set_variable = { professionals_jobs@var:bu_token = i_professionals_jobs }
		set_variable = { specialists_jobs@var:bu_token = i_specialists_jobs }
		
		set_variable = { jobs_available@var:bu_token = i_jobs_available }
		set_variable = { laborers_jobs_available@var:bu_token = i_laborers_jobs_available }
		set_variable = { workers_jobs_available@var:bu_token = i_workers_jobs_available }
		set_variable = { professionals_jobs_available@var:bu_token = i_professionals_jobs_available }
		set_variable = { specialists_jobs_available@var:bu_token = i_specialists_jobs_available }

	}
	industry_update_employment = {
		# Employment
		set_temp_variable = { i_employment_target = employment_target@var:bu_token }
		set_temp_variable = { i_employed = 0 }

		set_temp_variable = { i_pop_jobs = laborers_jobs@var:bu_token }
		set_temp_variable = { i_pop_token = token:laborers }
		industry_update_employment_pop_type = yes
		set_variable = { laborers_employed@var:bu_token = i_pop_employed }
		add_to_temp_variable = { i_employed = i_pop_employed }

		set_temp_variable = { i_pop_jobs = workers_jobs@var:bu_token }
		set_temp_variable = { i_pop_token = token:workers }
		industry_update_employment_pop_type = yes
		set_variable = { workers_employed@var:bu_token = i_pop_employed }
		add_to_temp_variable = { i_employed = i_pop_employed }

		set_temp_variable = { i_pop_jobs = professionals_jobs@var:bu_token }
		set_temp_variable = { i_pop_token = token:professionals }
		industry_update_employment_pop_type = yes
		set_variable = { professionals_employed@var:bu_token = i_pop_employed }
		add_to_temp_variable = { i_employed = i_pop_employed }

		set_temp_variable = { i_pop_jobs = specialists_jobs@var:bu_token }
		set_temp_variable = { i_pop_token = token:specialists }
		industry_update_employment_pop_type = yes
		set_variable = { specialists_employed@var:bu_token = i_pop_employed }
		add_to_temp_variable = { i_employed = i_pop_employed }
		
		set_variable = { employed@var:bu_token = i_employed }

		#Employment/Production Percentage
		set_temp_variable = { i_employment_percentage = i_employed }
		if = {
			limit = {
				check_variable = { i_employed < 1000 }
			}
			multiply_temp_variable = { i_employment_percentage = 100 }
			divide_temp_variable = { i_employment_percentage = jobs@var:bu_token }
		}
		else = {
			divide_temp_variable = { i_employment_percentage = jobs@var:bu_token }
			multiply_temp_variable = { i_employment_percentage = 100 }
		}
		set_variable = { employment_percentage@var:bu_token = i_employment_percentage }

		set_temp_variable = { total_prod_temp = 0 }
		set_temp_variable = { production_temp = laborers_employed@var:bu_token }
		multiply_temp_variable = { production_temp = 2 }
		add_to_temp_variable = { total_prod_temp = production_temp }
		set_temp_variable = { production_temp = workers_employed@var:bu_token }
		multiply_temp_variable = { production_temp = 5 }
		add_to_temp_variable = { total_prod_temp = production_temp }
		set_temp_variable = { production_temp = professionals_employed@var:bu_token }
		multiply_temp_variable = { production_temp = 10 }
		add_to_temp_variable = { total_prod_temp = production_temp }
		set_temp_variable = { production_temp = specialists_employed@var:bu_token }
		multiply_temp_variable = { production_temp = 20 }
		add_to_temp_variable = { total_prod_temp = production_temp }
		if = {
			limit = {
				check_variable = { total_prod_temp > 10000 }
			}
			divide_temp_variable = { total_prod_temp = payroll_multiplier@var:bu_token }
			divide_temp_variable = { total_prod_temp = size@var:bu_token }
			multiply_temp_variable = { total_prod_temp = 100 }
		}
		else = {
			multiply_temp_variable = { total_prod_temp = 100 }
			divide_temp_variable = { total_prod_temp = payroll_multiplier@var:bu_token }
			divide_temp_variable = { total_prod_temp = size@var:bu_token }
		}
		if = {
			limit = { NOT = { production_percentage@var:bu_token = total_prod_temp } }
			set_variable = { industry_dirty_var@var:bu_token = 1 }
		}
		
		set_variable = { production_percentage@var:bu_token = total_prod_temp }
	}
	industry_update_employment_pop_type = {

				divide_temp_variable = { i_pop_employed = 100 }
				multiply_temp_variable = { i_pop_employed = job_fulfillment@var:i_pop_token }
			}
			else = {
				set_temp_variable = { i_pop_employed = i_pop_jobs }
				multiply_temp_variable = { i_pop_employed = i_employment_target }
				divide_temp_variable = { i_pop_employed = 100 }
				multiply_temp_variable = { i_pop_employed = job_fulfillment@var:i_pop_token }
				divide_temp_variable = { i_pop_employed = 100 }
			}
		}
		else = {
			set_temp_variable = { i_pop_employed = 0 }
		}
	}

	industry_production_update = {
		# Needs bu_token and i_size

		set_temp_variable = { i_base_output_value = global.building_base_output_value@var:bu_token }

		# Core production methods
		set_temp_variable = { production_method_temp = production_method@var:bu_token }
		set_temp_variable = { automation_method_temp = automation_method@var:bu_token }

		# Output/input/throughput modifiers
		set_temp_variable = { output_modifier_temp = controller:industry_output_factor@var:bu_token }
		add_to_temp_variable = { output_modifier_temp = 1 }

		set_temp_variable = { throughput_modifier_temp = controller:industry_throughput_factor@var:bu_token }
		add_to_temp_variable = { throughput_modifier_temp = 1 }

		set_temp_variable = { input_modifier_temp = controller:industry_input_factor@var:bu_token }
		add_to_temp_variable = { input_modifier_temp = 1 }

		# Economy of scale modifier
		set_temp_variable = { eco_of_scale_modifier_temp = i_size }
		subtract_from_temp_variable = { eco_of_scale_modifier_temp = 1 }
		clamp_temp_variable = { var = eco_of_scale_modifier_temp min = 0 max = global.buildings_economy_of_scale_max }
		multiply_temp_variable = { eco_of_scale_modifier_temp = global.buildings_economy_of_scale_factor }
		add_to_temp_variable = { eco_of_scale_modifier_temp = 1 }

		# Total Input Modifier
		set_temp_variable = { i_input_modifier = i_base_output_value }
		multiply_temp_variable = { i_input_modifier = input_modifier_temp }
		multiply_temp_variable = { i_input_modifier = throughput_modifier_temp }
		multiply_temp_variable = { i_input_modifier = eco_of_scale_modifier_temp }

		# Total Ouput Modifier
		set_temp_variable = { i_output_modifier = output_modifier_temp }
		multiply_temp_variable = { i_output_modifier = throughput_modifier_temp }
		multiply_temp_variable = { i_output_modifier = eco_of_scale_modifier_temp }

		# Final Production Modifier
		set_temp_variable = { i_final_prod_modifier = i_size }
		multiply_temp_variable = { i_final_prod_modifier = production_percentage@var:bu_token }
		divide_temp_variable = { i_final_prod_modifier = 100 }

		# Capacity Reqs & Production
			# Industrial Capacity Use
			set_temp_variable = { i_industrial_capacity_use = controller:pm_industrial_capacity_use@var:production_method_temp }
			add_to_temp_variable = { i_industrial_capacity_use = controller:pm_industrial_capacity_use@var:automation_method_temp }
			if = {
				limit = { check_variable = { i_industrial_capacity_use > 0 } }
				multiply_temp_variable = { i_industrial_capacity_use = i_input_modifier }
				divide_temp_variable = { i_industrial_capacity_use = global.base_price_industrial_capacity }

				set_variable = { industrial_capacity_use_max@var:bu_token = i_industrial_capacity_use }

				multiply_temp_variable = { i_industrial_capacity_use = i_final_prod_modifier }
				set_variable = { industrial_capacity_use@var:bu_token = i_industrial_capacity_use }
			}
			else = {
				set_variable = { industrial_capacity_use@var:bu_token = 0 }
				set_variable = { industrial_capacity_use_max@var:bu_token = 0 }
			}
			
			# Industrial Capacity
			set_temp_variable = { i_industrial_capacity = controller:pm_industrial_capacity@var:production_method_temp }
			add_to_temp_variable = { i_industrial_capacity = controller:pm_industrial_capacity@var:automation_method_temp }
			if = {
				limit = { check_variable = { i_industrial_capacity > 0 } }
				multiply_temp_variable = { i_industrial_capacity = i_base_output_value }
				multiply_temp_variable = { i_industrial_capacity = i_output_modifier }
				divide_temp_variable = { i_industrial_capacity = global.base_price_industrial_capacity }

				set_variable = { industrial_capacity_max@var:bu_token = i_industrial_capacity }

				multiply_temp_variable = { i_industrial_capacity = i_final_prod_modifier }
				set_variable = { industrial_capacity@var:bu_token = i_industrial_capacity }
			}
			else = {
				set_variable = { industrial_capacity@var:bu_token = 0 }
				set_variable = { industrial_capacity_max@var:bu_token = 0 }
			}
			
			# Shipping Capacity
			set_temp_variable = { i_shipping_capacity = controller:pm_shipping_capacity@var:production_method_temp }
			add_to_temp_variable = { i_shipping_capacity = controller:pm_shipping_capacity@var:automation_method_temp }
			if = {
				limit = { check_variable = { i_shipping_capacity > 0 } }
				multiply_temp_variable = { i_shipping_capacity = i_base_output_value }
				multiply_temp_variable = { i_shipping_capacity = i_output_modifier }
				divide_temp_variable = { i_shipping_capacity = global.base_price_shipping_capacity }

				set_variable = { shipping_capacity_max@var:bu_token = i_shipping_capacity }

				multiply_temp_variable = { i_shipping_capacity = i_final_prod_modifier }
				set_variable = { shipping_capacity@var:bu_token = i_shipping_capacity }
			}
			else = {
				set_variable = { shipping_capacity@var:bu_token = 0 }
				set_variable = { shipping_capacity_max@var:bu_token = 0 }
			}
		#
		
		# Output Production
			clear_array = production@var:bu_token
			clear_array = production_max@var:bu_token
			clear_array = revenue_goods@var:bu_token
			set_temp_variable = { i_production_output_multiplier = controller:pm_output_multiplier@var:production_method_temp }
			set_variable = { production_output_multiplier@var:bu_token = i_production_output_multiplier }
			for_each_loop = {
				array = global.building_base_production@var:bu_token
				value = br_token
				index = ii

				# Max Production
				set_temp_variable = { prod_temp = global.building_base_production_amount@var:bu_token^ii }
				multiply_temp_variable = { prod_temp = i_production_output_multiplier }
				multiply_temp_variable = { prod_temp = i_output_modifier }
				add_to_array = { production_max@var:bu_token = prod_temp }

				# Actual Production
				multiply_temp_variable = { prod_temp = i_final_prod_modifier }
				add_to_array = { production@var:bu_token = prod_temp }

				add_to_array = { revenue_goods@var:bu_token = 0 }
			}
		#

		# Input Requirements
			# Compile Inputs from the PM and AM Together
			clear_array = input@var:bu_token
			clear_array = input_base_amount@var:bu_token
			clear_array = input_max@var:bu_token
			clear_array = input_amount@var:bu_token
			clear_array = input_cost@var:bu_token
			for_each_loop = {
				array = controller:pm_input@var:production_method_temp
				value = br_token
				index = bu_i
				add_to_array = { input@var:bu_token = br_token }
				add_to_array = { input_base_amount@var:bu_token = controller:pm_input_amount@var:production_method_temp^bu_i }
				add_to_array = { input_max@var:bu_token = 0 }
				add_to_array = { input_amount@var:bu_token = 0 }
				add_to_array = { input_cost@var:bu_token = 0 }
			}
			for_each_loop = {
				array = controller:pm_input@var:automation_method_temp
				value = br_token
				index = bu_i

				if = {
					limit = { is_in_array = { input@var:bu_token = br_token } }

					for_each_loop = {
						array = input@var:bu_token
						value = br_token2
						index = bu_ii
						break = bu_brk

						if = {
							limit = { check_variable = { br_token2 = br_token } }
							add_to_variable = { input_base_amount@var:bu_token^bu_ii = controller:pm_input_amount@var:automation_method_temp^bu_i }
							clamp_variable = { var = input_base_amount@var:bu_token^bu_ii min = 0 }

							set_temp_variable = { bu_brk = 1 }
						}
					}
				}
				else = {
					add_to_array = { input@var:bu_token = br_token }
					add_to_array = { input_base_amount@var:bu_token = controller:pm_input_amount@var:automation_method_temp^bu_i }
					add_to_array = { input_max@var:bu_token = 0 }

			set_temp_variable = { prod_temp = global.building_base_production_amount@token:subsistence_farm^ii }	# Base resource production per base employee count
			multiply_temp_variable = { prod_temp = controller:pm_output_multiplier@token:pm_subsistence_farming }   # Multiply by PM Output Multiplier
			multiply_temp_variable = { prod_temp = peasants_employed@token:subsistence_farm }					   # Multiply by Employee count

			#   Peasants excess of productive capacity are 70% as productive
			set_temp_variable = { marginal_temp = peasants_marginal_land }
			multiply_temp_variable = { marginal_temp = 0.7 }
			add_to_temp_variable = { prod_temp = marginal_temp }

			divide_temp_variable = { prod_temp = global.building_base_employees@token:subsistence_farm }			# Divide by Base Employee Count (to get per K)
			divide_temp_variable = { prod_temp = controller:pm_workforce_peasants@token:pm_subsistence_farming }	# Divide by PM Workforce Multiplier
			multiply_temp_variable = { prod_temp = output_modifier_temp }

			# Goods Revenue
			set_temp_variable = { money_temp = prod_temp }
			multiply_temp_variable = { money_temp = controller:price@var:r_token }

			add_to_array = { production@token:subsistence_farm = prod_temp }
			add_to_array = { revenue_goods@token:subsistence_farm = money_temp }
			add_to_variable = { revenue_goods@token:subsistence_farm = money_temp }

			# For productivity
			set_temp_variable = { productivity_temp = global.building_base_production_amount@token:subsistence_farm^ii }
			multiply_temp_variable = { productivity_temp = controller:pm_output_multiplier@token:pm_subsistence_farming }
			divide_temp_variable = { productivity_temp = global.building_base_employees@token:subsistence_farm }
			divide_temp_variable = { productivity_temp = controller:pm_workforce_peasants@token:pm_subsistence_farming }
			multiply_temp_variable = { productivity_temp = controller:price@var:r_token }
			add_to_temp_variable = { revenue_temp = productivity_temp }
		}

		# Revenue
		set_variable = { revenue@token:subsistence_farm = revenue_goods@token:subsistence_farm }

		# Expenses
		set_variable = { expenses_payroll@token:subsistence_farm = revenue@token:subsistence_farm }
		multiply_variable = { expenses_payroll@token:subsistence_farm = 0.8 }				 # Aristocrats take 20%
		set_variable = { expenses@token:subsistence_farm = expenses_payroll@token:subsistence_farm }

		set_variable = { average_wage@token:subsistence_farm = expenses_payroll@token:subsistence_farm }
		if = { limit = { check_variable = { pops_workforce@token:peasants > 0 } } divide_variable = { average_wage@token:subsistence_farm = pops_workforce@token:peasants } }

		# Dividends
		set_variable = { dividends@token:subsistence_farm = revenue@token:subsistence_farm }
		subtract_from_variable = { dividends@token:subsistence_farm = expenses@token:subsistence_farm }

		# Productivity
		if = { 
			limit = { check_variable = { pops_workforce@token:peasants > 0 } check_variable = { revenue_goods@token:peasants > 0 } } 
			set_variable = { productivity@token:subsistence_farm = revenue_goods@token:subsistence_farm }
			divide_variable = { productivity@token:subsistence_farm = pops_workforce@token:peasants }
		}
		else = {
			set_variable = { productivity@token:subsistence_farm = revenue_temp }
		}


		set_variable = { profit@token:subsistence_farm = 0 }

	industry_productivity_sort = {
		# Puts industries into an array sorted by the base productivity

		clear_array = industries_productivity_sort
		for_each_loop = {
			array = industries
			value = om_building

			set_temp_variable = { effective_wage_om = productivity@var:om_building }

			for_each_loop = {
				array = industries_productivity_sort

					}
					add_to_array = {
						array = industries_productivity_sort
						value = om_building
						index = ii
					}
					set_temp_variable = { bb = 1 }
				}
			}
			if = {
				limit = {
					check_variable = { bb = 0 }
				}
				add_to_array = { industries_productivity_sort = om_building }
			}
		}

	base_wage_calculation = {
		set_temp_variable = { subs_farm_productivity = 1 }
		if = {
			limit = {
				check_variable = { productivity@token:subsistence_farm > subs_farm_productivity }
			}
			set_temp_variable = { subs_farm_productivity = productivity@token:subsistence_farm }
		}

		# Is the sum of payroll_multiplier * size * productivity across every industry.
		set_temp_variable = { base_wage_counting = 0 }

		# Sourcing all job types
		set_temp_variable = { total_jobs_weight = 0.1 }
		add_to_temp_variable = { total_jobs_weight = pops_workforce@token:slaves }
		add_to_temp_variable = { total_jobs_weight = pops_workforce@token:peasants }
		set_temp_variable = { weight_temp = pops_workforce@token:laborers }
		multiply_temp_variable = { weight_temp = 2 }
		add_to_temp_variable = { total_jobs_weight = weight_temp }
		set_temp_variable = { weight_temp = pops_workforce@token:workers }
		multiply_temp_variable = { weight_temp = 5 }
		add_to_temp_variable = { total_jobs_weight = weight_temp }
		set_temp_variable = { weight_temp = pops_workforce@token:professionals }
		multiply_temp_variable = { weight_temp = 10 }
		add_to_temp_variable = { total_jobs_weight = weight_temp }
		set_temp_variable = { weight_temp = pops_workforce@token:specialists }
		multiply_temp_variable = { weight_temp = 20 }
		add_to_temp_variable = { total_jobs_weight = weight_temp }
		# log = "Jobs Weight: [?total_jobs_weight]"

		set_temp_variable = { remaining_jobs = total_jobs_weight }

		for_each_loop = {
			array = industries_productivity_sort
			value = b_temp

			# Base Wage Addition
			if = {
				limit = {
					check_variable = { remaining_jobs > 0 }
					NOT = { check_variable = { b_temp = token:subsistence_farm } }
				}
				# If there are still jobs to account for, then weight the jobs from this industry and add them up
				set_temp_variable = { jobs_wage_to_count = payroll_multiplier@var:b_temp }
				multiply_temp_variable = { jobs_wage_to_count = size@var:b_temp }
				multiply_temp_variable = { jobs_wage_to_count = employment_target@var:b_temp }
				divide_temp_variable = { jobs_wage_to_count = 100 }
				
				if = { limit = { check_variable = { jobs_wage_to_count > remaining_jobs } } set_temp_variable = { jobs_wage_to_count = remaining_jobs } }
				subtract_from_temp_variable = { remaining_jobs = jobs_wage_to_count }
				#set_temp_variable = { jobs_counted = jobs_wage_to_count }

				# Add to base_wage_counting for calculating the base wage using the productivity of the most productive industries
				if = {
					limit = {
						check_variable = { productivity@var:b_temp > subs_farm_productivity }
					}
					multiply_temp_variable = { jobs_wage_to_count = productivity@var:b_temp }
				}
				else = {
					multiply_temp_variable = { jobs_wage_to_count = subs_farm_productivity }
				}
				if = { limit = { check_variable = { b_temp = token:valuables } } divide_temp_variable = { jobs_wage_to_count = 3 } }
				add_to_temp_variable = { base_wage_counting = jobs_wage_to_count }
				#log = "[?b_temp.GetTokenLocalizedKey]: [?jobs_counted] | [?jobs_wage_to_count]"
			}
		}

		# log = "Remaining Jobs: [?remaining_jobs]"
		# These remaining jobs get assigned the productivity of subsistence farms
		if = {
			limit = {
				check_variable = { remaining_jobs > 0 }
			}
			# Multiply by subs farm prod and add to base wage counting
			multiply_temp_variable = { remaining_jobs = subs_farm_productivity } # Productivity of 1 or more
			add_to_temp_variable = { base_wage_counting = remaining_jobs }
		}

		# set_temp_variable = { base_wage_temp = base_wage_counting }
		# divide_temp_variable = { base_wage_temp = total_jobs_weight }
		# multiply_temp_variable = { base_wage_temp = wage_factor }
		# log = "Base Wage: [?base_wage_temp]"

		set_variable = { base_wage = base_wage_counting } # payroll_multiplier * size * productivity * employment_target
		divide_variable = { base_wage = total_jobs_weight } # payroll_multiplier * size (pops)
		if = {
			limit = {
				check_variable = { base_wage < subs_farm_productivity }
			}
			set_variable = { base_wage = subs_farm_productivity }
		}

		multiply_variable = { base_wage = wage_factor }

		set_variable = { laborers_wage = base_wage }
		multiply_variable = { laborers_wage = global.wage_multiplier@token:laborers }

		set_variable = { workers_wage = base_wage }
		multiply_variable = { workers_wage = global.wage_multiplier@token:workers }

		set_variable = { professionals_wage = base_wage }
		multiply_variable = { professionals_wage = global.wage_multiplier@token:professionals }

		set_variable = { specialists_wage = base_wage }
		multiply_variable = { specialists_wage = global.wage_multiplier@token:specialists }
	}

				set_variable = { construction_financier@var:industry_token = financer_temp }
			}
			else = {
				set_temp_variable = { financer_temp = construction_financier@var:industry_token }
			}

			add_to_variable = { construction@var:industry_token = x_temp }

			#Add building to array if it doesn't exist yet
			if = {
				limit = {
					NOT = { is_in_array = { industries = industry_token } }
				}
				add_to_array = { industries = industry_token }
				set_variable = { production_method@var:industry_token = production_method_temp }
				set_variable = { automation_method@var:industry_token = automation_method_temp }
			}

			# Cost
			set_temp_variable = { iuc_industry = industry_token }
			industry_update_construction_retooling = yes

			set_temp_variable = { pm_temp = production_method@var:industry_token }
			set_temp_variable = { am_temp = automation_method@var:industry_token }
			if = {
				limit = {
					check_variable = { no_cost_temp = 0 }
					get_construction_cost = yes
				}
				set_temp_variable = { initial_cost_temp = total_construction_cost_temp }
				multiply_temp_variable = { initial_cost_temp = global.start_construction_cost_multiplier }
				if = {
					limit = {
						check_variable = { financer_temp < 0 } #Corporation
					}
				}
				else_if = {
					limit = {
						check_variable = { financer_temp > 0 } #Government
					}
					divide_temp_variable = { initial_cost_temp = financer_temp:modifier@government_investment_efficiency_factor }
					subtract_from_variable = { financer_temp:treasury = initial_cost_temp }
					var:financer_temp = {
						spending_update = yes
					}
				}
				else = { #Local
					subtract_from_variable = { controller:investment_fund = initial_cost_temp }
				}
				add_to_variable = { construction_investment@var:industry_token = initial_cost_temp }
			}
			refresh_construction_list = yes
			reload_industries_array = yes
			add_to_variable = { update_state_econ_window = 1 }
		}
	}
