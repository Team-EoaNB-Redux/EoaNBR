
# Industry Update
	industries_update = {
		# The upper class owns the generic share of every industry here: its size less what companies hold (Q11).
		# A sector pays dividends nationally, by size (sectors_update): its dividends@ here are this size's share, from last
		# month's accounts. Subsistence rent and the government buildings pay their own dividends@
		set_variable = { pops_dividends@token:upper_class = 0 }

		for_each_loop = {
			array = industries
			value = bu_token
			if = {
				limit = {
					check_variable = { size@var:bu_token > 0 }
				}

				# The incorporated share is collected by companies in companies_update. owner_sizes is empty for unincorporated
				# industries, so the generic owner holds the whole size
				set_temp_variable = { incorporated_size_temp = 0 }
				for_each_loop = {
					array = owner_sizes@var:bu_token
					value = owner_size_temp
					add_to_temp_variable = { incorporated_size_temp = owner_size_temp }
				}
				set_variable = { owner_sizes_local@var:bu_token = { value = size@var:bu_token subtract = incorporated_size_temp max = 0 } }

				set_temp_variable = { local_dividends_temp = dividends@var:bu_token }
				if = {
					limit = { is_sector = yes }
					set_variable = { dividends@var:bu_token = { value = controller:sector_dividend_per_size@var:bu_token multiply = size@var:bu_token } }
					set_temp_variable = { local_dividends_temp = { value = controller:sector_dividend_per_size@var:bu_token multiply = owner_sizes_local@var:bu_token } }
				}
				add_to_variable = { pops_dividends@token:upper_class = local_dividends_temp }
			}
		}

		for_each_loop = {
			array = company_hqs
			value = cid
			# Add company dividends
			add_to_variable = { pops_dividends@token:upper_class = global.company_dividends@var:cid }
		}

		special_buildings_update = yes

		for_each_loop = {
			array = buildings
			value = bu_token
			index = ind
			break = ind_break

			industry_update = yes

		}
		for_each_loop = {
			array = rgos
			value = bu_token
			index = ind
			break = ind_break

			industry_update = yes
		}
		for_each_loop = {
			array = rgos
			value = r_token
			index = ind
			break = ind_break

			update_prospecting = yes
		}
	}

	# National sectors (Q18, Q85): one per industry token on the country's sectors list, which cache_register_industry fills.
	# Subsistence farms and the government buildings aren't sectors. Each sector keeps its accounts nationally, and its
	# state-industries are where the work happens
	sectors_monthly = {
		# Country scope. Runs once a tick, before the state loop, from last month's accounts (sectors_update):
		# - Retained earnings become sector capital (Q21), which M09 spends. A loss drawn from capital is negative retained
		#   earnings, and never takes capital below 0 (Q22).
		# - Target utilisation (Q20), sector_utilisation@, in %: a damped step on the national margin. It rises while the margin
		#   is above global.sector_utilisation_rise_margin and falls while it's negative, by 1 point plus
		#   global.sector_utilisation_gain points per unit of margin beyond those, at most global.sector_utilisation_step_max
		#   a month, between global.sector_utilisation_min and 100. A sector with no industries holds its target
		for_each_loop = {
			array = sectors
			value = s_token
			add_to_variable = { sector_capital@var:s_token = sector_retained@var:s_token }

			if = {
				limit = { check_variable = { sector_size@var:s_token > 0 } }
				set_variable = {
					sector_utilisation@var:s_token = {
						value = sector_utilisation@var:s_token
						if = { limit = { value = sector_margin@var:s_token greater_than = global.sector_utilisation_rise_margin }
							add = {
								value = sector_margin@var:s_token
								subtract = global.sector_utilisation_rise_margin
								multiply = global.sector_utilisation_gain
								add = 1
								min = global.sector_utilisation_step_max
							}
						}
						if = { limit = { value = sector_margin@var:s_token less_than = 0 }
							subtract = {
								value = sector_margin@var:s_token
								multiply = global.sector_utilisation_gain
								multiply = -1
								add = 1
								min = global.sector_utilisation_step_max
							}
						}
						clamp = { min = global.sector_utilisation_min max = 100 }
					}
				}
			}
		}
	}
	industries_monthly = {
		# State scope. Runs once a tick, before industries_update, from last month's figures:
		# - Each sector state-industry's trailing operating margin, margin@, moves global.industry_margin_weight of the way
		#   toward last month's (profit over revenue). It sets the local shift in utilisation (industry_utilisation_calc) and
		#   capacity decay. It's 0 until the industry first earns revenue, and then starts at that month's margin.
		# - Capacity decay (Q23, Q87): a building whose trailing margin is below global.industry_decay_margin loses
		#   global.industry_decay_rate of its generic owner's size a month (companies decide for their own), and the generic
		#   share closes once the building is below global.industry_decay_floor. Only while investment is on
		#   (global.eco_investment_enabled), since nothing rebuilds capacity before M09. RGOs follow their deposits
		for_each_loop = {
			array = industries
			value = bu_token
			if = {
				limit = {
					is_sector = yes
					check_variable = { revenue@var:bu_token > 0 }
				}
				set_temp_variable = {
					margin_temp = {
						value = revenue@var:bu_token
						subtract = expenses_payroll@var:bu_token
						subtract = expenses_input@var:bu_token
						subtract = expenses_maintenance@var:bu_token
						divide = revenue@var:bu_token
					}
				}
				if = {
					limit = { check_variable = { margin@var:bu_token = 0 } }
					set_variable = { margin@var:bu_token = margin_temp }
				}
				else = {
					set_variable = {
						margin@var:bu_token = {
							value = margin_temp
							subtract = margin@var:bu_token
							multiply = global.industry_margin_weight
							add = margin@var:bu_token
						}
					}
				}
			}
		}

		if = {
			limit = { check_variable = { global.eco_investment_enabled = 1 } }
			for_each_loop = { # Not the state's buildings array, which closing a building changes
				array = global.building_token_array
				value = decay_token
				if = {
					limit = {
						check_variable = { owner_sizes_local@var:decay_token > 0 }
						check_variable = { construction@var:decay_token = 0 }
						check_variable = { margin@var:decay_token < global.industry_decay_margin }
						NOT = { check_variable = { margin@var:decay_token = 0 } }
					}
					set_temp_variable = { industry_token = decay_token }
					set_temp_variable = { x_temp = { value = owner_sizes_local@var:decay_token multiply = global.industry_decay_rate multiply = -1 } }
					if = {
						limit = { check_variable = { size@var:decay_token < global.industry_decay_floor } }
						set_temp_variable = { x_temp = { value = owner_sizes_local@var:decay_token multiply = -1 } }
					}
					set_temp_variable = { financer_temp = 0 }
					set_temp_variable = { production_method_temp = production_method@var:decay_token }
					set_temp_variable = { automation_method_temp = automation_method@var:decay_token }
					add_x_industry_internal = yes
				}
			}
		}
	}
	industry_utilisation_calc = {
		# State scope. Input: bu_token. Output: i_employment_target, the share of its jobs it offers, in %
		# A sector's state-industries work at the national target utilisation (sector_utilisation@, sectors_monthly), shifted by
		# global.local_utilisation_gain points per unit by which their trailing margin beats the national margin, at most
		# global.local_utilisation_shift points either way (Q20). The government buildings always work at 100%
		if = {
			limit = { is_sector = yes }
			set_temp_variable = { i_employment_target = controller:sector_utilisation@var:bu_token }
			if = { # A sector the controller hasn't registered yet, such as a conquered state's, starts where every sector does
				limit = { check_variable = { i_employment_target = 0 } }
				set_temp_variable = { i_employment_target = global.sector_utilisation_start }
			}
			if = {
				limit = { NOT = { check_variable = { margin@var:bu_token = 0 } } } # 0 until its first month's accounts
				set_temp_variable = { local_shift_floor_temp = { value = global.local_utilisation_shift multiply = -1 } }
				set_temp_variable = {
					i_employment_target = {
						value = margin@var:bu_token
						subtract = controller:sector_margin@var:bu_token
						multiply = global.local_utilisation_gain
						min = global.local_utilisation_shift
						max = local_shift_floor_temp
						add = i_employment_target
						clamp = { min = global.sector_utilisation_min max = 100 }
					}
				}
			}
		}
		else = {
			set_temp_variable = { i_employment_target = 100 }
		}
	}
	sectors_update = {
		# Country scope. Each sector's national figures for the month, from its state-industries (industries_update):
		#   sector_size@, sector_generic_size@ (less what companies hold)
		#   sector_working_jobs@, sector_middle_jobs@, sector_working_employed@, sector_middle_employed@ (K workers)
		#   sector_revenue@, sector_inputs@, sector_payroll@, sector_maintenance@
		#   sector_profit@ (operating profit) and sector_margin@ (profit / revenue)
		#   sector_transport_demand@ - the industrial capacity its industries use (industrial_capacity_use@)
		#   Railways and ports: sector_industrial_capacity@, sector_shipping_capacity@, and sector_transport_revenue@ (what
		#     they're paid for transport at each state's rate, part of sector_revenue@)
		#   sector_construction@ (size being built), sector_retooling@ (size retooling)
		#   RGOs: sector_known_reserves@, sector_potential_reserves@
		# Profit pays corporate tax when it's positive (Q86), then the sector keeps global.sector_retained_share of what's left,
		# scaled by investment appetite (incentives, Q39), and pays the rest as dividends. A loss comes out of the sector's
		# capital first, then out of dividends, which go negative (Q22). Sectors don't borrow until M11.
		# Each state's upper class is paid sector_dividend_per_size@ on the generic size there, next month (industries_update).
		# sector_states@<sector> lists its five biggest states by size, biggest first. The method mix is
		# national_market_states_update's (sector_pm_size@, sector_am_size@).
		# Outputs: sector_*@<sector>, and the country totals sectors_profit_tax, sectors_retained and sectors_dividends
		for_each_loop = {
			array = sectors
			value = s_token
			set_variable = { sector_size@var:s_token = 0 }
			set_variable = { sector_generic_size@var:s_token = 0 }
			set_variable = { sector_working_jobs@var:s_token = 0 }
			set_variable = { sector_middle_jobs@var:s_token = 0 }
			set_variable = { sector_working_employed@var:s_token = 0 }
			set_variable = { sector_middle_employed@var:s_token = 0 }
			set_variable = { sector_revenue@var:s_token = 0 }
			set_variable = { sector_inputs@var:s_token = 0 }
			set_variable = { sector_payroll@var:s_token = 0 }
			set_variable = { sector_maintenance@var:s_token = 0 }
			set_variable = { sector_transport_demand@var:s_token = 0 }
			set_variable = { sector_construction@var:s_token = 0 }
			set_variable = { sector_retooling@var:s_token = 0 }
			if = {
				limit = { check_variable = { global.building_type@var:s_token = 2 } }
				set_variable = { sector_industrial_capacity@var:s_token = 0 }
				set_variable = { sector_shipping_capacity@var:s_token = 0 }
				set_variable = { sector_transport_revenue@var:s_token = 0 }
			}
			if = {
				limit = { check_variable = { global.building_rgo@var:s_token > 0 } }
				set_variable = { sector_known_reserves@var:s_token = 0 }
				set_variable = { sector_potential_reserves@var:s_token = 0 }
			}
			clear_array = sector_states@var:s_token
		}
		set_temp_variable = { sectors_country_temp = THIS }
		every_controlled_state = {
			for_each_loop = {
				array = industries
				value = bu_token
				if = {
					limit = { is_sector = yes }
					if = {
						limit = { check_variable = { size@var:bu_token > 0 } }
						add_to_variable = { PREV.sector_size@var:bu_token = size@var:bu_token }
						add_to_variable = { PREV.sector_generic_size@var:bu_token = owner_sizes_local@var:bu_token }
						add_to_variable = { PREV.sector_working_jobs@var:bu_token = working_jobs@var:bu_token }
						add_to_variable = { PREV.sector_middle_jobs@var:bu_token = middle_jobs@var:bu_token }
						add_to_variable = { PREV.sector_working_employed@var:bu_token = working_employed@var:bu_token }
						add_to_variable = { PREV.sector_middle_employed@var:bu_token = middle_employed@var:bu_token }
						add_to_variable = { PREV.sector_revenue@var:bu_token = revenue@var:bu_token }
						add_to_variable = { PREV.sector_inputs@var:bu_token = expenses_input@var:bu_token }
						add_to_variable = { PREV.sector_payroll@var:bu_token = expenses_payroll@var:bu_token }
						add_to_variable = { PREV.sector_maintenance@var:bu_token = expenses_maintenance@var:bu_token }
						add_to_variable = { PREV.sector_transport_demand@var:bu_token = industrial_capacity_use@var:bu_token }
						if = { # Railways and ports, paid at this state's rate as in industry_update
							limit = { check_variable = { global.building_type@var:bu_token = 2 } }
							add_to_variable = { PREV.sector_industrial_capacity@var:bu_token = industrial_capacity@var:bu_token }
							add_to_variable = { PREV.sector_shipping_capacity@var:bu_token = shipping_capacity@var:bu_token }
							set_temp_variable = {
								s_transport_temp = {
									value = industrial_capacity@var:bu_token
									subtract = {
										value = shipping_capacity@var:bu_token
										multiply = global.port_transport_share
									}
									add = shipping_capacity@var:bu_token
									multiply = price_industrial_capacity
								}
							}
							add_to_variable = { PREV.sector_transport_revenue@var:bu_token = s_transport_temp }
						}
						if = {
							limit = { NOT = { check_variable = { retooling_pm@var:bu_token = 0 } } }
							add_to_variable = { PREV.sector_retooling@var:bu_token = size@var:bu_token }
						}

						# Biggest states, sorted
						set_temp_variable = { s_break_temp = 0 }
						for_each_loop = {
							array = sectors_country_temp:sector_states@var:bu_token
							index = s_i value = s_state break = s_break_temp
							if = { limit = { check_variable = { s_state:size@var:bu_token < size@var:bu_token } }
								add_to_array = { array = sectors_country_temp:sector_states@var:bu_token value = THIS index = s_i }
								set_temp_variable = { s_break_temp = 1 }
							}
						}
						if = { limit = { check_variable = { s_break_temp = 0 } } add_to_array = { sectors_country_temp:sector_states@var:bu_token = THIS } }
						if = { limit = { check_variable = { sectors_country_temp:sector_states@var:bu_token^num > 5 } }
							remove_from_array = { array = sectors_country_temp:sector_states@var:bu_token index = 5 }
						}
					}

					# Construction counts before the first size is built, and reserves whether or not a deposit is worked
					if = {
						limit = { NOT = { check_variable = { construction@var:bu_token = 0 } } }
						add_to_variable = { PREV.sector_construction@var:bu_token = construction@var:bu_token }
					}
					if = {
						limit = { check_variable = { global.building_rgo@var:bu_token > 0 } }
						add_to_variable = { PREV.sector_known_reserves@var:bu_token = known_reserves@var:bu_token }
						add_to_variable = { PREV.sector_potential_reserves@var:bu_token = potential_reserves@var:bu_token }
					}
				}
			}
		}

		set_temp_variable = { profit_tax_rate_temp = { value = corporate_tax_percentage_actual divide = 100 } }
		set_temp_variable = { incentives_temp = modifier@incentives_modifier }
		calculate_investment_appetite = yes
		set_temp_variable = { retained_share_temp = { value = global.sector_retained_share multiply = investment_appetite_temp min = 0.8 } }

		set_variable = { sectors_profit_tax = 0 }
		set_variable = { sectors_retained = 0 }
		set_variable = { sectors_dividends = 0 }
		for_each_loop = {
			array = sectors
			value = s_token

			set_variable = {
				sector_profit@var:s_token = {
					value = sector_revenue@var:s_token
					subtract = sector_inputs@var:s_token
					subtract = sector_payroll@var:s_token
					subtract = sector_maintenance@var:s_token
				}
			}
			set_variable = { sector_margin@var:s_token = 0 }
			if = {
				limit = { check_variable = { sector_revenue@var:s_token > 0 } }
				set_variable = { sector_margin@var:s_token = { value = sector_profit@var:s_token divide = sector_revenue@var:s_token } }
			}

			set_variable = { sector_profit_tax@var:s_token = { value = sector_profit@var:s_token max = 0 multiply = profit_tax_rate_temp } }
			set_temp_variable = { s_after_tax_temp = { value = sector_profit@var:s_token subtract = sector_profit_tax@var:s_token } }
			if = {
				limit = { check_variable = { s_after_tax_temp > 0 } }
				set_variable = { sector_retained@var:s_token = { value = s_after_tax_temp multiply = retained_share_temp } }
			}
			else = { # As much of the loss as capital covers. The rest comes out of dividends
				set_variable = { sector_retained@var:s_token = { value = sector_capital@var:s_token multiply = -1 max = s_after_tax_temp } }
			}
			set_variable = { sector_dividends@var:s_token = { value = s_after_tax_temp subtract = sector_retained@var:s_token } }
			set_variable = { sector_dividend_per_size@var:s_token = 0 }
			if = {
				limit = { check_variable = { sector_size@var:s_token > 0 } }
				set_variable = { sector_dividend_per_size@var:s_token = { value = sector_dividends@var:s_token divide = sector_size@var:s_token } }
			}

			add_to_variable = { sectors_profit_tax = sector_profit_tax@var:s_token }
			add_to_variable = { sectors_retained = sector_retained@var:s_token }
			add_to_variable = { sectors_dividends = sector_dividends@var:s_token }
		}
	}

	industry_update_size_jobs = {
		# Calculates Industry Size & Job Figures
		set_temp_variable = { production_method_temp = production_method@var:bu_token }
		set_temp_variable = { automation_method_temp = automation_method@var:bu_token }

		# Update Industry Size
		if = { # RGOs
			limit = {
				check_variable = { global.building_rgo@var:bu_token > 0 }
			}
			set_temp_variable = { size_temp = base_resource@var:bu_token }
			set_variable = { size@var:bu_token = size_temp }
		}
		else = {
			set_temp_variable = { size_temp = size@var:bu_token }
		}

		# Jobs
		set_temp_variable = { base_jobs_temp = { value = global.building_base_employees@var:bu_token multiply = size_temp } }

		# Share of base jobs per tier: the PM's share plus the AM's delta, each scaled by its employment modifier
		set_temp_variable = { pm_employment_temp = { value = 1 add = controller:pm_employment_factor@var:production_method_temp } }
		set_temp_variable = { am_employment_temp = { value = 1 add = controller:pm_employment_factor@var:automation_method_temp } }
		set_temp_variable = {
			i_working_jobs = {
				value = global.pm_workforce_working@var:production_method_temp
				multiply = pm_employment_temp
				add = {
					value = global.pm_workforce_working@var:automation_method_temp
					multiply = am_employment_temp
				}
				max = 0
			}
		}
		set_temp_variable = {
			i_middle_jobs = {
				value = global.pm_workforce_middle@var:production_method_temp
				multiply = pm_employment_temp
				add = {
					value = global.pm_workforce_middle@var:automation_method_temp
					multiply = am_employment_temp
				}
				max = 0
			}
		}

		# Payroll Multiplier
		set_variable = {
			payroll_multiplier@var:bu_token = {
				value = i_working_jobs
				multiply = global.wage_multiplier@token:working_class
				add = {
					value = i_middle_jobs
					multiply = global.wage_multiplier@token:middle_class
				}
				multiply = global.building_base_employees@var:bu_token
			}
		}

		multiply_temp_variable = { i_working_jobs = base_jobs_temp }
		multiply_temp_variable = { i_middle_jobs = base_jobs_temp }

		# Jobs Available (employment_update adds them up for the state)
		industry_utilisation_calc = yes
		set_temp_variable = { i_working_jobs_available = { value = i_working_jobs multiply = i_employment_target divide = 100 } }
		set_temp_variable = { i_middle_jobs_available = { value = i_middle_jobs multiply = i_employment_target divide = 100 } }

		# Decaching. Owner places are about 2% of jobs and go to the upper class (Q11)
		set_variable = { owner_jobs@var:bu_token = { value = base_jobs_temp multiply = 0.02 multiply = i_employment_target divide = 100 } }

		set_variable = { working_jobs@var:bu_token = i_working_jobs }
		set_variable = { middle_jobs@var:bu_token = i_middle_jobs }
	}

	industry_update_employment = {
		# Employment
		industry_utilisation_calc = yes

		set_variable = {
			working_employed@var:bu_token = {
				value = 0
				if = { limit = { value = working_jobs@var:bu_token greater_than = 0 }
					add = working_jobs@var:bu_token
					multiply = i_employment_target
					divide = 100
					multiply = working_job_fulfillment # Slaves and free workers together
					divide = 100
				}
			}
		}

		set_variable = {
			middle_employed@var:bu_token = {
				value = 0
				if = { limit = { value = middle_jobs@var:bu_token greater_than = 0 }
					add = middle_jobs@var:bu_token
					multiply = i_employment_target
					divide = 100
					multiply = job_fulfillment@token:middle_class
					divide = 100
				}
			}
		}

		# Production Percentage, from the share of its payroll it employs
		set_temp_variable = {
			total_prod_temp = {
				value = working_employed@var:bu_token
				multiply = global.wage_multiplier@token:working_class
				add = {
					value = middle_employed@var:bu_token
					multiply = global.wage_multiplier@token:middle_class
				}
				divide = payroll_multiplier@var:bu_token
				divide = size@var:bu_token

			}

		set_variable = {
			production_percentage@var:bu_token = {
				value = total_prod_temp
				multiply = {
					value = 120
					subtract = {
						value = total_prod_temp
						pow = 3
						multiply = 20
					}
				}
			}
		}

	industry_production_update = {
		# Needs bu_token and i_size

		# Core production methods
		set_temp_variable = { production_method_temp = production_method@var:bu_token }
		set_temp_variable = { automation_method_temp = automation_method@var:bu_token }
		set_temp_variable = { i_base_employees = global.building_base_employees@var:bu_token }

		# Modifiers shared by outputs and inputs
		set_temp_variable = {
			i_common_modifier = {
				value = 1
				add = controller:industry_throughput_factor@var:bu_token
				multiply = { # Economy of scale
					value = i_size
					subtract = 1
					clamp = { min = 0 max = global.buildings_economy_of_scale_max }
					multiply = global.buildings_economy_of_scale_factor
					add = 1
				}
				multiply = {
					value = 1
					add = controller:pm_throughput_factor@var:production_method_temp
					add = controller:pm_throughput_factor@var:automation_method_temp
				}
			}
		}
		set_temp_variable = { i_output_modifier = { value = 1 add = controller:industry_output_factor@var:bu_token multiply = i_common_modifier } }
		set_temp_variable = { i_input_modifier = { value = 1 add = controller:industry_input_factor@var:bu_token multiply = i_common_modifier } }

		# Final Production Modifier
		set_temp_variable = { i_final_prod_modifier = { value = i_size multiply = production_percentage@var:bu_token divide = 100 } }

		# Transport demand. Measured before Infrastructure Efficiency, so that efficiency can't feed back into its own demand
		set_variable = {
			industrial_capacity_use@var:bu_token = {
				value = global.pm_industrial_capacity_use@var:production_method_temp
				add = global.pm_industrial_capacity_use@var:automation_method_temp
				multiply = i_base_employees
				multiply = i_input_modifier
				multiply = i_final_prod_modifier
			}
		}

		# Infrastructure Efficiency scales goods throughput, never employment
		set_temp_variable = {
			i_infrastructure_modifier = {
				value = infrastructure_efficiency
				subtract = 1
				multiply = global.building_infrastructure_dependence@var:bu_token
				add = 1
			}
		}
		multiply_temp_variable = { i_output_modifier = i_infrastructure_modifier }
		multiply_temp_variable = { i_input_modifier = i_infrastructure_modifier }

		# Transport & Shipping Supply
		if = {
			limit = { check_variable = { global.building_type@var:bu_token = 2 } }
			set_temp_variable = {
				i_shipping_max_temp = {
					value = global.pm_shipping_capacity@var:production_method_temp
					add = global.pm_shipping_capacity@var:automation_method_temp
					multiply = i_base_employees
					multiply = i_output_modifier
				}
			}
			set_variable = { shipping_capacity@var:bu_token = { value = i_shipping_max_temp multiply = i_final_prod_modifier } }

			# Ports also move goods within the state
			set_variable = {
				industrial_capacity@var:bu_token = {
					value = global.pm_industrial_capacity@var:production_method_temp
					add = global.pm_industrial_capacity@var:automation_method_temp
					multiply = i_base_employees
					multiply = i_output_modifier
					add = {
						value = i_shipping_max_temp
						multiply = global.port_transport_share
					}
					multiply = i_final_prod_modifier
				}
			}
		}

		# Output: bundles of the industry's base production (global.building_base_production_amount@) made this month. It makes
		# output@ × each good's amount, which the country adds up by sector (national_market_states_update). Its value comes
		# from the PM/AM cache in industry_update
		set_variable = {
			output@var:bu_token = {
				value = global.pm_output_amount@var:production_method_temp
				multiply = global.building_production_multiplier@var:bu_token
				multiply = i_base_employees
				multiply = i_output_modifier
				multiply = i_final_prod_modifier
			}
		}

		# Inputs: K of base workforce at work, in input terms. It uses input_scale@ × each of its PM's and AM's input amounts
		# (global.pm_input_amount@), which the country adds up by method. Their cost comes from the PM/AM cache in industry_update
		set_variable = { input_scale@var:bu_token = { value = i_base_employees multiply = i_input_modifier multiply = i_final_prod_modifier } }

		# Maintenance - flat upkeep from what the plant cost to build, paid on idle capacity too.
		# Depends on size ONLY (not production/employment %), scaled by the national regulatory_burden_factor.
		set_variable = {
			expenses_maintenance@var:bu_token = {
				value = global.pm_construction_value@var:production_method_temp
				add = global.pm_construction_value@var:automation_method_temp
				multiply = global.maintenance_rate
				multiply = i_base_employees
				multiply = i_size
				multiply = {
					value = 1
					add = controller:modifier@regulatory_burden_factor
				}
			}
		}
	}
	industry_update = {

		#log = "[?bu_token.GetTokenLocalizedKey]"

		cache_register_industry = yes

		# Finances. A state-industry stores only what the sector accounts, the state GDP and its trailing margin need (revenue@,

		if = {
			limit = { NOT = { check_variable = { global.building_government_control@var:bu_token = 1 } } }

			# Goods revenue and input costs from the PM/AM cache: the market's values for what it made and used. This country's

			set_variable = { revenue@var:bu_token = { value = controller:market_output_value@var:bu_token multiply = output@var:bu_token } }
			set_variable = {

				# A port's industrial capacity includes the share of its shipping that moves goods within the state, paid once
				limit = { check_variable = { global.building_type@var:bu_token = 2 } }
				set_variable = {
					revenue@var:bu_token = {
						value = industrial_capacity@var:bu_token
						subtract = {
							value = shipping_capacity@var:bu_token
							multiply = global.port_transport_share
						}
						add = shipping_capacity@var:bu_token
						multiply = price_industrial_capacity
						add = revenue@var:bu_token
					}
				}
			}

			# Expenses. Wages for the free share of the working class and for the middle class: slaves cost nothing (Q12, Q62)
			set_variable = {
				expenses_payroll@var:bu_token = {
					value = 1

					multiply = working_employed@var:bu_token

					add = {
						value = middle_employed@var:bu_token
						multiply = global.wage_multiplier@token:middle_class
					}
					multiply = base_wage
				}
			}
		}

	industry_asset_value_calc = {

		# (size, PM and AM, from the PM/AM cache) scaled by its trailing margin. A profitable plant carries a goodwill premium, a

	# PM/AM cache (Q55). Layers, from the top:
	#   Global: static PM and AM data (global.pm_*).
	#   Market: market_cache_update, at the market's prices and before modifiers. On every country, copied from its market leader:
	#     market_output_value@<sector> - the value of one bundle of the sector's base production (global.building_base_production_amount@).
	#       A state-industry's output@ counts the bundles it makes
	#     method_input_cost@<method> - what a PM's or AM's inputs cost, per K of base workforce
	#     method_construction_cost@<method> - what a PM's or AM's construction list costs, per K of base workforce
	#   Country: the country's own sector and method modifiers, industry_*_factor@<sector> and pm_*_factor@<method>
	#     (r_update_modifiers_country_effects, update_pm_modifier_factors). Output value and input cost stay separate, since some
	#     modifiers scale only one of them, and a state-industry combines its PM's and AM's with them when it reads them.
	#   State: local wage, Infrastructure Efficiency, economy of scale (industry_production_update).
	# A country lists the sectors and methods its state-industries use (sectors, methods_in_use), and its market works out only those.
	cache_register_industry = {
		# State scope. Input: bu_token
		# Puts the industry's sector and methods on the controller's lists, working out their values the first time
		set_temp_variable = { cache_token = bu_token }
		if = {
			limit = { NOT = { is_in_array = { controller:sectors = cache_token } } }
			controller = {
				add_to_array = { sectors = cache_token }
				cache_sector_calc = yes
				if = {
					limit = { check_variable = { sector_utilisation@var:cache_token = 0 } }
					set_variable = { sector_utilisation@var:cache_token = global.sector_utilisation_start }
				}
			}
		}
		set_temp_variable = { cache_token = production_method@var:bu_token }
		if = {
			limit = {
				NOT = { check_variable = { cache_token = 0 } }
				NOT = { is_in_array = { controller:methods_in_use = cache_token } }
			}
			controller = {
				add_to_array = { methods_in_use = cache_token }
				cache_method_calc = yes
			}
		}
		set_temp_variable = { cache_token = automation_method@var:bu_token }
		if = {
			limit = {
				NOT = { check_variable = { cache_token = 0 } }
				NOT = { is_in_array = { controller:methods_in_use = cache_token } }
			}
			controller = {
				add_to_array = { methods_in_use = cache_token }
				cache_method_calc = yes
			}
		}
	}
	cache_sector_calc = {

		# Output: market_output_value@<sector>, the value of one bundle of its base production at this country's prices

		for_each_loop = {

		}
	}

		}

								get_pm_values = yes

	base_wage_calculation = {
		# The startup wage when neither the state's history nor its controller's sets one (base_wage_seed): what peasants make
		# on the land, at least 1, times wage_factor. No state on the 1857 map needs it, since every set_tech_* sets a default.
		# It used to weight in the most productive industries, but state-industries no longer store their productivity (M04)
		set_temp_variable = { subs_farm_productivity = 1 }
		if = {
			limit = {
				check_variable = { productivity@token:subsistence_farm > subs_farm_productivity }
			}
			set_temp_variable = { subs_farm_productivity = productivity@token:subsistence_farm }
		}
		set_variable = { base_wage = subs_farm_productivity }
		multiply_variable = { base_wage = wage_factor }
	}
	base_wage_seed = {
		# State scope. Seeds the persistent base_wage ONCE at startup (called from _init_state / _state_fin).
		# Priority: state-history base_wage > controller (country history / set_tech_* default) > productivity formula.
		# History runs before on_startup, so an authored base_wage is already present here.
		if = {
			limit = { NOT = { has_variable = base_wage } }
			if = {
				limit = { controller = { has_variable = base_wage } }
				set_variable = { base_wage = controller:base_wage }
			}
			else = {
				base_wage_calculation = yes # Subsistence fallback
			}
		}

		base_wage_finalize = yes
	}
	base_wage_update = {
		# State scope. Monthly floating wage: base_wage *= (1 + labor_pressure), clamped to +/-wage_change_cap.
		# labor_pressure is piecewise on labor_utilization (jobs/workforce): rises >high, sticky-falls <low, flat between.
		# Uses LAST tick's labor_utilization (employment_update runs later in the monthly flow) - deliberate one-tick lag.
		# (Inflation pressure to be added here once the money-supply system provides a real inflation rate.)
		if = {
			limit = { check_variable = { industrial_workforce > 0 } }

			# The multiply arg builds the change from 0 (the two util branches are mutually exclusive since
			# high > low), clamps it, then +1 -> the growth factor applied to base_wage.
			set_variable = { base_wage = {
				value = base_wage
				multiply = {
					value = 0
					if = {
						limit = { value = labor_utilization  greater_than = global.wage_labor_util_high }
						add = labor_utilization
						subtract = global.wage_labor_util_high
						multiply = global.wage_labor_gain_up
					}
					if = {
						limit = { value = labor_utilization  less_than = global.wage_labor_util_low }
						add = labor_utilization
						subtract = global.wage_labor_util_low
						multiply = global.wage_labor_gain_down
					}
					clamp = { min = { value = global.wage_change_cap  multiply = -1 }  max = global.wage_change_cap }
					add = 1
				}
			} }
		}

		base_wage_finalize = yes
	}
	base_wage_finalize = {
		# State scope. Applies the wage floor. Shared by seed + update.
		# Floor = subsistence_productivity / 2, at least global.minimum_base_wage. The working class earns 2x base, so it floors at >= subsistence.
		set_temp_variable = { wage_floor_temp = {
			value = productivity@token:subsistence_farm
			divide = 2
			max = global.minimum_base_wage
		} }
		if = {
			limit = { check_variable = { base_wage < wage_floor_temp } }
			set_variable = { base_wage = wage_floor_temp }
		}
	}

		}

		# Employment target & ownership follow the active-size change (deposit_temp), so add_rgo_internal is
		# self-sufficient for RGOs (no add_x_industry_internal needed). Mirrors add_x_industry_internal.
		if = {
			limit = { NOT = { check_variable = { deposit_temp = 0 } } }
			set_temp_variable = { rgo_new_size_temp = rgo_old_size_temp }
			add_to_temp_variable = { rgo_new_size_temp = deposit_temp }

			if = { limit = { check_variable = { owner_sizes_local@var:r_token < 0 } } set_variable = { owner_sizes_local@var:r_token = 0 } }

			# Corporation-financed change: attribute the size delta to the funding company (same as add_x_industry_internal).
			if = {
				limit = {
					check_variable = { financer_temp > 0 }
					NOT = { check_variable = { financer_temp > 10500 } }
				}
				set_temp_variable = { company_temp = financer_temp }
				set_temp_variable = { holding_state_temp = THIS }
				set_temp_variable = { holding_building_temp = r_token }

					limit = { check_variable = { deposit_temp > 0 } }

					company_add_holding = yes
				}
				else = {
					set_temp_variable = { holding_size_temp = deposit_temp }
					multiply_temp_variable = { holding_size_temp = -1 }
					company_remove_holding = yes
				}
			}
		}

		# Update prospecting - do this regardless b/c changing any of the above will require this
		update_prospecting = yes

		set_temp_variable = { bu_token = r_token }
		industry_update_size_jobs = yes
		industry_update = yes
	}

		# r_token
		# Takes some potential deposits and makes them known
		set_temp_variable = { resource_amount_temp = discovery_high@var:r_token }
		subtract_from_temp_variable = { resource_amount_temp = discovery_low@var:r_token }
		multiply_temp_variable = { resource_amount_temp = random }
		round_temp_variable = resource_amount_temp
		
		add_to_variable = { known_reserves@var:r_token = resource_amount_temp }
		subtract_from_variable = { potential_reserves@var:r_token = resource_amount_temp }
		set_variable = { prospecting@var:r_token = 0 }

		set_temp_variable = { known_temp = resource_amount_temp }
		set_temp_variable = { potential_temp = resource_amount_temp }
		multiply_temp_variable = { potential_temp = -1 }
		set_temp_variable = { deposit_temp = 0 } # discovery only moves potential->known; no active-size change
		add_rgo_internal = yes
	}
	rgo_major_discovery = {
		# r_token
		# Adds a random amount of known & potential deposits
		set_variable = { major_discovery_resource = r_token }

		set_variable = { major_discovery_known = discovery_high@var:r_token }

		set_temp_variable = { random_temp = random }
		multiply_temp_variable = { random_temp = 100 }
		add_to_temp_variable = { random_temp = 100 }
		set_variable = { major_discovery_potential = random_temp }
		divide_variable = { major_discovery_potential = global.base_price@var:r_token }

		round_variable = major_discovery_known
		round_variable = major_discovery_potential
		set_variable = { prospecting@var:r_token = 0 }
		
		add_to_variable = { major_discoveries@var:r_token = 1 }
		set_temp_variable = { known_temp = major_discovery_known }
		set_temp_variable = { potential_temp = major_discovery_potential }
		set_temp_variable = { deposit_temp = 0 } # discovery only adds known/potential; no active-size change
		add_rgo_internal = yes
		if = {
			limit = {
				controller = { 
					is_ai = no 
					NOT = { has_variable = major_discovery_target }
				}
			}
			controller = {
				set_variable = { major_discovery_target = PREV }
				country_event = { id = economy.100 }
			}
		}
		else = {
			clear_variable = major_discovery_known 
			clear_variable = major_discovery_potential 
			clear_variable = major_discovery_resource
		}
	}

	construct_x_industries_no_update = {
		# x_temp
		# financer_temp
		# industry_token
		# production_method_temp
		# automation_method_temp
		# no_cost_temp

		# State


		if = {
			limit = {
				check_variable = { x_temp > 0 }
				is_in_array = { global.industry_token_array = industry_token }
				if = {
					limit = {
						is_in_array = { global.building_token_array = industry_token }
						NOT = { is_in_array = { buildings = industry_token } }
					}
					check_variable = { buildings^num < 16 }
					check_variable = { buildings^num < modifier@local_industry_slots }
					is_in_array = { controller:building_unlocked_pms@var:industry_token = production_method_temp }
					is_in_array = { controller:building_unlocked_ams@var:industry_token = automation_method_temp }
				}
				if = {
					limit = {
						check_variable = { financer_temp > 10500 } #Government
					}
					country_exists = var:financer_temp

				}
				else_if = {
					limit = {
						check_variable = { financer_temp > 0 } #Corporation
					}
					has_variable = global.company_hq_state@var:financer_temp
				}
				else = { #Local

				}
			}

			#Financer
			if = {
				limit = {
					check_variable = { construction@var:industry_token = 0 }
				}
				set_variable = { construction_financier@var:industry_token = financer_temp }
			}
			else = {
				set_temp_variable = { financer_temp = construction_financier@var:industry_token }
			}

			add_to_variable = { construction@var:industry_token = x_temp }

			#Add building to array if it doesn't exist yet
			if = {
				limit = {
					NOT = { is_in_array = { industries = industry_token } }
				}
				add_to_array = { industries = industry_token }
				set_variable = { production_method@var:industry_token = production_method_temp }
				set_variable = { automation_method@var:industry_token = automation_method_temp }
			}

			# Cost
			set_temp_variable = { iuc_industry = industry_token }
			industry_update_construction_retooling = yes

			set_temp_variable = { pm_temp = production_method@var:industry_token }
			set_temp_variable = { am_temp = automation_method@var:industry_token }
			if = {
				limit = {
					check_variable = { no_cost_temp = 0 }
					get_construction_cost = yes
				}
				set_temp_variable = { initial_cost_temp = total_construction_cost_temp }
				multiply_temp_variable = { initial_cost_temp = global.start_construction_cost_multiplier }
				if = {
					limit = {
						check_variable = { financer_temp > 10500 } #Government (country scope)
					}
					divide_temp_variable = { initial_cost_temp = financer_temp:modifier@government_investment_efficiency_factor }
					subtract_from_variable = { financer_temp:treasury = initial_cost_temp }
					var:financer_temp = {
						spending_update = yes
					}
				}
				else_if = {
					limit = {
						check_variable = { financer_temp > 0 } #Corporation (company id 1..10000)
					}
					subtract_from_variable = { global.company_treasury@var:financer_temp = initial_cost_temp }
				}
				else = { #Local (0; negative = legacy state ref, treated as local)
					subtract_from_variable = { controller:investment_fund = initial_cost_temp }
				}
				add_to_variable = { construction_investment@var:industry_token = initial_cost_temp }
			}
			refresh_construction_list = yes
			reload_industries_array = yes
			add_to_variable = { update_state_econ_window = 1 }
		}
	}

	retool_industry_no_update = {
		# financer_temp
		# industry_token
		# production_method_temp - only one of these two
		# automation_method_temp - only one of these two
		# no_cost_temp

		# State

		if = {
			limit = {
				is_in_array = { global.industry_token_array = industry_token }
				is_in_array = { industries = industry_token }
				OR = {
					NOT = { check_variable = { production_method_temp = 0 } }
					NOT = { check_variable = { automation_method_temp = 0 } }
				}
				if = {
					limit = {
						NOT = { check_variable = { production_method_temp = 0 } }
					}
					is_in_array = { controller:building_unlocked_pms@var:industry_token = production_method_temp }
					check_variable = { automation_method_temp = 0 }
				}
				if = {
					limit = {
						NOT = { check_variable = { automation_method_temp = 0 } }
					}
					is_in_array = { controller:building_unlocked_ams@var:industry_token = automation_method_temp }
					check_variable = { production_method_temp = 0 }
				}
				if = {
					limit = {
						check_variable = { financer_temp > 0 } #Government
					}
					country_exists = var:financer_temp

				}
				else_if = {
					limit = {
						check_variable = { financer_temp < 0 } #Corporation
					}
				}
				else = { #Local

				}
			}

			#Financer
			if = {
				limit = {
					NOT = { check_variable = { retooling_pm@var:industry_token = 0 } }
				}
				set_variable = { retooling_financier@var:industry_token = financer_temp }
			}
			else = {
				set_temp_variable = { financer_temp = retooling_financier@var:industry_token }
			}

			if = {
				limit = {
					NOT = { check_variable = { production_method_temp = 0 } }
				}
				set_variable = { retooling_pm@var:industry_token = production_method_temp }
			}
			else = {
				set_variable = { retooling_pm@var:industry_token = automation_method_temp }
			}

			# Cost
			set_temp_variable = { iuc_industry = industry_token }
			industry_update_construction_retooling = yes

			set_temp_variable = { pm_temp = retooling_pm@var:industry_token }
			if = {
				limit = {
					check_variable = { no_cost_temp = 0 }
					get_retooling_cost = yes
				}
				set_temp_variable = { initial_cost_temp = total_retooling_cost_temp }
				multiply_temp_variable = { initial_cost_temp = global.start_construction_cost_multiplier }
				multiply_temp_variable = { initial_cost_temp = size@var:industry_token }
				if = {
					limit = {
						check_variable = { financer_temp < 0 } #Corporation
					}
				}
				else_if = {
					limit = {
						check_variable = { financer_temp > 0 } #Government
					}
					divide_temp_variable = { initial_cost_temp = financer_temp:modifier@government_investment_efficiency_factor }
					subtract_from_variable = { financer_temp:treasury = initial_cost_temp }
				}
				else = { #Local
					subtract_from_variable = { controller:investment_fund = initial_cost_temp }
				}
				add_to_variable = { retooling_investment@var:industry_token = initial_cost_temp }
			}
			add_to_variable = { update_state_econ_window = 1 }
		}
	}
