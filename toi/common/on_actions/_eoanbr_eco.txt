on_actions = {
	on_startup = {
		effect = {
			UCT = {
				_reset = yes # Economy Reset

				# Get things going
				# r_weekly_trade = yes
				# _advance_time = yes
				# r_weekly_trade = yes
				# _advance_time = yes 
				# _advance_time = yes 
			}
		}
	}
	on_monthly_UCT = { # Economy monthly-tick
		effect = {
			_advance_time = yes
		}
	}
	on_weekly_UCT = { # Trade tick
		effect = {
			r_weekly_trade = yes
		}
	}


	on_declare_war = { # On Civil War
		effect = {
			if = {
				limit = {
					ROOT = { original_tag = FROM }
				}
				FROM = {
					ROOT = {
						eoanbr_institutions_inherit = yes
						eoanbr_technologies_inherit_hidden = yes
					}
				}
				ROOT = {
					if = {
						limit = {
							NOT = { has_dynamic_modifier = resource_trade }
						}
					}
					initialize_country = yes
				}

				if = {
					limit = {
						FROM = {
							#check for US guarantee
							is_guaranteed_by = USA
							#capital is in north or south america
							capital_scope = {
								OR = {
									is_on_continent = north_america
									is_on_continent = south_america
								}
							}
						}
					}
					USA = {
						give_guarantee = ROOT
					}
				}
			}
		}
	}
}
