					set_country_leader_ideology = christian
				}
				else_if = {
					limit = {
						check_variable = { global.family@var:religion_temp = token:religion_family_muslim }
					}
					set_country_leader_ideology = islamist
				}
				else_if = {
					limit = {
						check_variable = { global.family@var:religion_temp = token:religion_family_buddhist }
					}
					set_country_leader_ideology = shogunate
				}
				else_if = {
					limit = {
						check_variable = { global.family@var:religion_temp = token:religion_family_pagan }
					}
					set_country_leader_ideology = shamanism
				}
				else = {
					set_country_leader_ideology = other_ideology
				}
			}
		}
	}
